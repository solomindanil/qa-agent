const log = { info: vi.fn(), warn: vi.fn(), error: vi.fn() } as never;

describe('payment checkout provisioning orchestrator', () => {
  beforeEach(() => {
    mocks.provisionPaymentProduct.mockReset();
    mocks.inspectPaymentProductCheckout.mockReset();
    mocks.inspectPaymentProductCheckout.mockResolvedValue({
      settlementState: 'absent',
      operationId: null,
      feeOperationId: null,
      entityTable: null,
      entityId: null,
    });
    mocks.inspectProviderCheckout.mockReset();
    mocks.inspectProviderCheckout.mockResolvedValue({
      state: 'pending',
      providerStatus: 'WAITING',
      paymentAmountMinor: null,
      paymentCurrency: null,
    });
    mocks.findPaymentProviderIdentity.mockReset();
    mocks.registerPaymentProviderIdentity.mockReset();
    mocks.claimExternalMoneyEvent.mockReset();
    mocks.completeExternalMoneyEvent.mockReset();
    mocks.failExternalMoneyEvent.mockReset();
    mocks.createMoneyOperation.mockReset();
    mocks.prepareFailedStagingVpnActivationRecovery.mockReset();
    mocks.prepareFailedStagingVpnActivationRecovery.mockResolvedValue({ state: 'not_applicable' });
    mocks.capturePaymentCheckoutTransition.mockReset();
    mocks.capturePaymentCheckoutTransition.mockResolvedValue(undefined);
    mocks.tableUpdates.length = 0;
    mocks.rpc.mockReset();
    mocks.rpc.mockImplementation(async (name: string, args: Record<string, unknown>) => {
      if (name === 'record_payment_checkout_confirmation') {
        mocks.currentRow = {
          ...mocks.currentRow,
          provider_ref: args.p_provider_contract_id,
          provider_status: args.p_provider_status,
          confirmed_payment_amount_minor: args.p_payment_amount_minor ?? mocks.currentRow?.confirmed_payment_amount_minor ?? null,
          confirmed_payment_currency: args.p_payment_currency ?? mocks.currentRow?.confirmed_payment_currency ?? null,
          confirmed_payment_at: args.p_payment_amount_minor && args.p_payment_currency
            ? (mocks.currentRow?.confirmed_payment_at ?? args.p_confirmed_at)
            : (mocks.currentRow?.confirmed_payment_at ?? null),
          paid_at: mocks.currentRow?.paid_at ?? args.p_confirmed_at,
        };
        return { data: mocks.currentRow, error: null };
      }
      if (name === 'attach_payment_checkout_reversal_loss') {
        return { data: mocks.reversalIncident, error: null };
      }
      if (name === 'apply_payment_checkout_reversal') {
        mocks.currentRow = {
          ...mocks.currentRow,
          status: 'reversed',
          failure_code: 'PAYMENT_REVERSED',
          failure_message: 'Provider reported a payment reversal',
        };
        return { data: mocks.currentRow, error: null };
      }
      if (name === 'apply_payment_checkout_provider_refund_signal') {
        mocks.currentRow = {
          ...mocks.currentRow,
          status: 'manual_review',
          failure_code: 'PAYMENT_PROVIDER_REFUND_OBSERVED',
          failure_message: 'Provider reported a voluntary refund; operator reconciliation is required',
        };
        return { data: mocks.currentRow, error: null };
      }
      if (name === 'start_payment_checkout_provisioning_attempt') {
        return { data: true, error: null };
      }
      if (name === 'complete_payment_checkout_provisioning_attempt') {
        return { data: mocks.currentRow, error: null };
      }
      return { data: null, error: null };
    });
    mocks.currentRow = checkoutRow();
    mocks.reversalIncident = null;
  });

  it('persists a late DELIVERED callback as a review exception after confirmed-unpaid resolution', async () => {
    const resolvedRow = {
      ...checkoutRow(),
      status: 'payment_failed',
      provider: 'payassist',
      provider_ref: 'ce7d0ae4-7011-4085-aa7b-7cbf9ce8e39c',
      provider_status: 'FAILED',
      payment_amount_minor: '39732',
      payment_currency: 'RUB',
      paid_at: null,
      confirmed_payment_amount_minor: null,
      confirmed_payment_currency: null,
      confirmed_payment_at: null,
      metadata: {
        financial_resolution: {
          kind: 'confirmed_unpaid',
          evidence_digest: 'b'.repeat(64),
        },
      },
    };
    mocks.currentRow = resolvedRow;
    mocks.findPaymentProviderIdentity.mockResolvedValue({
      provider: 'payassist',
      providerContractId: resolvedRow.provider_ref,
      ownerTable: 'payment_checkouts',
      ownerId: resolvedRow.id,
      authUserId: resolvedRow.auth_user_id,
    });
    mocks.claimExternalMoneyEvent.mockResolvedValue({
      id: 'event-late-delivered', duplicate: false, replay: false, inFlight: false, claimToken: 'late-claim',
    });

    const { processPaymentProviderSignal } = await import('/Users/danilsolomin/projectsnew/qa-agent/.local/freeland-pr430-review-20260920.LOurYb/product/apps/api/src/services/payment-checkouts');
    const result = await processPaymentProviderSignal({
      provider: 'payassist',
      providerContractId: resolvedRow.provider_ref,
      eventType: 'payment.success',
      status: 'DELIVERED',
      paymentAmountMinor: 39732n,
      paymentCurrency: 'RUB',
      source: 'webhook',
      log,
    });

    expect(result.checkout).toMatchObject({
      status: 'manual_review',
      failureCode: 'PAYMENT_PAID_AFTER_CONFIRMED_UNPAID_RESOLUTION',
    });
    expect(mocks.currentRow).toMatchObject({
      metadata: {
        financial_resolution_exception: {
          kind: 'late_valid_payment',
          provider_event_id: 'event-late-delivered',
        },
      },
    });
    expect(mocks.provisionPaymentProduct).not.toHaveBeenCalled();
    expect(mocks.completeExternalMoneyEvent).toHaveBeenCalledWith(
      'event-late-delivered',
      null,
      'late-claim',
    );

    const beforeRepeat=JSON.stringify(mocks.currentRow);
    const rpcCount=mocks.rpc.mock.calls.length;
    const completeCount=mocks.completeExternalMoneyEvent.mock.calls.length;
    mocks.claimExternalMoneyEvent.mockResolvedValue({
      id:'event-late-delivered',duplicate:true,replay:false,inFlight:false,
    });
    const repeated=await processPaymentProviderSignal({
      provider:'payassist',providerContractId:resolvedRow.provider_ref,
      eventId:'event-late-delivered',eventType:'payment.success',status:'DELIVERED',
      paymentAmountMinor:39732n,paymentCurrency:'RUB',source:'webhook',log,
    });
    expect(repeated.duplicate).toBe(true);
    expect(repeated.checkout?.status).toBe('manual_review');
    expect(JSON.stringify(mocks.currentRow)).toBe(beforeRepeat);
    expect(mocks.rpc.mock.calls.length).toBe(rpcCount);
    expect(mocks.completeExternalMoneyEvent.mock.calls.length).toBe(completeCount);
    expect(mocks.provisionPaymentProduct).not.toHaveBeenCalled();
    expect(mocks.createMoneyOperation).not.toHaveBeenCalled();

  });

});
