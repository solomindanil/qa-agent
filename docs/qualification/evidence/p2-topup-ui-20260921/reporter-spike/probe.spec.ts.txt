test('early projection summary mismatch', async ({page}) => {
  await page.route('**/*', route => route.abort());
  await page.setContent(process.env.SPIKE_HTML!);
  await expect.poll(async () => {
    const values = await page.locator('p').allTextContents();
    return values.length === 1 && values[0] === 'expected';
  }, {timeout: 100, message: 'SUMMARY_VALUE_MISMATCH'}).toBe(true);
});
test('early projection ambiguous locator', async ({page}) => {
  await page.route('**/*', route => route.abort());
  await page.setContent(process.env.SPIKE_HTML!);
  await expect.poll(() => page.locator('p').count(), {timeout: 100, message: 'SUMMARY_CARDINALITY_MISMATCH'}).toBe(2);
});
