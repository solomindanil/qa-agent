import assert from 'node:assert/strict';
import { createHash } from 'node:crypto';
import { readFile } from 'node:fs/promises';
import { QaCampaignPlanV0Schema } from '/Users/danilsolomin/projectsnew/qa-agent/components/console/src/lib/qa-campaign-v0.ts';
import { digestCanonical } from '/Users/danilsolomin/projectsnew/qa-agent/components/console/src/lib/canonical-digest.ts';
import { readCampaignPlan, readCampaignWorkspace } from '/Users/danilsolomin/projectsnew/qa-agent/components/console/src/node/qa-campaign-files.ts';

const actorRoot = '/private/tmp/qa-api-consumer-20260922.iVqZ2s/actor';
const firstBytes = await readFile(`${actorRoot}/first-design.json`);
const rationaleBytes = await readFile(`${actorRoot}/first-design.md`);
const firstDigest = createHash('sha256').update(firstBytes).digest('hex');
const rationaleDigest = createHash('sha256').update(rationaleBytes).digest('hex');
assert.equal(firstDigest, 'e3ba08fe284e02b405991dd4ec14be599719536f8236b0caeb867c4241a7103c');
assert.equal(rationaleDigest, 'ffd7bfdffb68a9c3ad2691ec799eeebeb9485d3ea7b032f134100da9bfdca8c9');
const first = JSON.parse(firstBytes.toString('utf8'));
const manifests = JSON.parse(await readFile(`${actorRoot}/bound-plan-manifest.json`, 'utf8'));
for (const [index, manifest] of manifests.entries()) {
  const workspace = await readCampaignWorkspace(manifest.workspacePath);
  const plan = QaCampaignPlanV0Schema.parse(await readCampaignPlan(manifest.workspacePath, manifest.planPath));
  const proposed = JSON.parse(await readFile(`${actorRoot}/${manifest.releaseId}/first-proposed-plan.json`, 'utf8'));
  assert.equal(digestCanonical(plan), manifest.planDigest);
  assert.equal(digestCanonical(proposed), manifest.planDigest);
  assert.equal(workspace.graph.semanticDigest, manifest.graphDigest);
  assert.equal(workspace.catalog.semanticDigest, manifest.catalogDigest);
  assert.deepEqual(plan.checks.map(check => ({ request: { method: check.method, path: check.path }, assertions: check.assertions })),
    first.checks.map(check => ({ request: check.request, assertions: check.assertions })));
  const validationRoot = `${actorRoot}/invocations/${String(index + 8).padStart(3, '0')}-validate-${manifest.releaseId}`;
  const invocation = JSON.parse(await readFile(`${validationRoot}/invocation.json`, 'utf8'));
  const validation = JSON.parse(await readFile(`${validationRoot}/stdout.txt`, 'utf8'));
  const stderr = await readFile(`${validationRoot}/stderr.txt`, 'utf8');
  assert.equal(invocation.exitCode, 0);
  assert.equal(stderr, '');
  assert.equal(validation.ok, true);
  assert.equal(validation.command, 'validate');
  assert.equal(validation.readyToRun, false);
  assert.equal(validation.executableTargetIds.length, 2);
  assert.equal(validation.blockedTargetIds.length, 3);
  assert.deepEqual(new Set([...validation.executableTargetIds, ...validation.blockedTargetIds]),
    new Set(workspace.graph.nodes.filter(node => node.coverageTarget).map(node => node.id)));
  console.log(JSON.stringify({ releaseId: manifest.releaseId, planDigest: manifest.planDigest,
    checks: plan.checks.length, assertions: plan.checks.reduce((sum, check) => sum + check.assertions.length, 0),
    graphTargets: 5, catalogEntries: 3, blockers: plan.blockers.length, cliValidation: 'ok',
    validationAt: invocation.endedAt, unchangedFirstProposal: true, unchangedFirstAssertions: true }));
}
console.log(JSON.stringify({ initialDesignDigest: firstDigest, initialRationaleDigest: rationaleDigest,
  verifiedPlans: manifests.length, executionPerformed: false, state: 'awaiting-independent-review' }));
