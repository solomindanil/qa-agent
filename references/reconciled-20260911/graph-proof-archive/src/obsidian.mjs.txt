export function renderObsidianVault({ graph = {}, impact = {}, plan = {}, productionReference = null }) {
  const context = buildContext({ graph, impact, plan, productionReference });
  const files = new Map([
    ["INDEX.md", renderIndex(context)],
    ["SYSTEM.md", renderSystem(context)],
    ["MONEY.md", renderMoney(context)],
    ["SURFACES.md", renderSurfaces(context)],
    ["PROVIDERS.md", renderProviders(context)],
    ["IMPACT.md", renderImpact(context)],
    ["TEST-PLAN.md", renderTestPlan(context)],
    ["NODES.md", renderRegistry(context)],
    ["Freeland Product Map.canvas", renderCanvas()],
    [".obsidian/app.json", `${JSON.stringify({ showInlineTitle: true }, null, 2)}\n`],
    [".obsidian/graph.json", `${JSON.stringify(graphSettings(), null, 2)}\n`],
  ]);

  for (const node of context.semanticNodes) {
    files.set(notePath(node), renderNodeNote(node, context));
  }

  const managedMarkdown = [...files.keys()].filter(path => path.endsWith(".md")).sort();
  files.set(".freeland-generated.json", `${JSON.stringify({ schemaVersion: 1, managedMarkdown }, null, 2)}\n`);
  return new Map([...files].sort(([left], [right]) => left.localeCompare(right)));
}

function buildContext({ graph, impact, plan, productionReference }) {
  const nodes = [...(graph.nodes ?? [])].sort((left, right) => left.id.localeCompare(right.id));
  const nodeById = new Map(nodes.map(node => [node.id, node]));
  const semanticNodes = nodes.filter(node => SEMANTIC_KINDS.has(node.kind));
  const edges = [...(graph.edges ?? [])].sort(compareEdges);
  const byKind = countBy(nodes, node => node.kind);
  const locatorCounts = countBy(graph.sourceLocators ?? [], locator => locator.disposition);
  const unmappedFiles = [...(plan.unmappedFiles ?? impact.unmappedFiles ?? [])].sort((left, right) => filePath(left).localeCompare(filePath(right)));
  const impactedNodes = [...(impact.impactedNodes ?? [])].sort((left, right) => left.id.localeCompare(right.id));
  const mappedFileIds = new Set(impactedNodes.flatMap(node => (node.reasons ?? []).flatMap(reason =>
    reason.type === "impact" && reason.edgePath?.[0]?.startsWith("file:") ? [reason.edgePath[0]] : []
  )));
  const directlyAutomated = impact.automated?.length ?? directlyVerifying("automated_test", impactedNodes, edges, nodeById).length;
  const directlyManual = impact.manual?.length ?? directlyVerifying("manual_test", impactedNodes, edges, nodeById).length;
  const { semanticDigest: _semanticDigest, ...auditGraph } = graph;
  const strictCoverageIssues = validateGraph(auditGraph, { strictReleaseCoverage: true }).issues
    .filter(issue => STRICT_COVERAGE_CODES.has(issue.code));
  return {
    graph,
    impact,
    plan,
    nodes,
    nodeById,
    semanticNodes,
    edges,
    byKind,
    locatorCounts,
    unmappedFiles,
    impactedNodes,
    mappedChangedCount: mappedFileIds.size,
    directlyAutomated,
    directlyManual,
    strictCoverageIssues,
    strictCoverageCounts: countBy(strictCoverageIssues, issue => issue.code),
    sourceSha: graph.generatedFrom?.sourceSha ?? plan.sourceSha ?? "unknown",
    productMapReferenceSha: graph.generatedFrom?.productMapReferenceSha ?? "unknown",
    candidateSha: plan.candidateSha ?? graph.generatedFrom?.sourceSha ?? "unknown",
    baselineSha: plan.baselineSha ?? "unknown",
    graphDigest: graph.semanticDigest ?? plan.graphDigest ?? "unknown",
    productionObservation: productionReference ?? graph.observedDeployments?.production ?? null,
  };
}

function page({ title, body, tags = ["freeland", "product-graph"] }) {
  return [
    "---",
    `tags: [${tags.join(", ")}]`,
    "generated: true",
    "---",
    GENERATED,
    `# ${title}`,
    "",
    NAV,
    "",
    body.trim(),
    "",
    "---",
    NAV,
    "",
  ].join("\n");
}

function renderImpact(context) {
  const totalChanged = context.mappedChangedCount + context.unmappedFiles.length;
  const autoSelected = context.plan.automatedTests?.length ?? 0;
  const productCiSelected = context.plan.productCiTests?.length ?? 0;
  const manualSelected = context.plan.manualTests?.length ?? 0;
  const impacted = context.impactedNodes.map(item => {
    const node = context.nodeById.get(item.id);
    const links = node && SEMANTIC_KINDS.has(node.kind) ? nodeLink(node) : `\`${item.id}\``;
    const paths = (item.reasons ?? []).filter(reason => reason.type === "impact").map(reason =>
      (reason.edgePath ?? []).map(id => id.replace(/^file:/, "")).join(" → ")
    );
    return `- ${links}${paths.length ? `\n${paths.map(path => `  - \`${escapeCode(path)}\``).join("\n")}` : ""}`;
  }).join("\n");
  return page({
    title: "Freeland · влияние текущего релиза",
    tags: ["freeland", "product-graph", "release-impact"],
    body: `
> [!warning] Conservative selection
> mode: **${context.plan.mode ?? "unknown"}** · fallback: **${context.plan.fallback ?? "none"}** · source identity: **${context.plan.sourceIdentity ?? "source_only"}**

${renderProductionObservation(context)}

\`\`\`mermaid
flowchart LR
  DIFF["${totalChanged} changed files"] --> MAPPED["${context.mappedChangedCount} mapped files"]
  DIFF --> UNKNOWN["${context.unmappedFiles.length} unmapped files"]
  MAPPED --> NODES["${context.impactedNodes.length} impacted semantic nodes"]
  NODES --> DAUTO["${context.directlyAutomated} directly mapped automated tests"]
  NODES --> DMAN["${context.directlyManual} directly impacted manual cases"]
  UNKNOWN --> FALLBACK["Safe fallback: ${context.plan.fallback ?? "none"}"]
  FALLBACK --> AUTO["${autoSelected} staging automated tests"]
  FALLBACK --> PCI["${productCiSelected} candidate-bound Product CI tests"]
  FALLBACK --> MAN["${manualSelected} manual cases pending"]
\`\`\`

## Impacted nodes

${impacted || "- Пока impact-план не построен."}

## Why full fallback

Неизвестный changed file не получает придуманную связь. Пока зависимость не классифицирована, план расширяется до полного staging automation и manual safety net.

<details>
<summary>${context.unmappedFiles.length} unmapped files</summary>

${context.unmappedFiles.map(file => `- \`${escapeCode(filePath(file))}\` (${file.status ?? "?"})`).join("\n") || "- none"}

</details>
`,
  });
}

> release SHA: \`${escapeCode(observation.releaseSha ?? "unknown")}\` · public origin: \`${escapeCode(observation.publicOrigin ?? "unknown")}\` · profile digest: \`${escapeCode(observation.profileDigest ?? "unknown")}\` · public build digest: \`${escapeCode(observation.publicBuildDigest ?? "unknown")}\` · manifest ID: \`${escapeCode(observation.manifestId ?? "unknown")}\`
>
> workers: **${Array.isArray(observation.enabledWorkerFamilies) ? observation.enabledWorkerFamilies.length : 0}** · observed: **${escapeInline(observation.observedAt ?? "unknown")}** · assurance: **${escapeInline(observation.assurance ?? "unknown")}**`;
}

function renderNodeNote(node, context) {
  const incoming = context.edges.filter(edge => edge.to === node.id);
  const outgoing = context.edges.filter(edge => edge.from === node.id);
  const code = incoming.filter(edge => edge.type === "implements").map(edge => context.nodeById.get(edge.from)).filter(Boolean);
  const tests = incoming.filter(edge => ["verifies", "always_run"].includes(edge.type)).map(edge => ({ edge, node: context.nodeById.get(edge.from) })).filter(item => item.node);
  const semanticRelations = [...incoming.map(edge => ({ direction: "incoming", edge, other: context.nodeById.get(edge.from) })),
    ...outgoing.map(edge => ({ direction: "outgoing", edge, other: context.nodeById.get(edge.to) }))]
    .filter(item => item.other && SEMANTIC_KINDS.has(item.other.kind));
  const locators = (context.graph.sourceLocators ?? []).filter(locator => locator.ownerNodeId === node.id);
  const home = node.kind === "flow" ? "[[SYSTEM|Продукт]]" : node.kind === "route" ? "[[SURFACES|Поверхности]]" : "[[MONEY|Деньги и оплаты]]";
  return [
    "---",
    `id: ${JSON.stringify(node.id)}`,
    `kind: ${JSON.stringify(node.kind)}`,
    `graph_digest: ${JSON.stringify(context.graphDigest)}`,
    `source_sha: ${JSON.stringify(context.sourceSha)}`,
    "generated: true",
    "---",
    GENERATED,
    `# ${nodeTitle(node)}`,
    "",
    `${home} · [[IMPACT|Текущий impact]] · [[TEST-PLAN|План тестов]] · [[NODES|Все узлы]]`,
    "",
    "## Канонический узел",
    "",
    `\`${node.id}\``,
    "",
    nodeDescription(node),
    "",
    "## Реализуют",
    "",
    code.length ? code.map(source => `- \`${escapeCode(source.path ?? source.id.replace(/^file:/, ""))}\``).join("\n") : "- Прямая code-edge пока не заведена.",
    "",
    "## Проверяют",
    "",
    tests.length ? tests.map(({ edge, node: test }) => `- ${test.kind === "manual_test" ? "manual" : "automated"} \`${test.id}\` — ${escapeInline(test.title ?? test.selector?.titlePath?.join(" › ") ?? "")} (${edge.type})`).join("\n") : "- Прямая test-edge пока не заведена.",
    "",
    "## Связанные семантические узлы",
    "",
    semanticRelations.length ? semanticRelations.map(item => `- ${item.direction} \`${item.edge.type}\` ${nodeLink(item.other)}`).join("\n") : "- Канонических semantic-to-semantic рёбер пока нет.",
    "",
    "## Source locators",
    "",
    locators.length ? [...countBy(locators, locator => locator.disposition)].sort().map(([status, count]) => `- ${status}: ${count}`).join("\n") : "- Нет locators.",
    "",
  ].join("\n");
}

function renderCanvas() {
  const pages = [
    ["index", "INDEX.md", 0, 260],
    ["system", "SYSTEM.md", 520, 0],
    ["money", "MONEY.md", 520, 260],
    ["surfaces", "SURFACES.md", 520, 520],
    ["providers", "PROVIDERS.md", 1040, 0],
    ["impact", "IMPACT.md", 1040, 260],
    ["tests", "TEST-PLAN.md", 1040, 520],
    ["nodes", "NODES.md", 1560, 260],
  ];
  const nodes = pages.map(([id, file, x, y]) => ({ id, type: "file", file, x, y, width: 400, height: 220 }));
  const pairs = [
    ["index", "system"], ["index", "money"], ["index", "surfaces"],
    ["system", "providers"], ["money", "impact"], ["surfaces", "tests"],
    ["impact", "tests"], ["providers", "nodes"], ["tests", "nodes"],
  ];
  const edges = pairs.map(([fromNode, toNode], index) => ({ id: `edge-${index + 1}`, fromNode, fromSide: "right", toNode, toSide: "left" }));
  return `${JSON.stringify({ nodes, edges }, null, 2)}\n`;
}

function groupFlows(flows) {
  const order = ["VPN", "eSIM", "Virtual Numbers", "Card", "Passport", "Wallet", "Invoices", "Referrals", "Cross-product"];
  const groups = new Map(order.map(name => [name, []]));
  for (const flow of flows) groups.get(flowGroup(flow.id)).push(flow);
  return new Map([...groups].filter(([, nodes]) => nodes.length));
}

function flowGroup(id) {
  if (id.startsWith("flow:vpn.")) return "VPN";
  if (id.startsWith("flow:esim.")) return "eSIM";
  if (id.startsWith("flow:virtual-number.")) return "Virtual Numbers";
  if (id.startsWith("flow:card.")) return "Card";
  if (id.startsWith("flow:passport.")) return "Passport";
  if (id.startsWith("flow:wallet.")) return "Wallet";
  if (id.startsWith("flow:invoice.")) return "Invoices";
  if (id.startsWith("flow:referral.")) return "Referrals";
  return "Cross-product";
}

function isMoneyFlow(node) {
  return /(?:card|wallet|invoice|purchase-intents|products\.internal-balance)/.test(node.id);
}

function directlyVerifying(kind, impactedNodes, edges, nodeById) {
  const impacted = new Set(impactedNodes.map(node => node.id));
  return [...new Set(edges.filter(edge => edge.type === "verifies" && impacted.has(edge.to) && nodeById.get(edge.from)?.kind === kind).map(edge => edge.from))];
}

function renderTestGroups(groups, kind) {
  if (!groups.size) return "- none";
  return [...groups].map(([group, tests]) => [
    "<details>",
    `<summary>${escapeInline(group)} · ${tests.length}</summary>`,
    "",
    ...tests.map(test => `- ${kind} \`${test.id}\`${test.title ? ` — ${escapeInline(test.title)}` : ""} — ${renderReasons(test.reasons, kind)}`),
    "",
    "</details>",
  ].join("\n")).join("\n\n");
}

function renderReasons(reasons = [], kind) {
  if (!reasons.length) return kind === "manual" ? "manual safety net" : "selected";
  return reasons.map(reason => {
    if (reason.type === "impact") return `impact: ${(reason.edgePath ?? []).join(" → ")}`;
    if (reason.type === "always_run") return `always_run: invariant=${reason.invariantId ?? "unknown"}, reason=${reason.reason ?? "unknown"}`;
    if (reason.type === "policy_fallback") return "policy_fallback: see [[IMPACT#Why full fallback|why full fallback]]";
    return String(reason.type ?? "unknown");
  }).sort().join("; ");
}

function nodeLink(node) {
  return `[[${notePath(node).replace(/\.md$/, "")}|${node.id}]]`;
}

function notePath(node) {
  return `nodes/${slug(node.id)}.md`;
}

function slug(value) {
  return String(value).normalize("NFKD").toLowerCase().replace(/[^a-z0-9]+/g, "-").replace(/^-|-$/g, "");
}

function searchKey(value) {
  return String(value).replace(/[^A-Za-z0-9]+/g, " ").trim();
}

function nodeTitle(node) {
  if (node.kind === "flow") return `${node.product ?? "Flow"} · ${node.action ?? node.id}`;
  if (node.kind === "route") return node.label ?? node.title ?? node.id;
  return node.title ?? node.id;
}

function nodeDescription(node) {
  if (node.kind === "flow") return `- product: ${escapeInline(node.product ?? "unknown")}\n- action: ${escapeInline(node.action ?? "unknown")}`;
  if (node.kind === "route") return `- route/gate: ${escapeInline(node.label ?? "unknown")}`;
  return `- invariant: ${escapeInline(node.title ?? "unknown")}`;
}

function groupBy(values, keyOf) {
  const groups = new Map();
  for (const value of values) {
    const key = keyOf(value);
    const group = groups.get(key) ?? [];
    group.push(value);
    groups.set(key, group);
  }
  return new Map([...groups].sort(([left], [right]) => left.localeCompare(right)));
}

function countBy(values, keyOf) {
  const counts = new Map();
  for (const value of values) {
    const key = keyOf(value);
    counts.set(key, (counts.get(key) ?? 0) + 1);
  }
  return counts;
}

function compareEdges(left, right) {
  return `${left.type}\u0000${left.from}\u0000${left.to}`.localeCompare(`${right.type}\u0000${right.from}\u0000${right.to}`);
}

function compareTest(left, right) {
  return `${left.file ?? ""}\u0000${left.id ?? ""}`.localeCompare(`${right.file ?? ""}\u0000${right.id ?? ""}`);
}

function filePath(file) {
  return file.path ?? file.id?.replace(/^file:/, "") ?? "unknown";
}

function shortSha(value) {
  return /^[a-f0-9]{40}$/i.test(String(value)) ? String(value).slice(0, 12) : String(value ?? "unknown");
}

function escapeCode(value) {
  return String(value ?? "").replaceAll("`", "'");
}

function escapeInline(value) {
  return String(value ?? "").replaceAll("\n", " ").replaceAll("|", "\\|").trim();
}
