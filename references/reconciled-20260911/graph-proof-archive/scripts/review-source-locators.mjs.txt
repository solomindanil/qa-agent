import { execFileSync } from "node:child_process";
import { readFile, rename, writeFile } from "node:fs/promises";
import { dirname, resolve } from "node:path";

import { parseCorpus } from "../src/model.mjs";
import { isCodeSourceLocator } from "../src/proof.mjs";

const sourceRoot = argument("--source-root");
const outputPath = argument("--output");
if (!sourceRoot) throw new TypeError("--source-root is required");
const sourceSha = execFileSync("git", ["rev-parse", "HEAD"], { cwd: sourceRoot, encoding: "utf8" }).trim();
const files = execFileSync("git", ["ls-tree", "-r", "--name-only", sourceSha], {
  cwd: sourceRoot,
  encoding: "utf8",
  maxBuffer: 16 * 1024 * 1024,
}).trim().split("\n").filter(Boolean).sort();

const mappings = await json("mappings/mappings.v1.json");
const corpus = parseCorpus({
  productMap: await readFile(new URL("../contracts/PRODUCT-MAP.md", import.meta.url), "utf8"),
  coverageMap: await readFile(new URL("../contracts/COVERAGE-MAP.md", import.meta.url), "utf8"),
  manualCases: await readFile(new URL("../contracts/MANUAL-TEST-CASES.md", import.meta.url), "utf8"),
  baseline: await json("contracts/baseline.manifest.json"),
  aliases: mappings.aliases,
});
const nodeById = new Map(corpus.nodes.map(node => [node.id, node]));
const rows = [];

for (const locator of corpus.sourceLocators.filter(isCodeSourceLocator)) {
  const resolverToken = locator.rawToken.replace(/:(\d+)(?:[-–]\d+)?$/, "");
  const requestedLines = lineInterval(locator.rawToken);
  const candidates = files.filter(path => path === resolverToken || path.endsWith(`/${resolverToken}`));
  const reviewedCandidates = [];
  for (const path of candidates) {
    const text = await readFile(resolve(sourceRoot, path), "utf8");
    const lines = text.endsWith("\n") ? text.slice(0, -1).split("\n") : text.split("\n");
    const inBounds = requestedLines ? requestedLines[1] <= lines.length : null;
    reviewedCandidates.push({
      path,
      lineCount: lines.length,
      inBounds,
      snippet: requestedLines && inBounds ? snippet(lines, requestedLines) : null,
    });
  }
  rows.push({
    ownerNodeId: locator.ownerNodeId,
    ownerTitle: nodeTitle(nodeById.get(locator.ownerNodeId)),
    rawToken: locator.rawToken,
    resolverToken,
    requestedLines,
    candidates: reviewedCandidates,
    resolutionHint: resolutionHint(reviewedCandidates),
  });
}

const counts = {
  total: rows.length,
  exactSingle: rows.filter(row => row.candidates.length === 1 && row.candidates[0].inBounds !== false).length,
  ambiguousByPath: rows.filter(row => row.candidates.length > 1).length,
  missing: rows.filter(row => row.candidates.length === 0).length,
  outOfBounds: rows.filter(row => row.requestedLines && row.candidates.length > 0 && row.candidates.every(candidate => candidate.inBounds === false)).length,
};
const payload = `${JSON.stringify({ schemaVersion: 1, sourceSha, counts, rows }, null, 2)}\n`;
if (!outputPath) process.stdout.write(payload);
else {
  const target = resolve(outputPath);
  const temporary = resolve(dirname(target), `.${target.split("/").at(-1)}.tmp`);
  await writeFile(temporary, payload, { mode: 0o600 });
  await rename(temporary, target);
}

function argument(name) {
  const index = process.argv.indexOf(name);
  return index >= 0 ? process.argv[index + 1] : null;
}

async function json(path) {
  return JSON.parse(await readFile(new URL(`../${path}`, import.meta.url), "utf8"));
}

function lineInterval(rawToken) {
  const match = rawToken.match(/:(\d+)(?:[-–](\d+))?$/);
  return match ? [Number(match[1]), Number(match[2] ?? match[1])] : null;
}

function snippet(lines, [start, end]) {
  const from = Math.max(1, start - 2);
  const to = Math.min(lines.length, end + 2);
  return lines.slice(from - 1, to).map((text, index) => `${from + index}: ${text}`).join("\n");
}

function resolutionHint(candidates) {
  if (candidates.length === 0) return "missing_or_stale_path";
  const inBounds = candidates.filter(candidate => candidate.inBounds !== false);
  if (inBounds.length === 1) return "single_line_bound_candidate_requires_semantic_review";
  if (inBounds.length > 1) return "ambiguous_requires_semantic_review";
  return "stale_line_range_requires_contract_correction";
}

function nodeTitle(node) {
  return node?.title ?? node?.label ?? node?.action ?? node?.id ?? "unknown";
}
