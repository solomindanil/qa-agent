import { readFile, rename, writeFile } from "node:fs/promises";
import { dirname, resolve } from "node:path";

import { sourceLocatorReviewKey } from "../src/proof.mjs";

const worksheetPath = argument("--worksheet");
const correctionPath = argument("--corrections");
const decisionPaths = argumentsFor("--decisions");
const outputPath = argument("--output");
if (!worksheetPath || !correctionPath || decisionPaths.length === 0 || !outputPath) {
  throw new TypeError("--worksheet, --corrections, --decisions and --output are required");
}

const worksheet = JSON.parse(await readFile(resolve(worksheetPath), "utf8"));
const corrections = JSON.parse(await readFile(resolve(correctionPath), "utf8"));
const candidateByKey = new Map();

for (const path of decisionPaths) {
  const document = JSON.parse(await readFile(resolve(path), "utf8"));
  for (const decision of document.approved ?? document.decisions ?? []) addCandidate(decision);
}
for (const correction of corrections.corrections ?? []) {
  for (const replacement of correction.assertionsByReplacement ?? []) {
    addCandidate(fromReplacement(correction.ownerNodeId, replacement, correction.rationale));
  }
  for (const evidence of correction.contractCorrection?.sourceEvidence ?? []) {
    addCandidate(fromReplacement(correction.ownerNodeId, evidence, correction.rationale));
  }
}

const approved = [];
const missing = [];
for (const row of worksheet.rows ?? []) {
  const key = sourceLocatorReviewKey(row);
  const candidates = candidateByKey.get(key) ?? [];
  if (candidates.length === 0) missing.push({ ownerNodeId: row.ownerNodeId, rawToken: row.rawToken });
  else approved.push(mergeCandidates(candidates));
}
approved.sort((left, right) => sourceLocatorReviewKey(left).localeCompare(sourceLocatorReviewKey(right), "en"));
missing.sort((left, right) => sourceLocatorReviewKey(left).localeCompare(sourceLocatorReviewKey(right), "en"));
const payload = `${JSON.stringify({
  schemaVersion: 1,
  sourceSha: worksheet.sourceSha,
  counts: { total: worksheet.rows?.length ?? 0, approved: approved.length, missing: missing.length },
  approved,
  missing,
}, null, 2)}\n`;
const target = resolve(outputPath);
const temporary = resolve(dirname(target), `.${target.split("/").at(-1)}.tmp`);
await writeFile(temporary, payload, { mode: 0o600 });
await rename(temporary, target);

function addCandidate(decision) {
  if (!decision?.ownerNodeId || !decision?.rawToken || !decision?.path) return;
  const key = sourceLocatorReviewKey(decision);
  const values = candidateByKey.get(key) ?? [];
  values.push(decision);
  candidateByKey.set(key, values);
}

function fromReplacement(ownerNodeId, replacement, rationale) {
  const rawToken = replacement.replacementToken;
  const match = String(rawToken ?? "").match(/^(.*):(\d+)(?:[-–](\d+))?$/);
  if (!match) throw new TypeError(`Replacement requires explicit lines: ${rawToken}`);
  return {
    ownerNodeId,
    rawToken,
    path: match[1],
    correctedLines: [Number(match[2]), Number(match[3] ?? match[2])],
    assertions: replacement.assertions,
    rationale,
  };
}

function mergeCandidates(candidates) {
  const first = candidates[0];
  if (candidates.some(candidate => candidate.path !== first.path
    || JSON.stringify(candidate.correctedLines) !== JSON.stringify(first.correctedLines))) {
    throw new TypeError(`Conflicting source decisions: ${sourceLocatorReviewKey(first)}`);
  }
  const assertions = [...new Map(candidates.flatMap(candidate => candidate.assertions ?? [])
    .map(assertion => [JSON.stringify(assertion), assertion])).values()];
  return {
    ownerNodeId: first.ownerNodeId,
    rawToken: first.rawToken,
    path: first.path,
    correctedLines: first.correctedLines,
    assertions,
    rationale: [...new Set(candidates.map(candidate => candidate.rationale).filter(Boolean))].join(" "),
  };
}

function argument(name) {
  const index = process.argv.indexOf(name);
  return index >= 0 ? process.argv[index + 1] : null;
}

function argumentsFor(name) {
  const values = [];
  for (let index = 0; index < process.argv.length; index += 1) {
    if (process.argv[index] === name && process.argv[index + 1]) values.push(process.argv[index + 1]);
  }
  return values;
}
