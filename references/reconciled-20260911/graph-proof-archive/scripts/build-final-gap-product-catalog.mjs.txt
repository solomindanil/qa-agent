import { readFile, rename, writeFile } from "node:fs/promises";
import { dirname, resolve } from "node:path";

import { buildProductTestCatalog } from "../src/model.mjs";

const sourceSha = argument("--source-sha");
const legacyPath = argument("--legacy-catalog");
const outputPath = argument("--output");
if (!/^[0-9a-f]{40}$/.test(sourceSha ?? "") || !legacyPath || !outputPath) {
  throw new TypeError("--source-sha, --legacy-catalog and --output are required");
}

const legacy = JSON.parse(await readFile(resolve(legacyPath), "utf8"));
const tests = new Map();
for (const row of legacy.tests ?? []) {
  const renamed = legacySelectorRename(`${row.file}\u0000${row.selector}`);
  add(row.file, renamed ?? row.selector, row.bindings);
}

for (const [file, selectors, nodeIds] of finalGapBindings()) {
  for (const selector of selectors) {
    add(file, selector, nodeIds.map(nodeId => ({ nodeId, coverage: "full" })));
  }
}

const catalog = buildProductTestCatalog({
  sourceSha,
  tests: [...tests.values()].map(({ file, selector, bindings }) => ({
    file,
    selector,
    bindings: [...bindings.values()],
  })),
});
const target = resolve(outputPath);
const temporary = resolve(dirname(target), `.${target.split("/").at(-1)}.tmp`);
await writeFile(temporary, `${JSON.stringify(catalog, null, 2)}\n`, { mode: 0o600 });
await rename(temporary, target);
process.stdout.write(`${JSON.stringify({ tests: catalog.tests.length, digest: catalog.catalogDigest })}\n`);

function add(file, selector, bindings) {
  const id = `${file}\u0000${selector}`;
  const row = tests.get(id) ?? { file, selector, bindings: new Map() };
  for (const binding of bindings) {
    const previous = row.bindings.get(binding.nodeId);
    const rank = { detection: 1, partial: 2, full: 3 };
    if (!previous || rank[binding.coverage] > rank[previous.coverage]) {
      row.bindings.set(binding.nodeId, binding);
    }
  }
  tests.set(id, row);
}

function argument(name) {
  const index = process.argv.indexOf(name);
  return index >= 0 ? process.argv[index + 1] : null;
}

function legacySelectorRename(key) {
  return new Map([
    [[
      "apps/api/tests/payment-checkout-create-conflicts.test.ts",
      "req:B2 replays identical target checkouts, rejects parameter drift, and serializes concurrent provider creation",
    ].join("\u0000"),
    "invariant:payment-creation-idempotency replays the active production creator, rejects amount/product/method/identity drift, and serializes concurrent provider creation"],
  ]).get(key);
}
