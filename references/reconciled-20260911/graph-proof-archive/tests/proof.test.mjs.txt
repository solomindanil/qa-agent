import assert from "node:assert/strict";
import { mkdtemp, mkdir, rm, writeFile } from "node:fs/promises";
import { tmpdir } from "node:os";
import { join } from "node:path";
import test from "node:test";

import { buildGraph } from "../src/model.mjs";
import {
  buildSourceLocatorReviewDocument,
  digestFileBlob,
  digestReviewedLines,
  validateSourceLocatorReviews,
} from "../src/proof.mjs";

const sourceSha = "a".repeat(40);
const locator = {
  displayLines: [2],
  ownerNodeId: "req:A1",
  rawToken: "config.ts:2",
  resolverToken: "config.ts",
};
const fileText = "first\nsecond\nthird\n";

async function fixture() {
  const sourceRoot = await mkdtemp(join(tmpdir(), "freeland-proof-"));
  await mkdir(join(sourceRoot, "apps", "api"), { recursive: true });
  await writeFile(join(sourceRoot, "apps", "api", "config.ts"), fileText);
  const review = {
    ownerNodeId: locator.ownerNodeId,
    rawToken: locator.rawToken,
    path: "apps/api/config.ts",
    lines: [2, 2],
    fileBlobOid: digestFileBlob(Buffer.from(fileText)),
    contentSha256: digestReviewedLines(fileText, [2, 2]),
    assertions: [{ kind: "contains", value: "second" }],
    rationale: "Line 2 is the exact configuration value asserted by requirement A1.",
  };
  return {
    sourceRoot,
    review,
    document: buildSourceLocatorReviewDocument({ sourceSha, reviews: [review] }),
  };
}

test("source locator proof validates exact SHA, blob and reviewed line bytes", async () => {
  const value = await fixture();
  try {
    const result = await validateSourceLocatorReviews({
      document: value.document,
      locators: [locator],
      sourceRoot: value.sourceRoot,
      sourceSha,
      requireComplete: true,
    });
    assert.equal(result.ok, true);
    assert.equal(result.reviewed, 1);
  } finally {
    await rm(value.sourceRoot, { recursive: true, force: true });
  }
});

test("source locator proof rejects another source SHA", async () => {
  const value = await fixture();
  try {
    await assert.rejects(
      validateSourceLocatorReviews({
        document: value.document,
        locators: [locator],
        sourceRoot: value.sourceRoot,
        sourceSha: "b".repeat(40),
      }),
      /SOURCE_REVIEW_SHA_MISMATCH/,
    );
  } finally {
    await rm(value.sourceRoot, { recursive: true, force: true });
  }
});

test("source locator proof rejects duplicate owner and token keys", async () => {
  const value = await fixture();

    const document = buildSourceLocatorReviewDocument({

    });
    await assert.rejects(
      validateSourceLocatorReviews({ document, locators: [locator], sourceRoot: value.sourceRoot, sourceSha }),
      /SOURCE_REVIEW_DUPLICATE/,
    );
  } finally {
    await rm(value.sourceRoot, { recursive: true, force: true });
  }
});

test("source locator proof rejects missing paths and traversal", async () => {
  const value = await fixture();

    await rm(value.sourceRoot, { recursive: true, force: true });
  }
});

test("graph builder resolves a source locator only from validated proof", async () => {
  const value = await fixture();
  try {
    const corpus = {
      nodes: [{ id: locator.ownerNodeId, kind: "requirement", title: "A1" }],
      edges: [],
      sourceLocators: [locator],
      unresolved: [],
    };
    const withoutProof = buildGraph({
      corpus,
      sourceFiles: [value.review.path],
      sourceSha,
    });
    assert.equal(withoutProof.sourceLocators[0].disposition, "review_required");

    const validated = await validateSourceLocatorReviews({
      document: value.document,
      locators: [locator],
      sourceRoot: value.sourceRoot,
      sourceSha,
      requireComplete: true,
    });
    const withProof = buildGraph({
      corpus,
      sourceFiles: [value.review.path],
      sourceSha,
      sourceReviewsByKey: validated.byKey,
    });
    assert.equal(withProof.sourceLocators[0].disposition, "resolved");
    assert.equal(withProof.sourceLocators[0].path, value.review.path);
    assert.equal(withProof.sourceLocators[0].proof.contentSha256, value.review.contentSha256);
    assert.deepEqual(withProof.sourceLocators[0].proof.assertions, value.review.assertions);
  } finally {
    await rm(value.sourceRoot, { recursive: true, force: true });
  }
});
