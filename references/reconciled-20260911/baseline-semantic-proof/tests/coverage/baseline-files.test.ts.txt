function listTypeScriptFiles(root: string, prefix = ''): string[] {
  return readdirSync(root, { withFileTypes: true })
    .flatMap((entry) => {
      const relative = path.posix.join(prefix, entry.name);
      if (entry.isDirectory()) return listTypeScriptFiles(path.join(root, entry.name), relative);
      return entry.isFile() && entry.name.endsWith('.ts') ? [relative] : [];
    })
    .sort();
}

function loadManifest(): BaselineManifest {
  assertManifestIsRepositoryOwned(manifestRelativePath, repoRoot);
  const manifestPath = path.join(repoRoot, manifestRelativePath);
  assert.ok(existsSync(manifestPath), 'repository-owned Freeland baseline manifest is required');
  assert.equal(sha256File(manifestPath), manifestSha256, 'canonical baseline manifest hash mismatch');
  const manifest = JSON.parse(readFileSync(manifestPath, 'utf8')) as BaselineManifest;
  assert.equal(manifest.schemaVersion, 1);
  assert.equal(manifest.sourceSnapshot.algorithm, 'sha256');
  assert.equal(manifest.sourceSnapshot.sha256, snapshotSha256);
  assert.equal(manifest.sourceIdentity.algorithm, 'sha256');
  assert.equal(manifest.target.algorithm, 'sha256');
  for (const [relative, hash] of Object.entries(manifest.sourceIdentity.files)) {
    assert.ok(!path.isAbsolute(relative) && !relative.includes('..'), `invalid source identity path: ${relative}`);
    assert.match(hash, /^[0-9a-f]{64}$/, `invalid source identity SHA-256: ${relative}`);
  }
  for (const [relative, hash] of Object.entries(manifest.target.files)) {
    assert.ok(!path.isAbsolute(relative) && !relative.includes('..'), `invalid target path: ${relative}`);
    assert.match(hash, /^[0-9a-f]{64}$/, `invalid target SHA-256: ${relative}`);
  }
  assert.deepEqual(Object.keys(manifest.sourceIdentity.files).sort(), Object.keys(manifest.target.files).sort());
  assert.equal(Object.values(manifest.discovery.files).reduce((sum, count) => sum + count, 0), manifest.discovery.total);
  assert.equal(Object.keys(manifest.discovery.files).length, manifest.discovery.fileCount);
  assert.equal(manifest.discovery.total, 164);
  assert.equal(manifest.discovery.fileCount, 24);
  assert.deepEqual(manifest.transformations, expectedTransformationCounts);
  assert.deepEqual(manifest.proofInventory, expectedProofInventory);
  return manifest;
}

const manifest = loadManifest();
const baselineFiles = Object.keys(manifest.target.files).sort();

function loadSnapshot(): Readonly<Record<string, string>> {
  assertManifestIsRepositoryOwned(snapshotRelativePath, repoRoot);
  const snapshotPath = path.join(repoRoot, snapshotRelativePath);
  assert.equal(sha256File(snapshotPath), snapshotSha256, 'canonical source snapshot hash mismatch');
  const parsed = JSON.parse(readFileSync(snapshotPath, 'utf8')) as { schemaVersion: number; files: Record<string, string> };
  assert.equal(parsed.schemaVersion, 1);
  assertBaselineProofContract(manifest, parsed.files, sha256File(snapshotPath));
  assertSnapshotHashes(parsed.files, manifest.transformationProof.files);
  return parsed.files;
}

const sourceSnapshot = loadSnapshot();

function targetText(relative: string): string {
  return readFileSync(path.join(targetRoot, relative), 'utf8');
}

test('the canonical baseline contract is self-contained and has exactly 29 target files', () => {
  const guardSource = readFileSync(fileURLToPath(import.meta.url), 'utf8');
  const contractSource = readFileSync(path.join(repoRoot, 'tests/coverage/baseline-contract.ts'), 'utf8');
  assert.doesNotMatch(guardSource, /NuanuFlowQA\/tests\/freeland|\/Users\//);
  assert.doesNotMatch(contractSource, /NuanuFlowQA\/tests\/freeland|\/Users\//);
  assert.doesNotMatch(readFileSync(path.join(repoRoot, manifestRelativePath), 'utf8'), /\/Users\/|NuanuFlowQA\/tests\/freeland/);
  assert.deepEqual(listTypeScriptFiles(targetRoot), baselineFiles);
  assert.equal(baselineFiles.length, 29);
  assertTargetAstContract(targetRoot, manifest);
  assertTargetHashes(manifest.target.files, targetRoot);
  for (const relative of baselineFiles) {
    const source = sourceSnapshot[relative];
    const target = targetText(relative);
    const proof = manifest.transformationProof.files[relative];
    assert.deepEqual(createSemanticTransformationRecord(relative, source, target), proof);
    assertReconstructedTarget(source, target, proof);
  }
});

test('discovery-only JSON enumeration matches every registered Freeland test', () => {
  const actual = assertDiscoveryInventory(runDiscoveryJson(repoRoot), manifest.discovery.files);
  assert.equal(Object.values(actual).reduce((sum, count) => sum + count, 0), 164);
  assert.equal(Object.keys(actual).length, 24);
});

test('all suite-level product URL skips are removed after migration', () => {
  const offenders = baselineFiles
    .filter((file) => file.endsWith('.spec.ts'))
    .filter((file) => /test\.skip\(\s*!process\.env\.FREELAND_BASE_URL/.test(targetText(file)));
  assert.deepEqual(offenders, []);
});

test('app, responsive, and freeman contain no production credential gates', () => {
  const offenders = ['app.spec.ts', 'responsive.spec.ts', 'freeman.spec.ts'].filter((file) =>
    /FREELAND_TEST_(?:EMAIL|PASSWORD)/.test(targetText(file)),
  );
  assert.deepEqual(offenders, []);
});

test('auth state belongs only to Playwright projects', () => {
  const offenders = baselineFiles
    .filter((file) => file.endsWith('.spec.ts'))
    .filter((file) => /FREELAND_AUTH_FILE|test\.use\(\s*\{\s*storageState/.test(targetText(file)));
  assert.deepEqual(offenders, []);
  assert.doesNotMatch(targetText('helpers.ts'), /FREELAND_AUTH_FILE/);
  assert.doesNotMatch(targetText('auth.setup.ts'), /FREELAND_AUTH_FILE/);
});

test('only retired live-purchase exports are removed and the dry pay-sheet gate remains', () => {
  const helpers = targetText('helpers.ts');
  for (const retired of ['purchaseGate', 'PURCHASABLE_PRODUCTS', 'PAYMENT_PROVIDERS']) {
    assert.doesNotMatch(helpers, new RegExp(`export\\s+(?:const|type)\\s+${retired}\\b`));
  }
  assert.match(helpers, /export const paymentDryRunGate/);
});

test('auth setup writes exactly the project-owned auth path', () => {
  const setup = targetText('auth.setup.ts');
  assert.match(setup, /testInfo\.project\.metadata/);
  assert.match(setup, /requireProjectAuthFile\(\s*testInfo\.project\.metadata,\s*testInfo\.project\.name/);
  assert.match(setup, /storageState\(\{\s*path:\s*authFile\s*\}\)/);
});
