import { existsSync } from 'node:fs';

export const NUANU_APP_BASE_URL_ENV = 'NUANU_APP_BASE_URL';
export const AUTH_STORAGE_STATE_ENV = 'NUANU_APP_AUTH_STORAGE_STATE';

export const authStorageStatePath = (): string | undefined =>
  process.env[AUTH_STORAGE_STATE_ENV];

export const hasAuthStorageState = (): boolean => {
  const path = authStorageStatePath();
  return Boolean(path && existsSync(path));
};
