function clone(value) {
  return structuredClone(value);
}

function sealSchemaFacts(schemaFacts, queryManifest, accessLock) {
  const content = clone(schemaFacts);
  delete content.schemaFactsDigest;
  schemaFacts.schemaFactsDigest = digestJson(content);
  queryManifest.schemaObservation.schemaFactsDigest = schemaFacts.schemaFactsDigest;
  accessLock.approvedWindow.schemaFactsDigest = schemaFacts.schemaFactsDigest;
}

function requestIdentity(queryHash, accessLock) {
  for (const [queryId, approved] of Object.entries(accessLock.approvedQueryHashes)) {
    if (approved.primary === queryHash) return { queryId, requestRole: "primary" };
    if (approved.sampleCount === queryHash) return { queryId, requestRole: "sample_count" };
    if (approved.auxiliary?.exceptionFingerprint === queryHash) {
      return { queryId: `${queryId}:fingerprint`, requestRole: "exception_fingerprint" };
    }
  }
  return { queryId: ".invalid", requestRole: ".invalid" };
}

function resultKindFor(queryId) {
  return queryId.startsWith("funnel-") ? "funnel" : "series";
}

function makeAuthorizedProvider(accessLock, mutate = (envelope) => envelope) {
  const calls = [];
  const provider = {
    postJsonWithHostSecret: async (request) => {
      calls.push(clone(request));
      const queryHash = digestStructuredPostHogQuery(request.body.query);
      const identity = requestIdentity(queryHash, accessLock);
      const envelope = {
        status: 200,
        redirected: false,
        payload: {
          schemaVersion: 1,
          queryId: identity.queryId,
          queryHash,
          windowDigest: WINDOW_DIGEST,
          resultKind: resultKindFor(identity.queryId),
          coordinates: defaultCoordinates(identity.queryId, identity.requestRole),
        },
      };
      return mutate(envelope, identity, calls.length, request);
    },
  };
  return { provider, calls };
}

function makeDeferredAuthorizedProvider(accessLock) {
  const calls = [];
  const frozenQueries = [];
  let releaseGate;
  const gate = new Promise((resolve) => { releaseGate = resolve; });
  const provider = {
    postJsonWithHostSecret: async (request) => {
      await gate;
      calls.push(clone(request));
      frozenQueries.push(Object.isFrozen(request.body.query));
      const queryHash = digestStructuredPostHogQuery(request.body.query);
      const identity = requestIdentity(queryHash, accessLock);
      return {
        status: 200,
        redirected: false,
        payload: {
          schemaVersion: 1,
          queryId: identity.queryId,
          queryHash,
          windowDigest: WINDOW_DIGEST,
          resultKind: resultKindFor(identity.queryId),
          coordinates: defaultCoordinates(identity.queryId, identity.requestRole),
        },
      };
    },
  };
  return { provider, calls, frozenQueries, release: releaseGate };
}

function makeSystem({
  accessLock = fixtures().accessLock,
  schemaFacts = fixtures().schemaFacts,
  queryManifest = fixtures().queryManifest,
  mutateEnvelope,
  now = () => NOW,
} = {}) {
  const { provider, calls } = makeAuthorizedProvider(accessLock, mutateEnvelope);
  const transport = createExactPostHogTransport({
    accessLock,
    queryManifest,
    schemaFacts,
    window: WINDOW,
    authorizedPostProvider: provider,
    now: () => NOW,
  });
  const adapter = createPostHogReadAdapter({
    accessLock,
    queryManifest,
    schemaFacts,
    window: WINDOW,
    transport,
    now,
  });
  return { adapter, transport, provider, calls };
}

  assert.equal(candidate.status, "BLOCKED");

  assert.equal(candidate.aggregate, null);
  assert.deepEqual(candidate.breakdowns, []);
  assert.equal(candidate.p75Pairing, null);
}

test("empty or overall-missing p75 pairs are unavailable rather than throwing", async () => {
  for (const mutateEnvelope of [
    (envelope, identity) => {
      if (identity.queryId === "web-vitals-lcp-p75") envelope.payload.coordinates = [];
      return envelope;
    },
    (envelope, identity) => {
      if (identity.queryId === "web-vitals-lcp-p75") {
        envelope.payload.coordinates = envelope.payload.coordinates.filter(
          (coordinate) => coordinate.dimensionIds.length !== 0,
        );
      }
      return envelope;
    },
  ]) {
    const system = makeSystem({ mutateEnvelope });
    const result = await system.adapter.readMetric({ queryId: "web-vitals-lcp-p75" });
    assert.equal(result.candidate.status, "UNAVAILABLE");
    assert.equal(result.candidate.queryHash, LCP_PRIMARY_HASH);
    assert.equal(result.candidate.p75Pairing, null);
    assert.deepEqual(result.candidate.limitations, ["P75_COUNT_PAIR_MISMATCH"]);
  }
});
