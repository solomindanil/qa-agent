function node(kind, suffix, overrides = {}) {
  return {
    id: `nuanu-app:${kind}:${suffix}`,
    kind,
    provenance: "runtime",
    ...overrides,
  };
}

function edge(source, kind, target) {
  return { source, kind, target };
}

function graphFixture({ nodes = [], edges = [] } = {}) {
  return {
    productId: "nuanu-app",
    schemaVersion: "1.0.0",
    nodes,
    edges,
  };
}

function runtimeOnlyImpact() {
  return {
    mode: "runtime_only",
    changedFiles: [],
    affectedNodeIds: [],
    unmappedFiles: [],
    paths: [],
    sourceImpact: "UNVERIFIED",
  };
}

function automatedTest(id, suffix, overrides = {}) {
  return node("automated_test", suffix, {
    id,
    execution: "automated",
    safety: "read_only",
    coverageDisposition: "AUTOMATED",
    ...overrides,
  });
}

function manualTest(suffix, overrides = {}) {
  return node("manual_test", suffix, {
    coverageDisposition: "MANUAL",
    ...overrides,
  });
}

test("planImpact records every changed-file to node to test path", () => {
  const source = node("source_file", "home", {
    path: "apps/nuanu-web/src/app/page.tsx",
    provenance: "source",
    coverageDisposition: "UNVERIFIED",
  });
  const route = node("route", "home", { provenance: "source_and_runtime" });
  const surface = node("surface", "home");
  const testNode = automatedTest(SMOKE_ID, "home-smoke");
  const graph = graphFixture({
    nodes: [source, route, surface, testNode],
    edges: [
      edge(source.id, "implements", route.id),
      edge(route.id, "affects", surface.id),
      edge(surface.id, "affects", route.id),
      edge(testNode.id, "verifies", surface.id),
    ],
  });

  const impact = planImpact({ graph, changedFiles: [source.path] });

  assert.deepEqual(impact.changedFiles, [source.path]);
  assert.deepEqual(impact.unmappedFiles, []);
  assert.deepEqual(impact.affectedNodeIds, [route.id, surface.id]);
  assert.deepEqual(impact.paths, [{
    changedFile: source.path,
    nodeIds: [source.id, route.id, surface.id, testNode.id],
    edges: [
      edge(source.id, "implements", route.id),
      edge(route.id, "affects", surface.id),
      edge(testNode.id, "verifies", surface.id),
    ],
  }]);
});

test("unmapped file selects all runnable safe tests and mobile manual tier", () => {
  const safeTest = automatedTest(SMOKE_ID, "safe");
  const unavailableAuth = automatedTest(SAFETY_ID, "auth", { coverageDisposition: "BLOCKED" });
  const mobileManual = manualTest("mobile", { manualTier: "mobile_safety" });
  const liveTarget = node("route", "fallback-live", { provenance: "source_and_runtime" });
  const graph = graphFixture({
    nodes: [safeTest, unavailableAuth, mobileManual, liveTarget],
    edges: [
      edge(safeTest.id, "verifies", liveTarget.id),
      edge(mobileManual.id, "verifies", liveTarget.id),
    ],
  });
  const impact = planImpact({ graph, changedFiles: ["apps/nuanu-web/src/new-unmapped.ts"] });

  const plan = buildTestPlan({
    graph,
    impact,
    runnableTestIds: new Set([SMOKE_ID, SAFETY_ID]),
  });

  assert.equal(plan.fallbackApplied, true);
  assert.deepEqual(plan.selectedAutomated, [{
    id: SMOKE_ID,
    reason: "full_fallback",
    policy: "full_fallback",
  }]);
  assert.deepEqual(plan.manual, [{
    id: mobileManual.id,
    reason: "mobile_safety",
    policy: "mobile_safety",
  }]);
  assert.deepEqual(plan.blocked, [{
    id: SAFETY_ID,
    disposition: "BLOCKED",
  }]);
});

test("all-unavailable impact paths use deterministic blocked-unverified-not-deployed precedence", () => {
  const source = node("source_file", "all-unavailable", {
    path: "apps/nuanu-web/src/all-unavailable.ts",
    provenance: "source",
    coverageDisposition: "UNVERIFIED",
  });
  const blockedTarget = node("route", "blocked", {
    provenance: "runtime",
    coverageDisposition: "BLOCKED",
  });
  const unverifiedTarget = node("capability", "unverified", {
    provenance: "source",
    coverageDisposition: "UNVERIFIED",
  });
  const notDeployedTarget = node("route", "not-deployed", {
    provenance: "source_and_runtime",
    coverageDisposition: "NOT_DEPLOYED",
  });
  const testNode = automatedTest(SMOKE_ID, "all-unavailable");
  const graph = graphFixture({
    nodes: [source, blockedTarget, unverifiedTarget, notDeployedTarget, testNode],
    edges: [
      edge(source.id, "implements", blockedTarget.id),
      edge(source.id, "implements", unverifiedTarget.id),
      edge(source.id, "implements", notDeployedTarget.id),
      edge(testNode.id, "verifies", blockedTarget.id),
      edge(testNode.id, "verifies", unverifiedTarget.id),
      edge(testNode.id, "verifies", notDeployedTarget.id),
    ],
  });
  const impact = planImpact({ graph, changedFiles: [source.path] });
  const plan = buildTestPlan({ graph, impact, runnableTestIds: new Set([testNode.id]) });

  assert.deepEqual(plan.selectedAutomated, []);
  assert.deepEqual(plan.blocked, [
    { id: testNode.id, disposition: "BLOCKED" },
    { id: blockedTarget.id, disposition: "BLOCKED" },
  ]);
  assert.deepEqual(plan.unverified, [{ id: unverifiedTarget.id, disposition: "UNVERIFIED" }]);
  assert.deepEqual(plan.notDeployed, [{ id: notDeployedTarget.id, disposition: "NOT_DEPLOYED" }]);
  const outputIds = [
    ...plan.selectedAutomated,
    ...plan.manual,
    ...plan.blocked,
    ...plan.unverified,
    ...plan.notDeployed,
  ].map(({ id }) => id);
  assert.equal(new Set(outputIds).size, outputIds.length);
});

    coverageDisposition: "UNVERIFIED",

  const edges = [edge(source.id, "implements", start.id)];
  let previous = [start];
  for (let level = 0; level < 15; level += 1) {
    const left = node("surface", `limit-${level}-left`);
    const right = node("surface", `limit-${level}-right`);
    nodes.push(left, right);
    for (const parent of previous) {
      edges.push(edge(parent.id, "affects", left.id), edge(parent.id, "affects", right.id));
    }
    previous = [left, right];
  }
  const graph = graphFixture({ nodes, edges });

  assert.throws(
    () => planImpact({ graph, changedFiles: [source.path] }),
    { code: "IMPACT_PATH_LIMIT_EXCEEDED" },
  );

let signalCounter = 0;

function telemetrySignal(mappingSuffix, targetNodeIds, disposition = "select", overrides = {}) {
  const id = `napp-telemetry-v1-${(signalCounter++).toString(16).padStart(64, "0")}`;
  return {
    id,
    productId: "nuanu-app",
    catalogDigest: "b".repeat(64),
    mappingId: `napp-telemetry-map-v1:${mappingSuffix}`,
    metricId: "napp-risk-score",
    targetNodeIds,
    disposition,
    kind: "observed",
    priority: "informational",
    ...overrides,
  };
}

function telemetryGraph() {
  const source = node("source_file", "mapped", {

  return {
    source,

      edges: [
        edge(source.id, "implements", mappedCapability.id),
        edge(observedTest.id, "verifies", observedRoute.id),
      ],
    }),
  };
}
