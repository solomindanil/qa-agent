function clone(value) {
  return structuredClone(value);
}

function bindingFor(queryId, { breakdown = null } = {}) {
  const definition = DEFINITIONS[queryId];
  const isP75 = queryId.startsWith("web-vitals-");
  const safeIdentifierForRole = (role) => ({
    eligible_session_population: "napp_test_eligible_population",
    exception_session_population: "napp_test_exception_population",
  })[role] ?? `napp_test_${role}`;
  return {
    queryId,
    builderId: definition.builderId,
    eventIdentifiers: definition.roles.map(([role]) => safeIdentifierForRole(role)),
    valuePropertyIdentifier: isP75 ? "napp_test_vital_value" : null,
    hostPropertyIdentifier: "napp_test_host",
    breakdown,
    p75SampleCountCompanion: isP75
      ? {
          builderId: "trends-total-sample-size-v1",
          exactPopulationSemanticsEvidenceDigest: HASH_A,
        }
      : null,
  };
}

function expectedFreshnessQuery() {
  return {
    kind: "TrendsQuery",
    series: [{
      kind: "EventsNode",
      event: "napp_test_population_event",
      math: "total",
    }],
    dateRange: {
      date_from: WINDOW.startAt,
      date_to: WINDOW.endAt,
    },
    interval: "day",
    properties: [{
      type: "event",
      key: "napp_test_host",
      operator: "exact",
      value: "app.nuanu.com",
    }],
    trendsFilter: { display: "ActionsLineGraph" },
  };
}

function mutateQuery(queryId, mutation) {
  const binding = bindingFor(queryId);
  const query = clone(buildStructuredPostHogQuery({ queryId, binding, window: WINDOW }));
  mutation(query);
  return { query, binding };
}

test("builds only TrendsQuery FunnelsQuery and nested EventsNode", () => {
  assert.deepEqual(ALLOWED_ROOT_QUERY_KINDS, ["TrendsQuery", "FunnelsQuery"]);
  assert.deepEqual(ALLOWED_SERIES_KINDS, ["EventsNode"]);
  assert.deepEqual(ALLOWED_AGGREGATIONS, ["total", "unique_session", "p75"]);
  assert.deepEqual(ALLOWED_PROPERTY_OPERATORS, ["exact"]);
  assert.equal(TARGET_APP_HOST, "app.nuanu.com");

  for (const queryId of Object.keys(DEFINITIONS)) {
    if (queryId.startsWith("web-vitals-")) continue;
    const binding = bindingFor(queryId);
    const query = buildStructuredPostHogQuery({ queryId, binding, window: WINDOW });
    assert.equal(query.kind, DEFINITIONS[queryId].queryKind);
    assert.ok(query.series.every((series) => series.kind === "EventsNode"));
    assert.deepEqual(
      validateStructuredPostHogQuery({ query, binding, window: WINDOW }),
      { ok: true },
    );
  }
});

test("accepts only a canonical UTC-midnight exact 24-hour day window", () => {
  assert.equal(POSTHOG_WINDOW_INTERVAL, "day");
  assert.equal(POSTHOG_WINDOW_DURATION_HOURS, 24);
  assert.deepEqual(validateAndFreezePostHogWindow(WINDOW), WINDOW);

  for (const invalid of [
    { ...WINDOW, extra: ".invalid" },
    { ...WINDOW, startAt: "2026-08-25T00:00:00Z" },
    { ...WINDOW, startAt: "2026-08-25T01:00:00.000Z" },
    { ...WINDOW, startAt: "2026-02-30T00:00:00.000Z" },
    { ...WINDOW, endAt: "2026-08-26T00:00:00.001Z" },
    { ...WINDOW, endAt: WINDOW.startAt },
    { ...WINDOW, interval: ".invalid" },
  ]) {
    assert.throws(() => validateAndFreezePostHogWindow(invalid), /window/i);
  }
});

test("defensively freezes the window and never derives one from now", () => {
  const mutable = { ...WINDOW };
  const frozen = validateAndFreezePostHogWindow(mutable);
  mutable.startAt = NEXT_WINDOW.startAt;
  assert.equal(frozen.startAt, WINDOW.startAt);
  assert.ok(Object.isFrozen(frozen));
  assert.throws(
    () => validateAndFreezePostHogWindow({ ...WINDOW, now: "2026-08-26T00:05:00.000Z" }),
    /window/i,
  );
});

test("query hashes change when the validated window changes", () => {
  const binding = bindingFor("telemetry-freshness");
  const first = buildStructuredPostHogQuery({
    queryId: "telemetry-freshness",
    binding,
    window: WINDOW,
  });
  const second = buildStructuredPostHogQuery({
    queryId: "telemetry-freshness",
    binding,
    window: NEXT_WINDOW,
  });
  assert.deepEqual(first, expectedFreshnessQuery());
  assert.equal(digestStructuredPostHogQuery(first), EXPECTED_FRESHNESS_QUERY_HASH);
  assert.notEqual(digestStructuredPostHogQuery(first), digestStructuredPostHogQuery(second));
});

test("p75 pair uses the shared canonical digest input and helper", () => {
  const binding = bindingFor("web-vitals-lcp-p75", {
    breakdown: {
      dimension: "browser",
      propertyIdentifier: "napp_test_browser",
      publicValueAllowlist: null,
    },
  });
  const result = buildP75PostHogQueryPair({
    metricId: "web-vitals-lcp-p75",
    binding,
    window: WINDOW,
  });
  const input = createP75PairDigestInput({
    metricId: "web-vitals-lcp-p75",
    dimensions: ["browser"],
    percentileQueryHash: digestStructuredPostHogQuery(result.pairBundle.percentileQuery),
    sampleCountQueryHash: digestStructuredPostHogQuery(result.pairBundle.sampleCountQuery),
  });
  assert.deepEqual(result.pairBundle.p75PairDigestInput, input);
  assert.equal(result.pairBundle.pairDigest, digestP75Pair(input));
});

test("rejects missing extra reversed or mismatched p75 pair fields before client use", () => {
  const binding = bindingFor("web-vitals-lcp-p75");
  const ready = buildP75PostHogQueryPair({
    metricId: "web-vitals-lcp-p75",
    binding,
    window: WINDOW,
  });
  const mutations = [
    (pair) => { delete pair.pairDigest; },
    (pair) => { pair.extra = ".invalid"; },
    (pair) => {
      const next = pair.p75PairDigestInput.percentile;
      pair.p75PairDigestInput.percentile = pair.p75PairDigestInput.sampleCount;
      pair.p75PairDigestInput.sampleCount = next;
    },
    (pair) => { pair.sampleCountQuery.dateRange.date_from = NEXT_WINDOW.startAt; },
    (pair) => { pair.sampleCountQuery.series[0].event = "napp_test_other_event"; },
  ];
  for (const mutation of mutations) {
    const pair = clone(ready.pairBundle);
    mutation(pair);
    assert.throws(
      () => validateP75PostHogQueryPair({ pair, binding, window: WINDOW }),
      /pair|query|hash|window|event|keys/i,
    );
  }
});
