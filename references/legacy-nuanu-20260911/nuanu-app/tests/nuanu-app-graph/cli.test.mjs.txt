function without(object, key) {
  return Object.fromEntries(Object.entries(object).filter(([candidate]) => candidate !== key));
}

function storedGraphDigest(graph) {
  const timestampSuffixes = [
    "timestamp",
    "timestamps",
    "timestampms",
    "generatedat",
    "createdat",
    "updatedat",
    "observedat",
    "capturedat",
    "collectedat",
    "fetchedat",
    "checkedat",
    "seenat",
    "timeobserved",
    "observationtime",
  ];
  const withoutTimestamps = (value) => {
    if (Array.isArray(value)) return value.map(withoutTimestamps);
    if (!value || typeof value !== "object") return value;
    return Object.fromEntries(Object.entries(value)
      .filter(([key]) => {
        const normalized = key.replace(/[^a-z0-9]/gi, "").toLowerCase();
        return !timestampSuffixes.some((suffix) => normalized.endsWith(suffix));
      })
      .map(([key, child]) => [key, withoutTimestamps(child)]));
  };
  const semantic = withoutTimestamps(graph);
  delete semantic.graphDigest;
  return digestJson(semantic);
}

async function captureCli(args, dependencies = {}) {
  let stdout = "";
  let stderr = "";
  const exitCode = await runCli(args, {
    now: () => new Date(FIXED_NOW),
    discoverPlaywright: async () => playwrightList(),
    observeRuntime: async () => runtimeObservation(),
    loadChromium: async () => ({ launch() { throw new Error("observer must be injected"); } }),
    stdout: { write(value) { stdout += value; } },
    stderr: { write(value) { stderr += value; } },
    ...dependencies,
  });
  return { exitCode, stdout, stderr };
}

    },

      forbiddenDependencyCalled = true;

    },

test("failure writing the second plan artifact leaves no new commit marker or temp", async (t) => {
  const fixture = await bootstrapGraph(t);
  const markerPath = join(fixture.root, "product-graph/current/plan-commit.json");
  await fixture.io.writeJsonAtomic(markerPath, { previous: true });
  const failingIo = createJsonIo({
    privateRoot: fixture.root,
    operations: {
      async rename(from, to, actualRename) {
        if (to.endsWith("test-plan.json")) {
          const error = new Error("synthetic second artifact failure");
          error.code = "EIO";
          throw error;
        }
        return actualRename(from, to);
      },
    },
  });
  const result = await captureCli(["plan", "--runtime-only", "--dry-run"], { io: failingIo });
  assert.equal(result.exitCode, 1);
  assert.deepEqual(JSON.parse(await readFile(markerPath, "utf8")), { previous: true });
  assert.deepEqual((await readdir(dirname(markerPath))).filter((name) => name.endsWith(".tmp")), []);
});

test("plan marker is restored or removed when its post-rename directory-sync event fails", async (t) => {
  for (const existing of [false, true]) {
    const fixture = await bootstrapGraph(t);
    const current = join(fixture.root, "product-graph/current");
    const markerPath = join(current, "plan-commit.json");
    const original = Buffer.from('{"previous":true}\n');
    if (existing) await writeFile(markerPath, original, { mode: 0o600 });
    let markerRenamed = false;
    const failingIo = createJsonIo({
      privateRoot: fixture.root,
      onEvent(event) {
        if (event.type === "renamed" && event.fileName === "plan-commit.json") {
          markerRenamed = true;
        } else if (markerRenamed && event.type === "directory-synced") {
          const failure = new Error("synthetic marker directory sync event failure");
          failure.code = "EIO";
          throw failure;
        }
      },
    });

    const result = await captureCli(["plan", "--runtime-only", "--dry-run"], { io: failingIo });
    assert.equal(result.exitCode, 1);
    assert.equal(markerRenamed, true);
    if (existing) assert.deepEqual(await readFile(markerPath), original);
    else await assert.rejects(() => lstat(markerPath), { code: "ENOENT" });
    assert.deepEqual(transactionLeftovers(await readdir(current)), []);
  }
});

test("spawn log permits only read-only Git commands and Playwright list discovery", async (t) => {
  const { io } = await bootstrapGraph(t);
  const repo = "/synthetic/repo";
  const lock = {
    productId: "nuanu-app", repositoryUrl: EXPECTED_REMOTE,
    commitSha: "b".repeat(40), clean: true, sourceRoots: ["apps/nuanu-web"],
    contentDigest: "c".repeat(64), mappingDigest: "d".repeat(64),
  };
  const log = [];
  const result = await captureCli([
    "plan", "--repo", repo, "--from", "a".repeat(40), "--to", "b".repeat(40), "--dry-run",
  ], {
    io,
    attestRepository: async () => repo,
    discoverPlaywright: async (invocation) => {
      log.push({ kind: "playwright", invocation });
      return playwrightList();
    },
    buildSourceLock: async (input) => {
      log.push({ kind: "source-lock", input });
      return { ...lock, mappingDigest: input.mappingDigest };
    },
    git: async (_repo, args, options) => {
      log.push({ kind: "git", args, options });
      if (args[0] === "rev-parse") return `${"b".repeat(40)}\n`;
      if (args[0] === "cat-file") return "";
      if (args[0] === "diff") return Buffer.alloc(0);
      throw new Error("unapproved Git command");
    },
  });
  assert.equal(result.exitCode, 1);
  assert.equal(log.some((entry) => ["checkout", "fetch", "pull", "add", "commit", "push"].includes(entry.args?.[0])), false);
  assert.ok(log.filter((entry) => entry.kind === "git").every((entry) =>
    entry.options.env.GIT_OPTIONAL_LOCKS === "0" &&
    entry.options.env.GIT_TERMINAL_PROMPT === "0" &&
    entry.options.env.GIT_FSMONITOR_TEST === "0"));
  assert.ok(log.filter((entry) => entry.kind === "playwright").every((entry) =>
    entry.invocation.args.includes("--list") && !entry.invocation.args.some((arg) => arg === "--ui")));
});
