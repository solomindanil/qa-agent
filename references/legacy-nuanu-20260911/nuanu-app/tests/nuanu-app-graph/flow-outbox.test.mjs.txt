function shaFor(value) {
  return `sha256:${createHash("sha256").update(String(value)).digest("hex")}`;

function escapeHtml(value) {
  return value
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll("\"", "&quot;")
    .replaceAll("'", "&#39;");
}

function variableStrings(value) {
  return [
    value.symptom,
    value.affectedObjectOrRoute,
    ...value.preconditions,
    ...value.steps,
    value.actual,
    value.expected,
    value.impact,
    value.evidenceSummary,
    ...value.attachments.flatMap((item) => [item.evidenceId, item.fileName, item.mediaType]),
    ...value.graph.nodeIds,
    ...(value.regression.testId === null ? [] : [value.regression.testId]),
    value.regression.design,
    value.deduplication.searchedAt,
  ];
}

function variableTextBytes(value) {
  return variableStrings(value).reduce((total, item) => total + Buffer.byteLength(item), 0);
}

function repeatedPattern(bytes, pattern = "&<>'\"") {
  return pattern.repeat(Math.ceil(bytes / pattern.length)).slice(0, bytes);
}

function findingAtAggregateBytes(target, pattern = "x") {
  const value = finding({
    symptom: "x",
    affectedObjectOrRoute: "/x",
    preconditions: ["x"],
    steps: ["x"],
    actual: "x",
    expected: "x",
    impact: "x",
    evidenceSummary: "x",
    graph: { graphDigest: SHA_C, nodeIds: ["nuanu-app:x"] },
    regression: { status: "AUTOMATED", testId: "NAPP-X-1", design: "x" },
  });
  let remaining = target - variableTextBytes(value);
  assert.ok(remaining >= 0);
  for (const key of ["actual", "expected", "impact", "evidenceSummary"]) {
    const add = Math.min(remaining, MAX_TEXT_BYTES - Buffer.byteLength(value[key]));
    value[key] += repeatedPattern(add, pattern);
    remaining -= add;
  }
  assert.equal(remaining, 0, "test fixture has enough bounded fields for aggregate bytes");
  assert.equal(variableTextBytes(value), target);
  return value;
}

function actionManifest(entry, {
  kind = "search_issues",
  issueId = remoteIdentity().id,
  actionId,
  batchDigest = exactBatchDigest(entry),
  proposedAt = NOW,
  descriptorCapturedAt,
  name,
  descriptionHtml,
  previousOutbox = { version: 1, entries: [entry] },
} = {}) {
  const codeOwnedCreatePayload = expectedCreatePayload(entry);
  const kindFields = {
    search_issues: {
      mutation: false,
      arguments: {
        workspaceSlug: "nuanu-app",
        projectIdentifier: "NAPP",
        fingerprint: entry.fingerprint,
        scopes: ["active", "recently_closed"],
        requireComplete: true,
      },
      expectedObservation: "search",
    },
    get_issue: {
      mutation: false,
      arguments: {
        workspaceSlug: "nuanu-app",
        projectIdentifier: "NAPP",
        issueId,
      },
      expectedObservation: "issue",
    },
    create_issue: {
      mutation: true,
      arguments: {
        project_identifier: "NAPP",
        name: name ?? codeOwnedCreatePayload.name,
        description_html: descriptionHtml ?? codeOwnedCreatePayload.description_html,
        state_name: "Bugs",
        label_names: ["Bug", entry.finding.platform].sort(),
        priority: entry.finding.priority,
      },
      expectedObservation: "mutation_ack",
    },
  }[kind];
  assert.ok(kindFields);
  const semantic = {
    version: 1,
    productId: "nuanu-app",
    fingerprint: entry.fingerprint,
    findingDigest: entry.findingDigest,
    batchDigest,
    proposedAt,
    expiresAt: new Date(Date.parse(proposedAt) + MAX_ACTION_AGE_MS).toISOString(),
    kind,
    mutation: kindFields.mutation,
    descriptorBinding: descriptorBinding(descriptorCapturedAt ?? proposedAt),
    arguments: kindFields.arguments,
    expectedObservation: kindFields.expectedObservation,
  };
  const derivedActionId = `napp-flow-action-v1-${digestJson({
    version: 1,
    outboxRevision: exactOutboxRevision(previousOutbox),
    action: semantic,
  })}`;
  const preceding = {
    version: semantic.version,
    productId: semantic.productId,
    actionId: actionId ?? derivedActionId,
    fingerprint: semantic.fingerprint,
    findingDigest: semantic.findingDigest,
    batchDigest: semantic.batchDigest,
    proposedAt: semantic.proposedAt,
    expiresAt: semantic.expiresAt,
    kind: semantic.kind,
    mutation: semantic.mutation,
    descriptorBinding: semantic.descriptorBinding,
    arguments: semantic.arguments,
    expectedObservation: semantic.expectedObservation,
  };
  return { ...preceding, actionDigest: `sha256:${digestJson(preceding)}` };
}

function proposedAction(entry, overrides = {}) {
  return {
    manifest: actionManifest(entry, overrides),
    lifecycle: "PROPOSED",
    dispatchedAt: null,
  };
}

function dispatchedAction(entry, { dispatchedAt = NOW, ...overrides } = {}) {
  return {
    manifest: actionManifest(entry, overrides),
    lifecycle: "DISPATCHED",
    dispatchedAt,
  };
}

function createReceipt({
  outcome = "ACKED",
  remoteId = remoteIdentity().id,
  candidateRemoteId,
  observedAt = NOW,
} = {}) {
  const common = {
    kind: "create_issue",
    actionId: `napp-flow-action-v1-${"e".repeat(64)}`,
    actionDigest: `sha256:${"f".repeat(64)}`,
    outcome,
  };
  return outcome === "ACKED"
    ? { ...common, remoteId, observedAt }
    : {
      ...common,
      candidateRemoteId: candidateRemoteId ?? null,
      observedAt,
    };
}

async function fixture(t, { ioOptions = {}, lockFs, limits } = {}) {
  const createdRoot = await mkdtemp(join(tmpdir(), "napp-flow-outbox-"));
  const root = await realpath(createdRoot);
  t.after(() => rm(createdRoot, { recursive: true, force: true }));
  const io = createJsonIo({ privateRoot: root, ...ioOptions });
  const outboxPath = join(root, OUTBOX_RELATIVE_PATH);
  const lockPath = join(root, OUTBOX_LOCK_RELATIVE_PATH);
  const service = createFlowOutboxService({ io, outboxPath, lockPath, lockFs, limits });
  return { root, io, outboxPath, lockPath, service };
}

function transactionInput(operation, overrides = {}) {
  return {
    ownerId: OWNER,
    now: () => new Date(NOW),
    leaseMs: 30_000,
    operation,
    ...overrides,
  };
}

async function enqueue(service, value, overrides = {}) {
  return service.enqueueConfirmedFinding({
    finding: value,
    ownerId: OWNER,
    now: () => new Date(NOW),
    leaseMs: 30_000,
    ...overrides,
  });
}

async function assertFindingAccepted(t, value) {
  const { service } = await fixture(t);
  const persisted = await enqueue(service, value);
  assert.equal(persisted.findingDigest, `sha256:${digestJson(value)}`);
}

async function assertFindingRejectedBeforePersistence(t, value, code = "INVALID_FLOW_FINDING") {
  const { outboxPath, lockPath, service } = await fixture(t);
  await assert.rejects(() => enqueue(service, value), { code });
  await assert.rejects(() => stat(outboxPath), { code: "ENOENT" });
  await assert.rejects(() => stat(lockPath), { code: "ENOENT" });
}

test("fingerprint rejects prototypes accessors symbols and non-enumerable fields without invoking accessors", () => {
  const minimal = {
    productId: "nuanu-app",
    symptom: "safe defect",
    affectedObjectOrRoute: "/safe",
    platform: "Desktop",
    runtimeIdentity: SHA_A,
    evidenceSignature: SHA_B,
  };
  const customPrototype = Object.assign(Object.create({ inherited: true }), minimal);
  assert.throws(() => fingerprintFinding(customPrototype), { code: "INVALID_FLOW_FINDING" });

  let accessed = false;
  const accessor = { ...minimal };
  Object.defineProperty(accessor, "symptom", {
    enumerable: true,
    get() {
      accessed = true;
      return "safe defect";
    },
  });
  assert.throws(() => fingerprintFinding(accessor), { code: "INVALID_FLOW_FINDING" });
  assert.equal(accessed, false);

  const symbolBearing = { ...minimal, [Symbol("secret")]: "unsafe" };
  assert.throws(() => fingerprintFinding(symbolBearing), { code: "INVALID_FLOW_FINDING" });

  const nonEnumerable = { ...minimal };
  Object.defineProperty(nonEnumerable, "hidden", { value: "unsafe", enumerable: false });
  assert.throws(() => fingerprintFinding(nonEnumerable), { code: "INVALID_FLOW_FINDING" });
});

test("fingerprint rejects root and nested proxies before any proxy trap executes", () => {
  const trapCounts = {
    get: 0,
    getOwnPropertyDescriptor: 0,
    getPrototypeOf: 0,
    ownKeys: 0,
  };
  const proxied = (target) => new Proxy(target, {
    get(...arguments_) {
      trapCounts.get += 1;
      return Reflect.get(...arguments_);
    },
    getOwnPropertyDescriptor(...arguments_) {
      trapCounts.getOwnPropertyDescriptor += 1;
      return Reflect.getOwnPropertyDescriptor(...arguments_);
    },
    getPrototypeOf(...arguments_) {
      trapCounts.getPrototypeOf += 1;
      return Reflect.getPrototypeOf(...arguments_);
    },
    ownKeys(...arguments_) {
      trapCounts.ownKeys += 1;
      return Reflect.ownKeys(...arguments_);
    },
  });

  const minimal = {
    productId: "nuanu-app",
    symptom: "safe defect",
    affectedObjectOrRoute: "/safe",
    platform: "Desktop",
    runtimeIdentity: SHA_A,
    evidenceSignature: SHA_B,
  };
  assert.throws(() => fingerprintFinding(proxied(minimal)), { code: "INVALID_FLOW_FINDING" });
  assert.deepEqual(trapCounts, {
    get: 0,
    getOwnPropertyDescriptor: 0,
    getPrototypeOf: 0,
    ownKeys: 0,
  });

  const nested = finding({ steps: proxied(["Open the reviewed route."]) });
  assert.throws(() => fingerprintFinding(nested), { code: "INVALID_FLOW_FINDING" });
  assert.deepEqual(trapCounts, {
    get: 0,
    getOwnPropertyDescriptor: 0,
    getPrototypeOf: 0,
    ownKeys: 0,
  });
});

test("literal replacement characters remain reportable while malformed UTF-8 bytes fail closed", async (t) => {
  const replacementFinding = finding({
    symptom: "place name renders \uFFFD",
    evidenceSignature: shaFor("replacement-character"),
  });
  assert.doesNotThrow(() => fingerprintFinding(replacementFinding));
  await assertFindingAccepted(t, replacementFinding);

  const normalizedSymptom = replacementFinding.symptom.toLowerCase();
  const canonicalReplacementFinding = { ...replacementFinding, symptom: normalizedSymptom };
  const persisted = {
    fingerprint: fingerprintFinding(replacementFinding),
    findingDigest: `sha256:${digestJson(canonicalReplacementFinding)}`,
    finding: canonicalReplacementFinding,
    state: "pending_sync",
    attempts: 0,
    createdAt: NOW,
    updatedAt: NOW,
    pendingAction: null,
    mutationReceipt: null,
    remote: null,
    failure: null,
  };

    ...baseIo,
    async readJson(path) {
      events.push("outbox-read");

      events.push("outbox-write");
      return baseIo.writeJsonAtomic(path, value);
    },

      events.push("lock-open");
      return open(...args);
    },

    unlink,
  };

    io,

    await transaction.commit(transaction.snapshot);
  }));
  assert.ok(events.indexOf("lock-open") < events.indexOf("outbox-read"));
  assert.ok(events.indexOf("lock-open") < events.indexOf("outbox-write"));
});

      () => service.withOutboxTransaction(transactionInput((transaction) =>
        transaction.commit(nextOutbox))),
      { code: "INVALID_FLOW_OUTBOX" },
    );
    await assert.rejects(() => stat(outboxPath), { code: "ENOENT" });
  }
});

test("safe read recovery preserves a known remote in pending_sync", async (t) => {
  const { io, outboxPath, service } = await fixture(t);
  const initial = await enqueue(service, finding());
  const remote = remoteIdentity();
  const current = (await service.readOutbox()).entries[0];
  assert.equal(current.fingerprint, initial.fingerprint);
  await io.writeJsonAtomic(outboxPath, {
    version: 1,
    entries: [{
      ...current,
      state: "failed",
      remote,
      failure: {
        code: "FLOW_READ_FAILED",
        failedAt: NOW,
        recoveryClass: "RETRY_SAFE_READ",
      },
    }],
  });
  const recovered = await service.recoverOutbox({
    fingerprint: initial.fingerprint,
    ownerId: OWNER,
    now: () => new Date(NOW),
    leaseMs: 30_000,
  });
  assert.equal(recovered.state, "pending_sync");
  assert.deepEqual(recovered.remote, remote);
  assert.equal(recovered.failure, null);

  const resumed = await service.withOutboxTransaction(transactionInput((transaction) => {
    const pending = transaction.snapshot.entries[0];
    const syncing = {
      ...pending,
      state: "syncing",
      attempts: pending.attempts + 1,
      pendingAction: proposedAction(pending, {
        kind: "get_issue",
        issueId: remote.id,
        actionSeed: "known-remote-safe-read",
      }),
    };
    return transaction.commit({ version: 1, entries: [syncing] });
  }));
  assert.equal(resumed.entries[0].state, "syncing");
  assert.deepEqual(resumed.entries[0].remote, remote);
  assert.equal(resumed.entries[0].pendingAction.manifest.kind, "get_issue");
});

test("transaction rejects a replaced lock inode and never unlinks the replacement", async (t) => {
  const { lockPath, service } = await fixture(t);
  const displacedPath = `${lockPath}.displaced`;
  await assert.rejects(
    () => service.withOutboxTransaction(transactionInput(async (transaction) => {
      await rename(lockPath, displacedPath);
      await writeFile(lockPath, `${JSON.stringify({
        ownerId: "other-worker",
        acquiredAt: NOW,
        expiresAt: "2026-08-26T00:01:00.000Z",
      })}\n`, { mode: 0o600 });
      await transaction.commit(transaction.snapshot);
    })),
    { code: "OUTBOX_LOCK_REPLACED" },
  );
  assert.equal(JSON.parse(await readFile(lockPath, "utf8")).ownerId, "other-worker");
});

test("enqueue durably replaces state at 0600 and leaves no transaction residue", async (t) => {
  const { root, outboxPath, lockPath, service } = await fixture(t);
  const persisted = await enqueue(service, finding());
  assert.equal(persisted.state, "pending_sync");
  assert.equal((await lstat(outboxPath)).mode & 0o777, 0o600);
  assert.deepEqual((await service.readOutbox()).entries, [persisted]);
  await assert.rejects(() => stat(lockPath), { code: "ENOENT" });
  assert.deepEqual(
    (await readdir(root)).filter((name) => name.includes(".tmp") || name.includes(".rollback")),
    [],
  );
});

test("previous outbox survives an injected atomic rename failure", async (t) => {
  const base = await fixture(t);
  await enqueue(base.service, finding());
  const before = await readFile(base.outboxPath);
  const failingIo = createJsonIo({
    privateRoot: base.root,
    operations: {
      async rename() {
        const error = new Error("synthetic rename failure");
        error.code = "EIO";
        throw error;
      },
    },
  });
  const failingService = createFlowOutboxService({
    io: failingIo,
    outboxPath: base.outboxPath,
    lockPath: base.lockPath,
  });
  await assert.rejects(
    () => enqueue(failingService, finding({ evidenceSignature: shaFor("new-finding") })),
    { code: "ATOMIC_WRITE_FAILED" },
  );
  assert.deepEqual(await readFile(base.outboxPath), before);
  assert.deepEqual(
    (await readdir(base.root)).filter((name) => name.includes(".tmp") || name.includes(".rollback")),
    [],
  );
});
