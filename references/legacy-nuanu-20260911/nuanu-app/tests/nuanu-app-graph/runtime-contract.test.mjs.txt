function canonical(value) {
  if (Array.isArray(value)) return value.map(canonical);
  if (!value || typeof value !== "object") return value;
  return Object.fromEntries(Object.keys(value).sort().map((key) => [key, canonical(value[key])]));
}

function digest(value) {
  return createHash("sha256").update(JSON.stringify(canonical(value))).digest("hex");
}

function without(object, key) {
  return Object.fromEntries(Object.entries(object).filter(([candidate]) => candidate !== key));
}

async function jsonFixture(name) {
  return JSON.parse(await readFile(join(FIXTURES, name), "utf8"));
}

async function syntheticBundle() {
  const capturePlan = await jsonFixture("selector-capture-plan.synthetic.json");
  capturePlan.capturePlanDigest = digest(without(capturePlan, "capturePlanDigest"));

  const catalog = normalizeRuntimeObservation({
    capturedAt: "2026-08-26T00:10:00.000Z",
    routes: ["/", "/__napp_runtime_probe__"],
    rootCandidates: { state: "root", route: "/", surfaces: [], controls: [], taxonomy: [] },
    probe: { route: "/__napp_runtime_probe__", status: 404 },
    apiHosts: [],
    dependencies: [],
    manifest: {},
  });

  const observation = await jsonFixture("selector-observation.synthetic.json");
  observation.runtimeDigest = catalog.runtimeDigest;
  observation.capturePlanDigest = capturePlan.capturePlanDigest;
  const aliases = new Map();
  for (const state of observation.states) {
    for (const record of state.records) {
      const alias = record.evidenceId;
      record.evidenceId = digest({
        state: record.state,
        role: record.role,
        name: record.name,
        matchCount: record.matchCount,
      });
      aliases.set(alias, record.evidenceId);
    }
    state.postcondition.evidenceId = aliases.get(state.postcondition.evidenceId);
  }
  observation.observationDigest = digest(without(observation, "observationDigest"));

  const bindings = await jsonFixture("selector-bindings.synthetic.json");
  bindings.observationDigest = observation.observationDigest;
  bindings.capturePlanDigest = capturePlan.capturePlanDigest;
  for (const selector of Object.values(bindings.selectors)) {
    if (selector.status === "AVAILABLE") selector.evidenceId = aliases.get(selector.evidenceId);
  }
  for (const items of Object.values(bindings.inventory)) {
    for (const item of items) item.evidenceId = aliases.get(item.evidenceId);
  }
  bindings.bindingDigest = digest(without(bindings, "bindingDigest"));

  const contract = buildReviewedRuntimeContract({ catalog, capturePlan, observation, bindings });
  return { catalog, capturePlan, observation, bindings, contract };
}

test("all Nuanu App desktop and mobile project lanes block service workers", async (t) => {
  const directory = await mkdtemp(join(tmpdir(), "napp-project-use-"));
  t.after(() => rm(directory, { recursive: true, force: true }));
  const reporter = join(directory, "project-use-reporter.mjs");
  await writeFile(reporter, `
    export default class ProjectUseReporter {
      onBegin(config) {
        const uses = config.projects
          .filter((project) => project.name.startsWith('nuanu-app'))
          .map((project) => ({ name: project.name, serviceWorkers: project.use.serviceWorkers ?? null }))
          .sort((left, right) => left.name.localeCompare(right.name));
        process.stdout.write('NUANU_PROJECT_USE=' + JSON.stringify(uses) + '\\n');
      }
    }
  `, "utf8");

  const { stdout } = await execFileAsync(process.execPath, [
    PLAYWRIGHT_CLI,
    "test",
    "--list",
    "--grep",
    "NAPP-SAFE-002",
    `--reporter=${reporter}`,
  ], {
    cwd: QA_ROOT,
    env: {
      ...process.env,
      NUANU_APP_BASE_URL: "https://app.nuanu.com",
      NUANU_APP_FIREFOX: "1",
      NUANU_APP_WEBKIT: "1",
      NUANU_APP_MOBILE: "1",
    },
  });
  const line = stdout.split("\n").find((entry) => entry.startsWith("NUANU_PROJECT_USE="));
  assert.deepEqual(JSON.parse(line.slice("NUANU_PROJECT_USE=".length)), [
    { name: "nuanu-app", serviceWorkers: "block" },
    { name: "nuanu-app-android-chrome", serviceWorkers: "block" },
    { name: "nuanu-app-firefox", serviceWorkers: "block" },
    { name: "nuanu-app-ios-webkit", serviceWorkers: "block" },
    { name: "nuanu-app-webkit", serviceWorkers: "block" },
  ]);
});

test("typed inventory preserves the reviewed evidence-ID order instead of re-sorting by display name", async () => {
  const bundle = await syntheticBundle();
  const record = {
    evidenceId: digest({ state: "root", role: "option", name: "Z synthetic item", matchCount: 1 }),
    state: "root",
    role: "option",
    name: "Z synthetic item",
    matchCount: 1,
  };
  const root = bundle.observation.states.find((state) => state.state === "root");
  root.records.push(record);
  root.records.sort((left, right) =>
    [left.state, left.role, left.name, left.matchCount, left.evidenceId].join("\u0000")
      .localeCompare([right.state, right.role, right.name, right.matchCount, right.evidenceId].join("\u0000")));
  bundle.observation.observationDigest = digest(without(bundle.observation, "observationDigest"));
  bundle.bindings.observationDigest = bundle.observation.observationDigest;
  bundle.bindings.inventory.categories.push({ evidenceId: record.evidenceId, matchCount: 1 });
  bundle.bindings.inventory.categories.sort((left, right) => left.evidenceId.localeCompare(right.evidenceId));
  bundle.bindings.bindingDigest = digest(without(bundle.bindings, "bindingDigest"));

  const contract = buildReviewedRuntimeContract(bundle);
  assert.deepEqual(contract.inventory.categories.map((item) => item.name), [
    "Z synthetic item",
    "Synthetic category",
  ]);
});

test("fixed loader classifies only absent or unreadable accepted contract as BLOCKED", async () => {
  const bundle = await syntheticBundle();
  const byName = new Map([
    ["runtime-contract.v1.json", bundle.contract],
    ["runtime-catalog.json", bundle.catalog],
    ["runtime-selector-observation.v1.json", bundle.observation],
    ["selector-capture-plan.v1.json", bundle.capturePlan],
    ["selector-bindings.v1.json", bundle.bindings],
  ]);
  const names = [];
  const loader = createRuntimeContractLoader({
    now: () => NOW,
    readFile: async (path) => {
      const name = path.split("/").at(-1);
      names.push(name);
      if (!byName.has(name)) throw Object.assign(new Error("not found"), { code: "ENOENT" });
      return JSON.stringify(byName.get(name));
    },
  });

  assert.deepEqual(await loader(), { status: "READY", contract: bundle.contract });
  assert.deepEqual(new Set(names), new Set(byName.keys()));
  assert.equal(names.some((name) => name.includes("..")), false);

  byName.delete("runtime-contract.v1.json");
  assert.deepEqual(await loader(), { status: "BLOCKED", reason: "RUNTIME_CONTRACT_MISSING" });

  byName.set("runtime-contract.v1.json", bundle.contract);
  const unreadable = createRuntimeContractLoader({
    now: () => NOW,
    readFile: async () => { throw Object.assign(new Error("private detail"), { code: "EACCES" }); },
  });
  assert.deepEqual(await unreadable(), { status: "BLOCKED", reason: "RUNTIME_CONTRACT_UNREADABLE" });
});

test("fixed loader maps malformed JSON and binding drift to non-sensitive UNVERIFIED reasons", async () => {
  const bundle = await syntheticBundle();
  const jsonByName = new Map([
    ["runtime-contract.v1.json", JSON.stringify(bundle.contract)],
    ["runtime-catalog.json", JSON.stringify(bundle.catalog)],
    ["runtime-selector-observation.v1.json", JSON.stringify(bundle.observation)],
    ["selector-capture-plan.v1.json", JSON.stringify(bundle.capturePlan)],
    ["selector-bindings.v1.json", JSON.stringify(bundle.bindings)],
  ]);
  const loader = createRuntimeContractLoader({
    now: () => NOW,
    readFile: async (path) => jsonByName.get(path.split("/").at(-1)),
  });

  jsonByName.set("runtime-contract.v1.json", "{private-value");
  assert.deepEqual(await loader(), {
    status: "UNVERIFIED",
    reason: "RUNTIME_CONTRACT_MALFORMED_JSON",
  });

  jsonByName.set("runtime-contract.v1.json", JSON.stringify(bundle.contract));
  const bindings = structuredClone(bundle.bindings);
  bindings.bindingDigest = "f".repeat(64);
  jsonByName.set("selector-bindings.v1.json", JSON.stringify(bindings));
  const gate = await loader();
  assert.deepEqual(gate, {
    status: "UNVERIFIED",
    reason: "RUNTIME_CONTRACT_DIGEST_MISMATCH",
  });
  assert.equal(JSON.stringify(gate).includes("private-value"), false);
});

test("fixed loader maps bounded private-I/O symlink race size JSON and missing failures without detail leakage", async () => {
  const bundle = await syntheticBundle();
  const byName = new Map([
    ["runtime-contract.v1.json", bundle.contract],
    ["runtime-catalog.json", bundle.catalog],
    ["runtime-selector-observation.v1.json", bundle.observation],
    ["selector-capture-plan.v1.json", bundle.capturePlan],
    ["selector-bindings.v1.json", bundle.bindings],
  ]);
  const loaderFor = (failedName, code) => createRuntimeContractLoader({
    now: () => NOW,
    readJson: async (path) => {
      const name = path.split("/").at(-1);
      if (name === failedName) throw Object.assign(new Error("private path and value"), { code });
      return structuredClone(byName.get(name));
    },
  });

  for (const code of [
    "MISSING_PRIVATE_ROOT",
    "MISSING_PARENT_DIRECTORY",
  ]) {
    assert.deepEqual(await loaderFor("runtime-contract.v1.json", code)(), {
      status: "BLOCKED",
      reason: "RUNTIME_CONTRACT_MISSING",
    });
  }
  for (const code of [
    "SYMLINK_PATH_REJECTED",
    "PATH_RACE_DETECTED",
    "JSON_ARTIFACT_TOO_LARGE",
    "JSON_READ_FAILED",
  ]) {
    assert.deepEqual(await loaderFor("runtime-contract.v1.json", code)(), {
      status: "BLOCKED",
      reason: "RUNTIME_CONTRACT_UNREADABLE",
    });
  }
  assert.deepEqual(await loaderFor("runtime-contract.v1.json", "MISSING_ARTIFACT")(), {
    status: "BLOCKED",
    reason: "RUNTIME_CONTRACT_MISSING",
  });
  assert.deepEqual(await loaderFor("runtime-contract.v1.json", "INVALID_JSON_ARTIFACT")(), {
    status: "UNVERIFIED",
    reason: "RUNTIME_CONTRACT_MALFORMED_JSON",
  });
  assert.deepEqual(await loaderFor("runtime-selector-observation.v1.json", "MISSING_ARTIFACT")(), {
    status: "UNVERIFIED",
    reason: "RUNTIME_SELECTOR_OBSERVATION_MISSING",
  });
  const siblingGate = await loaderFor("selector-bindings.v1.json", "SYMLINK_PATH_REJECTED")();
  assert.deepEqual(siblingGate, {
    status: "UNVERIFIED",
    reason: "RUNTIME_SELECTOR_OBSERVATION_MISMATCH",
  });
  assert.equal(JSON.stringify(siblingGate).includes("private path"), false);
});
