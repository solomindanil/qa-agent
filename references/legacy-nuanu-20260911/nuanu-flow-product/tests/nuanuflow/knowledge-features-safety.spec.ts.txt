const nestedErrorMessages = (error: unknown): string[] => {
  if (!(error instanceof Error)) return [String(error)];
  return [
    error.message,
    ...(error instanceof AggregateError
      ? error.errors.flatMap(nestedErrorMessages)
      : []),
  ];
};

const artifactCleanupHarness = async (
  page: Page,
  options: {
    body?: unknown;
    detailStatus?: number;
    deleteStatus?: number;
    suffix: string;
  },
) => {
  const detailEndpoint =
    `${SYNTHETIC_ORIGIN}/artifacts/${IDS.artifact}/`;
  const collectionEndpoint = `${SYNTHETIC_ORIGIN}/artifacts/`;
  const uiPath = `${SYNTHETIC_ORIGIN}/artifact-ui-${options.suffix}/`;
  let exists = options.detailStatus !== 404;
  const calls: string[] = [];
  const body = options.body ?? privateArtifactBody('pending');
  const requestContext = {
    get: async (url: string) => {
      if (url === detailEndpoint) {
        calls.push('GET detail');
        return syntheticResponse(exists ? (options.detailStatus ?? 200) : 404, body);
      }
      if (url === collectionEndpoint) {
        calls.push('GET collection');
        return syntheticResponse(200, exists ? [body] : []);
      }
      throw new Error(`Unexpected synthetic GET: ${url}`);
    },
    delete: async (url: string) => {
      if (url !== detailEndpoint) {
        throw new Error(`Unexpected synthetic DELETE: ${url}`);
      }
      calls.push('DELETE detail');
      const status = options.deleteStatus ?? 204;
      if (status === 204) exists = false;
      return syntheticResponse(status);
    },
  };
  await page.route(uiPath, async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'text/html',
      body: exists ? `<div>${PRIVATE_ARTIFACT_NAME}</div>` : '<div>empty</div>',
    });
  });
  return {
    calls,
    created: privateArtifact(),
    detailEndpoint,
    collectionEndpoint,
    uiPath,
    requestContext,
    assertSandbox: async () => {
      calls.push('assertSandbox');
    },
    exists: () => exists,
  };
};

const savedViewPomHarness = async (
  page: Page,
  options: {
    createBody?: unknown;
    createLeavesRecord?: boolean;
    detailResponse?: (
      ordinal: number,
      deleted: boolean,
    ) => ReturnType<typeof syntheticResponse>;
  } = {},
) => {
  const viewPath =
    `/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/views/`;
  const apiPath =
    `/be/api/workspaces/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/views/`;
  const calls: string[] = [];
  let collectionResidual = false;
  let uiResidual = false;
  let deleted = false;
  let detailOrdinal = 0;

  await page.route(`**${viewPath}**`, async (route) => {
    calls.push('GET UI');
    await route.fulfill({
      status: 200,
      contentType: 'text/html',
      body: [
        '<div>Views</div>',
        ...(uiResidual
          ? [`<div>${IDS.view}</div>`, `<div>${OWNED_NAME}</div>`]
          : []),
      ].join(''),
    });
  });
  await page.route(`**${apiPath}`, async (route) => {
    if (route.request().method() !== 'POST') {
      await route.fulfill({ status: 405 });
      return;
    }
    collectionResidual = options.createLeavesRecord ?? true;
    uiResidual = options.createLeavesRecord ?? true;
    await route.fulfill({
      status: 201,
      contentType: 'application/json',
      body: JSON.stringify(options.createBody ?? savedViewBody()),
    });
  });

  const requestContext = {
    get: async (url: string) => {
      const pathname = new URL(url).pathname;
      if (pathname === apiPath) {
        calls.push('GET collection');
        return syntheticResponse(
          200,
          collectionResidual ? [savedViewBody()] : [],
        );
      }
      if (pathname === `${apiPath}${IDS.view}/`) {
        calls.push('GET detail');
        const ordinal = detailOrdinal++;
        return options.detailResponse?.(ordinal, deleted) ??
          (deleted
            ? syntheticResponse(200, { deleted: true })
            : syntheticResponse(200, savedViewBody()));
      }
      throw new Error(`Unexpected saved-view POM GET: ${url}`);
    },
    delete: async (url: string) => {
      if (new URL(url).pathname !== `${apiPath}${IDS.view}/`) {
        throw new Error(`Unexpected saved-view POM DELETE: ${url}`);
      }
      calls.push('DELETE detail');
      deleted = true;
      collectionResidual = false;
      uiResidual = false;
      return syntheticResponse(204);
    },
  };
  const assertSandbox = async () => {
    calls.push('assertSandbox');
  };
  return {
    calls,
    knowledge: new KnowledgeFeaturesPage(
      page,
      assertSandbox,
      requestContext as unknown as APIRequestContext,
    ),
    setCollectionResidual: (value: boolean) => {
      collectionResidual = value;
    },
    setUiResidual: (value: boolean) => {
      uiResidual = value;
    },
  };
};

const artifactBatchHarness = async (
  page: Page,
  created: CreatedArtifact[],
  options: { failDeleteId?: string; failGetId?: string; suffix: string },
) => {
  const live = new Map(created.map((artifact) => [artifact.id, artifact]));
  const detailRoot = `${SYNTHETIC_ORIGIN}/artifact-batch/`;
  const collectionEndpoint = `${SYNTHETIC_ORIGIN}/artifact-batch/`;
  const uiPath = `${SYNTHETIC_ORIGIN}/artifact-batch-ui-${options.suffix}/`;
  const calls: string[] = [];
  const resolved: string[] = [];
  const requestContext = {
    get: async (url: string) => {
      if (url === collectionEndpoint) {
        calls.push('GET collection');
        return syntheticResponse(
          200,
          [...live.values()].map(({ id }) =>
            privateArtifactBody('processing', { id }),
          ),
        );
      }
      const id = url.slice(detailRoot.length, -1);
      calls.push(`GET ${id}`);
      if (id === options.failGetId) {
        throw new Error(`Synthetic detail GET failed for ${id}.`);
      }
      return live.has(id)
        ? syntheticResponse(200, privateArtifactBody('processing', { id }))
        : syntheticResponse(404);
    },
    delete: async (url: string) => {
      const id = url.slice(detailRoot.length, -1);
      calls.push(`DELETE ${id}`);
      if (id === options.failDeleteId) return syntheticResponse(500);
      return syntheticResponse(live.delete(id) ? 204 : 404);
    },
  };
  await page.route(uiPath, async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'text/html',
      body: [...live.values()]
        .map(({ name }) => `<div>${name}</div>`)
        .join(''),
    });
  });
  return {
    calls,
    collectionEndpoint,
    detailEndpoint: (id: string) => `${detailRoot}${id}/`,
    live,
    onResolved: (artifact: CreatedArtifact) => resolved.push(artifact.id),
    requestContext,
    resolved,
    uiPath,
  };
};
