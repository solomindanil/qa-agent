const responseAtEndpoint = (body: BodyInit | null, init: ResponseInit): Response => {
  const response = new Response(body, init);
  Object.defineProperty(response, 'url', {
    configurable: false,
    enumerable: false,
    value: NUANU_FLOW_MCP_ENDPOINT,
  });
  return response;
};

const jsonRpcResponse = (
  id: number,
  result: unknown,
  headers: Record<string, string> = {},
) =>
  responseAtEndpoint(JSON.stringify({ jsonrpc: '2.0', id, result }), {
    status: 200,
    headers: { 'content-type': 'application/json', ...headers },
  });

const sseResponse = (events: readonly string[]) =>
  responseAtEndpoint(events.map((event) => `${event}\r\n\r\n`).join(''), {
    status: 200,
    headers: { 'content-type': 'text/event-stream; charset=utf-8' },
  });

const syntheticProtocolFetch = (
  searchResponse: (requestId: number) => Response,
  initializeResponse?: (requestId: number) => Response,
): FetchLike =>
  async (input, init) => {
    expect(String(input)).toBe(NUANU_FLOW_MCP_ENDPOINT);
    const body = JSON.parse(String(init?.body)) as Record<string, unknown>;
    const id = body.id;
    if (body.method === 'initialize' && typeof id === 'number') {
      return (
        initializeResponse?.(id) ??
        jsonRpcResponse(
          id,
          {
            protocolVersion: '2025-06-18',
            capabilities: {},
            serverInfo: { name: 'synthetic-server', version: '1.0.0' },
          },
          { 'mcp-session-id': 'synthetic-session' },
        )
      );
    }
    if (body.method === 'notifications/initialized') {
      return responseAtEndpoint(null, { status: 202 });
    }
    if (body.method === 'tools/call' && typeof id === 'number') {
      return searchResponse(id);
    }
    throw new Error('Unexpected synthetic protocol request.');
  };

const withImmediateInternalTimers = async <T>(run: () => Promise<T>): Promise<T> => {
  const nativeSetTimeout = globalThis.setTimeout;
  const nativeClearTimeout = globalThis.clearTimeout;
  globalThis.setTimeout = ((
    handler: (...args: any[]) => void,
    _delay?: number,
    ...args: any[]
  ) =>
    nativeSetTimeout(handler, 0, ...args)) as typeof setTimeout;
  let guard: ReturnType<typeof setTimeout> | undefined;
  try {
    const pending = run();
    return await Promise.race([
      pending,
      new Promise<T>((_, reject) => {
        guard = nativeSetTimeout(
          () => reject(new Error('synthetic-deadline-guard')),
          250,
        );
      }),
    ]);
  } finally {
    if (guard !== undefined) nativeClearTimeout(guard);
    globalThis.setTimeout = nativeSetTimeout;
  }
};

test('normalizes hostile constructor inputs into local safe errors', () => {
  const fake = neverFetch();
  const tokenProvider = () => 'synthetic-token';
  const accessor = Object.create(null) as Record<string, unknown>;
  Object.defineProperties(accessor, {
    workspace: { enumerable: true, value: 'nf-qa-sandbox' },
    tokenProvider: {
      enumerable: true,
      get: () => {
        throw new Error('constructor-accessor-sentinel');
      },
    },
  });
  const reflectionProxy = new Proxy(
    {},
    {
      ownKeys: () => {
        throw new Error('constructor-proxy-sentinel');
      },
    },
  );
  const symbolKey = {
    workspace: 'nf-qa-sandbox',
    tokenProvider,
    [Symbol('secret')]: true,
  };

  for (const options of [
    null,
    7,
    [],
    new Date(0),
    accessor,
    reflectionProxy,
    symbolKey,
    { workspace: 'nf-qa-sandbox', tokenProvider, logger: () => {} },
    { workspace: 'nf-qa-sandbox', tokenProvider, headers: {} },
    { workspace: 'nf-qa-sandbox', tokenProvider, proxy: 'https://proxy.invalid' },

    let captured: unknown;
    try {
      new NuanuFlowMcpClient(options as never);
    } catch (error) {
      captured = error;
    }
    expect(captured).toBeInstanceOf(NuanuMcpError);
    expect(captured).toMatchObject({ code: 'ARGUMENT_REFUSED' });
    expect(String(captured)).not.toMatch(/constructor-(?:accessor|proxy)-sentinel/);
  }
  expect(fake.calls()).toBe(0);
});

test('poisons a client when the current per-request token is reflected in session, tool name, or schema key', async () => {
  const reflectedToken = `plane_api_${'ab'.repeat(20)}`;
  const consoleCalls: unknown[][] = [];
  const originalConsole = {
    debug: console.debug,
    error: console.error,
    info: console.info,
    log: console.log,
    warn: console.warn,
  };
  for (const method of Object.keys(originalConsole) as Array<keyof typeof originalConsole>) {
    console[method] = ((...args: unknown[]) => {
      consoleCalls.push(args);
    }) as never;
  }
  try {
    for (const reflection of [
      'session',
      'session-prefixed',
      'name',
      'name-prefixed',
      'key',
      'key-prefixed',
    ] as const) {
      let fetchCalls = 0;
      let tokenCalls = 0;
      const fallback = syntheticProtocolFetch(
        (id) =>
          jsonRpcResponse(id, {
            content: [
              {
                type: 'text',
                text: JSON.stringify({
                  tools: [
                    {
                      name:
                        reflection === 'name'
                          ? reflectedToken
                          : reflection === 'name-prefixed'
                            ? `tool_${reflectedToken}`
                            : 'get_work_item',
                      inputSchema: {
                        type: 'object',
                        properties:
                          reflection === 'key' || reflection === 'key-prefixed'
                            ? {
                                [reflection === 'key'
                                  ? reflectedToken
                                  : `field_${reflectedToken}`]: { type: 'string' },
                              }
                            : {},
                        required:
                          reflection === 'key' || reflection === 'key-prefixed'
                            ? [
                                reflection === 'key'
                                  ? reflectedToken
                                  : `field_${reflectedToken}`,
                              ]
                            : [],
                      },
                    },
                  ],
                }),
              },
            ],
            isError: false,
          }),
        (id) =>
          jsonRpcResponse(
            id,
            {
              protocolVersion: '2025-06-18',
              capabilities: {},
              serverInfo: { name: 'synthetic-server', version: '1.0.0' },
            },
            reflection === 'session' || reflection === 'session-prefixed'
              ? {
                  'mcp-session-id':
                    reflection === 'session'
                      ? reflectedToken
                      : `sid_${reflectedToken}`,
                }
              : { 'mcp-session-id': 'synthetic-session' },
          ),
      );
      const client = new NuanuFlowMcpClient({
        workspace: 'nf-qa-sandbox',
        tokenProvider: () => {
          tokenCalls += 1;
          return reflectedToken;
        },
        fetchImpl: async (input, init) => {
          fetchCalls += 1;
          return fallback(input, init);
        },
      });

      let captured: unknown;
      try {
        await client.searchTools('work-item-parity');
      } catch (error) {
        captured = error;
      }
      expect(captured, reflection).toMatchObject({
        code: 'TOKEN_REFLECTION_REFUSED',
      });
      const callsAtRefusal = { fetchCalls, tokenCalls };
      await expect(client.searchTools('work-item-parity'), reflection).rejects.toMatchObject({
        code: 'TOKEN_REFLECTION_REFUSED',
      });
      expect({ fetchCalls, tokenCalls }, reflection).toEqual(callsAtRefusal);
      expect(String(captured), reflection).not.toContain(reflectedToken);
      expect(JSON.stringify(captured), reflection).not.toContain(reflectedToken);
      expect(JSON.stringify(client), reflection).not.toContain(reflectedToken);
    }
  } finally {
    Object.assign(console, originalConsole);
  }
  expect(consoleCalls).toEqual([]);
});

test('refuses a retained session that matches a freshly rotated request token before fetch', async () => {
  const initialToken = `plane_api_${'11'.repeat(20)}`;
  const rotatedToken = `plane_api_${'22'.repeat(20)}`;
  const tokens = [initialToken, rotatedToken];
  let tokenCalls = 0;
  let fetchCalls = 0;
  const client = new NuanuFlowMcpClient({
    workspace: 'nf-qa-sandbox',
    tokenProvider: () => tokens[tokenCalls++] ?? rotatedToken,
    fetchImpl: async (_input, init) => {
      fetchCalls += 1;
      const body = JSON.parse(String(init?.body)) as Record<string, unknown>;
      if (body.method !== 'initialize' || typeof body.id !== 'number') {
        throw new Error('rotated-session-must-be-refused-before-second-fetch');
      }
      return jsonRpcResponse(
        body.id,
        {
          protocolVersion: '2025-06-18',
          capabilities: {},
          serverInfo: { name: 'synthetic-server', version: '1.0.0' },
        },
        { 'mcp-session-id': rotatedToken },
      );
    },
  });

  await expect(client.searchTools('work-item-parity')).rejects.toMatchObject({
    code: 'TOKEN_REFLECTION_REFUSED',
  });
  expect({ tokenCalls, fetchCalls }).toEqual({ tokenCalls: 2, fetchCalls: 1 });
  await expect(client.searchTools('work-item-parity')).rejects.toMatchObject({
    code: 'TOKEN_REFLECTION_REFUSED',
  });
  expect({ tokenCalls, fetchCalls }).toEqual({ tokenCalls: 2, fetchCalls: 1 });
});

test('normalizes a session-fingerprint entropy failure after clearing initialization state', async () => {
  const nodeRequire = createRequire(import.meta.url);
  const crypto = nodeRequire('node:crypto') as typeof import('node:crypto');
  const nativeRandomBytes = crypto.randomBytes;
  let randomBytesCalls = 0;
  let fetchCalls = 0;
  crypto.randomBytes = ((size: number) => {
    randomBytesCalls += 1;
    if (randomBytesCalls === 3) {
      throw new Error('session-fingerprint-entropy-sentinel');
    }
    return nativeRandomBytes(size);
  }) as typeof crypto.randomBytes;
  syncBuiltinESMExports();

  let captured: unknown;
  try {
    const fallback = syntheticProtocolFetch(() => {
      throw new Error('search-must-not-run-after-entropy-failure');
    });
    const client = new NuanuFlowMcpClient({
      workspace: 'nf-qa-sandbox',
      tokenProvider: () => 'synthetic-token',
      fetchImpl: async (input, init) => {
        fetchCalls += 1;
        return fallback(input, init);
      },
    });
    captured = await client
      .searchTools('work-item-parity')
      .catch((error: unknown) => error);
    await expect(client.searchTools('work-item-parity')).rejects.toMatchObject({
      code: 'RESULT_SCHEMA_MISMATCH',
      operation: 'client.initialize',
    });
  } finally {
    crypto.randomBytes = nativeRandomBytes;
    syncBuiltinESMExports();
  }

  expect(captured).toMatchObject({
    code: 'RESULT_SCHEMA_MISMATCH',
    operation: 'client.initialize',
  });
  expect({ randomBytesCalls, fetchCalls }).toEqual({
    randomBytesCalls: 3,
    fetchCalls: 1,
  });
  expect(String(captured)).not.toContain('session-fingerprint-entropy-sentinel');
  expect(JSON.stringify(captured)).not.toContain(

  );
});
