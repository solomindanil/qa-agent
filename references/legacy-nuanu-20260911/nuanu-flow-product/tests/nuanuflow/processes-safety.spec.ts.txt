const parseProcess = (
  body: unknown,
  overrides: Partial<{
    requestedName: string;
    requestOrdinal: number;
    projectId: string;
  }> = {},
) =>
  parseCreatedProcess(body, {
    requestedName: overrides.requestedName ?? PROCESS_NAME,
    requestOrdinal: overrides.requestOrdinal ?? 0,
    workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
    projectId: overrides.projectId,
    expectedDescription: 'Synthetic inert BPMN process.',
    expectedCategory: 'operational',
    expectedIsActive: false,

test('accepts only the exact observed process-create response schema', () => {
  expect(parseProcess(processBody())).toEqual({
    kind: 'process',
    id: IDS.process,
    name: PROCESS_NAME,
    description: 'Synthetic inert BPMN process.',
    category: 'operational',
    isActive: false,
    workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
    requestOrdinal: 0,
  });

  const rejected: unknown[] = [
    processBody({ id: 'not-a-canonical-id' }),
    processBody({ name: `${QA_PREFIX} synthetic-run` }),
    processBody({ workspace_id: IDS.foreign }),
    processBody({ project_id: IDS.foreign }),
    processBody({ id: undefined, data: { id: IDS.process } }),
    processBody({ description: 'Changed response description.' }),
    processBody({ category: 'service' }),
    processBody({ is_active: true }),
    processBody({ data: { id: IDS.foreign } }),
    processBody({ payload: { id: IDS.foreign } }),
    processBody({ template: { uuid: IDS.foreign } }),
    processBody({
      envelope: {
        records: [{ process_id: IDS.foreign }],
      },
    }),
    processBody({
      metadata: [{ unrelatedValue: IDS.foreign }],
    }),
    processBody({
      payload: { workspace_id: PROCESS_SANDBOX_WORKSPACE_ID },
    }),
    processBody({ payload: { nested_id: 'not-canonical' } }),
    [processBody()],
  ];
  for (const body of rejected) {
    expect(() => parseProcess(body)).toThrow();
  }
  expect(() =>
    parseProcess(processBody({ project_id: SANDBOX.projectId }), {
      projectId: IDS.foreign,
    }),
  ).toThrow();

  expect(
    parseProcess(
      processBody({ project_id: SANDBOX.projectId }),
      { projectId: SANDBOX.projectId },
    ),
  ).toEqual(
    expect.objectContaining({
      id: IDS.process,
      workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
      projectId: SANDBOX.projectId,
    }),
  );
});

test('planned metadata transitions preserve trusted ownership for later autosave and cleanup', async ({
  page,
}) => {
  const endpoint = syntheticProcessDetailEndpoint(IDS.process);
  const familyPattern = `${SYNTHETIC_ORIGIN}/**/process-templates/**`;
  const allowedNames = new Set([PROCESS_NAME]);
  const created = parseProcess(processBody());
  const serverState = {
    name: PROCESS_NAME,
    description: 'Synthetic inert BPMN process.',
    category: 'operational',
  };
  let backendWrites = 0;
  await page.route(endpoint, async (route) => {
    backendWrites += 1;
    const body = route.request().postDataJSON() as Record<string, unknown>;
    if (typeof body.name === 'string') serverState.name = body.name;
    if (typeof body.description === 'string') {
      serverState.description = body.description;
    }
    if (typeof body.category === 'string') {
      serverState.category = body.category;
    }
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify(processBody(serverState)),
    });
  });
  await openSyntheticOwnedProcessPage(page);

  const patches = [
    { name: RENAMED_PROCESS_NAME },
    { description: 'Authorized description transition.' },
    { category: 'service' },
  ] satisfies ProcessMetadataPatch[];
  for (const patch of patches) {
    const result = await captureExactProcessMetadataUpdate({
      page,
      created,
      allowedNames,
      endpoint,
      familyPattern,
      patch,
      assertSandbox: async () => {},
      action: () =>
        page.evaluate(
          ({ requestEndpoint, exactPatch }) =>
            fetch(requestEndpoint, {
              method: 'PATCH',
              headers: { 'content-type': 'application/json' },
              body: JSON.stringify(exactPatch),
            }),
          { requestEndpoint: endpoint, exactPatch: patch },
        ),
      quietMs: 20,
      maxAdmissionMs: 100,
    });
    expect(result.capture.successfulEntries).toHaveLength(1);
    expect(result.capture.contractErrors).toEqual([]);
  }

  const autosave = await captureExactProcessSaves({
    page,
    created,
    allowedNames,
    assertSandbox: async () => {},
    action: () =>
      page.evaluate(
        ({ requestEndpoint }) =>
          fetch(requestEndpoint, {
            method: 'PATCH',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({ bpmn_xml: '<definitions />' }),
          }),
        { requestEndpoint: endpoint },
      ),
    validatePayload: () => {},
    quietMs: 20,
    maxAdmissionMs: 100,
  });

  expect(created).toEqual(
    expect.objectContaining({
      id: IDS.process,
      name: RENAMED_PROCESS_NAME,
      description: 'Authorized description transition.',
      category: 'service',
      requestOrdinal: 0,
    }),
  );
  expect(autosave.capture.successfulEntries).toHaveLength(1);
  expect(() =>
    assertExactProcessCleanupTarget(
      processBody(serverState),
      created,
      allowedNames,
    ),
  ).not.toThrow();
  expect(backendWrites).toBe(4);
});

const nestedErrorMessages = (error: unknown): string[] => {
  if (!(error instanceof Error)) return [String(error)];
  return [
    error.message,
    ...(error instanceof AggregateError
      ? error.errors.flatMap(nestedErrorMessages)
      : []),
  ];
};

const cleanupHarness = async (
  page: Page,
  options: {
    detailStatus?: Partial<Record<string, number>>;
    deleteStatus?: Partial<Record<string, number>>;
    tombstoneIds?: ReadonlySet<string>;
    retainedUiIds?: ReadonlySet<string>;
    retainedCollectionIds?: ReadonlySet<string>;
    foreignIds?: ReadonlySet<string>;
    stickyAfterDeleteIds?: ReadonlySet<string>;
  } = {},
) => {
  const created = [
    parseProcess(processBody(), { requestOrdinal: 0 }),
    parseProcess(processBody({ id: IDS.process2 }), {
      requestOrdinal: 1,
    }),
  ];
  const exists = new Set(created.map(({ id }) => id));
  const calls: string[] = [];
  const collectionEndpoint = `${SYNTHETIC_ORIGIN}/cleanup/process-templates/`;
  const detailEndpoint = (id: string) =>
    `${collectionEndpoint}${id}/`;
  const uiPath = `${SYNTHETIC_ORIGIN}/cleanup/process-ui/`;
  const bodyFor = (entity: CreatedProcess) =>
    processBody({
      id: entity.id,
      name: entity.name,
      description: entity.description,
      category: entity.category,
      is_active: entity.isActive,
      workspace_id: options.foreignIds?.has(entity.id)
        ? IDS.foreign
        : PROCESS_SANDBOX_WORKSPACE_ID,
      ...(entity.projectId === undefined
        ? {}
        : { project_id: entity.projectId }),
    });

  const requestContext = {
    get: async (url: string) => {
      if (url === collectionEndpoint) {
        calls.push('GET collection');
        return syntheticResponse(
          200,
          created
            .filter(
              ({ id }) =>
                exists.has(id) ||
                options.retainedCollectionIds?.has(id),
            )
            .map(bodyFor),
        );
      }
      const entity = created.find(({ id }) => detailEndpoint(id) === url);
      if (!entity) throw new Error(`Unexpected process GET: ${url}`);
      calls.push(`GET ${entity.id}`);
      if (options.tombstoneIds?.has(entity.id)) {
        return syntheticResponse(200, { deleted: true });
      }
      const status =
        options.detailStatus?.[entity.id] ??
        (exists.has(entity.id) ? 200 : 404);
      return syntheticResponse(status, bodyFor(entity));
    },
    delete: async (url: string) => {
      const entity = created.find(({ id }) => detailEndpoint(id) === url);
      if (!entity) throw new Error(`Unexpected process DELETE: ${url}`);
      calls.push(`DELETE ${entity.id}`);
      const status = options.deleteStatus?.[entity.id] ?? 204;
      if (
        status === 204 &&
        !options.stickyAfterDeleteIds?.has(entity.id)
      ) {
        exists.delete(entity.id);
      }
      return syntheticResponse(status);
    },
  };
  await page.route(uiPath, async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'text/html',
      body:
        '<button>Create Process</button>' +
        created
          .filter(
            ({ id }) =>
              exists.has(id) || options.retainedUiIds?.has(id),
          )
          .map(
            ({ id, name }) =>
              `<a href="${PROCESS_PATHS.detail(id)}">${name}</a>`,
          )
          .join(''),
    });
  });
  return {
    calls,
    created,
    exists,
    collectionEndpoint,
    detailEndpoint,
    uiPath,
    requestContext: requestContext as unknown as APIRequestContext,
    assertSandbox: async () => {
      calls.push('assertSandbox');
    },
  };
};

}) => {

    syntheticProcessDetailEndpoint(IDS.foreign);
  let foreignBackendWrites = 0;

  await page.route(exactEndpoint, async (route) => {
    ownedBackendWrites += 1;
    await route.fulfill({
      status: 200,
      contentType: 'application/json',
      body: JSON.stringify({
        id: IDS.process,
        workspace_id: PROCESS_SANDBOX_WORKSPACE_ID,
      }),
    });
  });

    page,

              method: 'PATCH',

              body: JSON.stringify({
                bpmn_xml: '<definitions />',
              }),
            }).catch(() => undefined),
          { endpoint: foreignEndpoint },
        ),
        page.evaluate(
          ({ endpoint }) =>
            fetch(endpoint, {
              method: 'PUT',
              headers: { 'content-type': 'application/json' },
              body: JSON.stringify({
                bpmn_xml: '<definitions />',
              }),
            }).catch(() => undefined),
          { endpoint: exactEndpoint },
        ),
      ]);
      await page.evaluate(
        ({ endpoint }) =>
          fetch(endpoint, {
            method: 'PATCH',
            headers: { 'content-type': 'application/json' },
            body: JSON.stringify({
              bpmn_xml: '<definitions />',
            }),
          }),
        { endpoint: exactEndpoint },
      );
    },
    validatePayload: () => {},
    quietMs: 60,
    maxAdmissionMs: 150,
  });

  expect(result.capture.attemptCount).toBe(1);
  expect(result.capture.successfulEntries).toEqual([

      requestOrdinal: 0,
      created: {
        processId: IDS.process,
        requestOrdinal: 0,
      },

  ]);
  expect(result.capture.unexpectedMutationAttempts).toEqual([
    {
      requestOrdinal: 0,

    },
  ]);
  expect(result.foreignWriteAttempts).toEqual([
    {
      method: 'PATCH',

      processId: IDS.foreign,
    },
  ]);
