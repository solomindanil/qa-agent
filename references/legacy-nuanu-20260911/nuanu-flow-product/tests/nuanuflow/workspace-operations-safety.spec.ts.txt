const IDS = Object.freeze({

const teamBody = (overrides: Record<string, unknown> = {}) => ({
  id: IDS.team,
  name: TEAM_NAME,
  description: TEAM_DESCRIPTION,
  workspace: WORKSPACE_OPERATIONS_SANDBOX_ID,

});

const syntheticResponse = (status: number, body: unknown = undefined) => ({
  status: () => status,
  json: async () => body,
});

const nestedErrorMessages = (error: unknown): string[] => {
  if (!(error instanceof Error)) return [String(error)];
  return [
    error.message,
    ...(error instanceof AggregateError
      ? error.errors.flatMap(nestedErrorMessages)
      : []),
  ];
};

const teamCleanupHarness = async (
  page: Page,
  options: {
    detailStatus?: Partial<Record<string, number>>;
    deleteStatus?: Partial<Record<string, number>>;
    extraCollectionRows?: Record<string, unknown>[];
  } = {},
) => {
  const secondName = TEAM_NAME;
  const created: CreatedTeam[] = [
    parseCreatedTeam(teamBody(), TEAM_NAME),
    parseCreatedTeam(
      teamBody({ id: IDS.other, name: secondName }),
      secondName,
    ),
  ];
  const exists = new Set(created.map(({ id }) => id));
  const calls: string[] = [];
  const collectionEndpoint = 'https://qa.invalid/teams/';
  const uiPath = 'https://qa.invalid/team-ui/';
  const detailEndpoint = (id: string) => `${collectionEndpoint}${id}/`;

  const requestContext = {
    get: async (url: string) => {
      if (url === collectionEndpoint) {
        calls.push('GET collection');
        return syntheticResponse(
          200,
          [
            ...created
              .filter(({ id }) => exists.has(id))
              .map(({ id, name }) => teamBody({ id, name })),
            ...(options.extraCollectionRows ?? []),
          ],
        );
      }
      const entity = created.find(({ id }) => detailEndpoint(id) === url);
      if (!entity) throw new Error(`Unexpected Team GET: ${url}`);
      calls.push(`GET ${entity.id}`);
      const status =
        options.detailStatus?.[entity.id] ??
        (exists.has(entity.id) ? 200 : 404);
      return syntheticResponse(
        status,
        teamBody({ id: entity.id, name: entity.name }),
      );
    },
    delete: async (url: string) => {
      const entity = created.find(({ id }) => detailEndpoint(id) === url);
      if (!entity) throw new Error(`Unexpected Team DELETE: ${url}`);
      calls.push(`DELETE ${entity.id}`);
      const status = options.deleteStatus?.[entity.id] ?? 204;
      if (status === 204) exists.delete(entity.id);
      return syntheticResponse(status);
    },
  };
  await page.route(uiPath, async (route) => {
    await route.fulfill({
      status: 200,
      contentType: 'text/html',
      body: created
        .filter(({ id }) => exists.has(id))
        .map(({ id, name }) => `<div>${id}</div><div>${name}</div>`)
        .join(''),
    });
  });
  return {
    calls,
    created,
    exists,
    collectionEndpoint,
    uiPath,
    detailEndpoint,
    requestContext: requestContext as unknown as APIRequestContext,
  };
};

};

test('registers restoration before mutation and restores the exact original value', async () => {
  const events: string[] = [];
  let value = false;
  let registeredRestore: (() => Promise<void>) | undefined;

  await runRestoredSandboxSetting({
    next: true,
    assertSandbox: async () => {
      events.push('guard');
    },
    read: async () => {
      events.push(`read:${value}`);
      return value;
    },
    mutate: async (next) => {
      events.push(`mutate:${next}`);
      value = next;
    },
    verify: async (expected) => {
      events.push(`verify:${expected}`);
      expect(value).toBe(expected);
    },
    registerRestoration: (restore) => {
      events.push('register');
      registeredRestore = restore;
    },
  });

  expect(value).toBe(false);
  expect(events).toEqual([
    'read:false',
    'register',
    'guard',
    'mutate:true',
    'verify:true',
    'guard',
    'mutate:false',
    'verify:false',
  ]);
  expect(registeredRestore).toBeDefined();
  await registeredRestore?.();
  expect(events).toHaveLength(8);
});

test('restores the original setting when changed-value verification fails', async () => {
  const events: string[] = [];
  let value = 'original';

  await expect(
    runRestoredSandboxSetting({
      next: 'changed',
      assertSandbox: async () => {
        events.push('guard');
      },
      read: async () => value,
      mutate: async (next) => {
        events.push(`mutate:${next}`);
        value = next;
      },
      verify: async (expected) => {
        events.push(`verify:${expected}`);
        if (expected === 'changed') {
          throw new Error('synthetic changed-value mismatch');
        }
        expect(value).toBe(expected);
      },
      registerRestoration: () => {
        events.push('register');
      },
    }),
  ).rejects.toThrow(/changed-value mismatch/);

  expect(value).toBe('original');
  expect(events).toEqual([
    'register',
    'guard',
    'mutate:changed',
    'verify:changed',
    'guard',
    'mutate:original',
    'verify:original',
  ]);
});

test('reports both the action and exact restoration failure', async () => {
  let mutationCount = 0;

  await expect(
    runRestoredSandboxSetting({
      next: true,
      assertSandbox: async () => {},
      read: async () => false,
      mutate: async () => {
        mutationCount += 1;
        if (mutationCount === 2) {
          throw new Error('synthetic restoration failure');
        }
      },
      verify: async () => {
        if (mutationCount === 1) {
          throw new Error('synthetic action failure');
        }
      },
      registerRestoration: () => {},
    }),
  ).rejects.toMatchObject({
    message: expect.stringMatching(/action and restoration/),
    errors: [
      expect.objectContaining({ message: 'synthetic action failure' }),
      expect.objectContaining({ message: 'synthetic restoration failure' }),
    ],
  });
});

test('batch cleanup continues after one Team ownership GET error', async ({
  page,
}) => {
  const harness = await teamCleanupHarness(page, {
    detailStatus: { [IDS.team]: 500 },
  });
  let caught: unknown;

  try {
    await cleanupCapturedTeamBatch({
      page,
      requestContext: harness.requestContext,
      assertSandbox: async () => {},
      created: harness.created,
      detailEndpoint: harness.detailEndpoint,
      collectionEndpoint: harness.collectionEndpoint,
      uiPath: harness.uiPath,
    });

    caught = error;

  const messages = nestedErrorMessages(caught);

  expect(harness.exists.has(IDS.other)).toBe(false);
  expect(harness.calls).toContain(`DELETE ${IDS.other}`);
});

test('collector drains duplicate same-name Team 201 responses in request order', async ({
  page,
}) => {
  await openSyntheticTeamPage(page);
  let requestOrdinal = 0;
  let confirmSecondAdmission = () => {};
  const secondAdmission = new Promise<void>((resolve) => {
    confirmSecondAdmission = resolve;
  });
  let releaseSecondResponse = () => {};
  const secondResponse = new Promise<void>((resolve) => {
    releaseSecondResponse = resolve;
  });
  await page.route(SYNTHETIC_TEAM_ENDPOINT, async (route) => {
    const ordinal = requestOrdinal++;
    if (ordinal === 1) {
      confirmSecondAdmission();
      await secondResponse;
    }
    await route.fulfill({
      status: 201,
      contentType: 'application/json',
      body: JSON.stringify(
        teamBody({ id: ordinal === 0 ? IDS.team : IDS.other }),
      ),
    });
  });
  const registered: string[] = [];

  const settling = captureWorkspaceTeamPosts({
    page,
    endpoint: SYNTHETIC_TEAM_ENDPOINT,
    requestedName: TEAM_NAME,
    register: async (created) => {
      registered.push(created.id);
    },
    recoverRegistrationFailure: async () => {},
    recoverRejectedExactId: async () => {},
    quietMs: 30,
    maxAdmissionMs: 500,
    action: async () => {
      await page.evaluate((endpoint) => {
        void fetch(endpoint, { method: 'POST' });
        void fetch(endpoint, { method: 'POST' });
      }, SYNTHETIC_TEAM_ENDPOINT);
    },
  });

  await secondAdmission;
  releaseSecondResponse();
  const capture = await settling;

  expect(capture.attemptCount).toBe(2);
  expect(capture.responseCount).toBe(2);
  expect(capture.createdEntries).toEqual([
    { requestOrdinal: 0, created: expect.objectContaining({ id: IDS.team }) },
    { requestOrdinal: 1, created: expect.objectContaining({ id: IDS.other }) },
  ]);
  expect(registered).toEqual([IDS.team, IDS.other]);
});
