export type CreatedTeam = {
  kind: 'team';
  id: string;
  name: string;
  description: string;
  workspaceId: string;
  leadId: typeof WORKSPACE_OPERATIONS_SELF_LEAD_ID;
};

type JsonRecord = Record<string, unknown>;

const isRecord = (value: unknown): value is JsonRecord =>
  typeof value === 'object' && value !== null && !Array.isArray(value);

const assertOwnedName = (name: string) => {
  if (!name.startsWith(`${QA_PREFIX} `)) {
    throw new Error(`Team name must begin with ${QA_PREFIX}.`);
  }
};

export const extractWorkspaceOperationResponseIdentity = (body: unknown) => {
  if (!isRecord(body)) return {};
  const observedId =
    typeof body.id === 'string' && body.id.length > 0 ? body.id : undefined;
  return {
    observedId,
    exactId: isCanonicalEntityId(observedId) ? observedId : undefined,
  };
};

export const parseCreatedTeam = (
  body: unknown,
  requestedName: string,
): CreatedTeam => {
  assertOwnedName(requestedName);
  if (!isRecord(body)) {
    throw new Error('Team create response was not an object.');
  }
  const {
    id,
    name,
    description,
    workspace,
    lead_id: leadId,
  } = body;
  if (
    !isCanonicalEntityId(id) ||
    typeof name !== 'string' ||
    typeof description !== 'string' ||
    typeof workspace !== 'string'
  ) {
    throw new Error('Team create response omitted its exact ID contract.');
  }
  if (name !== requestedName) {
    throw new Error('Team create response identity did not match the request.');
  }
  if (workspace !== WORKSPACE_OPERATIONS_SANDBOX_ID) {
    throw new Error('Team create response crossed its exact workspace scope.');
  }
  if (leadId !== WORKSPACE_OPERATIONS_SELF_LEAD_ID) {
    throw new Error(
      'Team create response included a foreign membership or lead side effect.',
    );
  }
  return {
    kind: 'team',
    id,
    name,
    description,
    workspaceId: workspace,
    leadId,
  };
};

export const assertExactTeamCreatePayload = (
  body: unknown,
  expectedName: string,
  expectedDescription: string,
) => {
  assertOwnedName(expectedName);
  if (!isRecord(body)) {
    throw new Error('Team create payload must be an exact JSON object.');
  }
  const keys = Object.keys(body);
  if (
    keys.length !== 4 ||
    !keys.includes('name') ||
    !keys.includes('description') ||
    !keys.includes('logo_props') ||
    !keys.includes('lead_id')
  ) {
    throw new Error('Team create payload did not match its exact schema.');
  }
  if (body.name !== expectedName) {
    throw new Error('Team create payload was not exact owned data.');
  }
  if (body.description !== expectedDescription) {
    throw new Error('Team create payload description was not exact.');
  }
  if (!isRecord(body.logo_props) || Object.keys(body.logo_props).length !== 0) {
    throw new Error('Team create payload logo must be an empty object.');
  }
  if (body.lead_id !== WORKSPACE_OPERATIONS_SELF_LEAD_ID) {
    throw new Error('Team create payload lead must be the sanctioned self.');
  }
  return body as {
    name: string;
    description: string;
    logo_props: Record<string, never>;
    lead_id: typeof WORKSPACE_OPERATIONS_SELF_LEAD_ID;
  };
};

export const assertExactTeamCleanupTarget = (
  body: unknown,
  created: CreatedTeam,
): JsonRecord => {
  assertOwnedName(created.name);
  if (
    !isCanonicalEntityId(created.id) ||
    created.workspaceId !== WORKSPACE_OPERATIONS_SANDBOX_ID ||
    created.leadId !== WORKSPACE_OPERATIONS_SELF_LEAD_ID ||
    !isRecord(body) ||
    body.id !== created.id ||
    body.name !== created.name ||
    body.workspace !== created.workspaceId ||
    body.lead_id !== WORKSPACE_OPERATIONS_SELF_LEAD_ID
  ) {
    throw new Error('Team cleanup target failed its exact ownership match.');
  }
  return body;
};

type RestoredSandboxSettingOptions<T> = {
  next: T;
  assertSandbox: () => Promise<void>;
  read: () => Promise<T>;
  mutate: (value: T) => Promise<void>;
  verify: (expected: T) => Promise<void>;
  registerRestoration: (restore: () => Promise<void>) => void;
};

export const runRestoredSandboxSetting = async <T>({
  next,
  assertSandbox,
  read,
  mutate,
  verify,
  registerRestoration,
}: RestoredSandboxSettingOptions<T>): Promise<void> => {
  const original = await read();
  let restored = false;
  const restore = async () => {
    if (restored) return;
    await assertSandbox();
    await mutate(original);
    await verify(original);
    restored = true;
  };
  registerRestoration(restore);

  let actionError: unknown;
  try {
    await assertSandbox();
    await mutate(next);
    await verify(next);
  } catch (error) {
    actionError = error;
  }

  let restorationError: unknown;
  try {
    await restore();
  } catch (error) {
    restorationError = error;
  }

  if (actionError !== undefined && restorationError !== undefined) {
    throw new AggregateError(
      [actionError, restorationError],
      'Sandbox setting action and restoration both failed.',
    );
  }
  if (actionError !== undefined) throw actionError;
  if (restorationError !== undefined) throw restorationError;
};

type TeamCleanupOptions = {
  page: Page;
  requestContext: APIRequestContext;
  assertSandbox: () => Promise<void>;
  created: CreatedTeam[];
  detailEndpoint: (id: string) => string;
  collectionEndpoint: string;
  uiPath: string;
  onResolved?: (created: CreatedTeam) => void;
};

const responseRows = (body: unknown): JsonRecord[] => {
  if (Array.isArray(body)) return body.filter(isRecord);
  if (isRecord(body) && Array.isArray(body.results)) {
    return body.results.filter(isRecord);
  }
  throw new Error('Team collection response was not a supported list.');
};

const cleanupResidualError = (
  id: string,
  error: unknown,
) =>
  new AggregateError(
    [error],
    `Team cleanup left exact residual ID: ${id}`,
  );

export const cleanupCapturedTeamBatch = async ({
  page,
  requestContext,
  assertSandbox,
  created,
  detailEndpoint,
  collectionEndpoint,
  uiPath,
  onResolved,
}: TeamCleanupOptions): Promise<void> => {
  const unique = new Map<string, CreatedTeam>();
  for (const entity of created) {
    assertOwnedName(entity.name);
    if (
      !isCanonicalEntityId(entity.id) ||
      entity.workspaceId !== WORKSPACE_OPERATIONS_SANDBOX_ID ||
      entity.leadId !== WORKSPACE_OPERATIONS_SELF_LEAD_ID
    ) {
      throw new Error('Team cleanup batch contained a non-exact identity.');
    }
    if (unique.has(entity.id)) {
      throw new Error('Team cleanup batch contained a duplicate exact ID.');
    }
    unique.set(entity.id, entity);
  }
  if (unique.size === 0) return;

  const errors: unknown[] = [];
  const residualIds = new Set<string>();
  const deleteTargets: Array<{ entity: CreatedTeam; endpoint: string }> = [];
  const markResidual = (entity: CreatedTeam, error: unknown) => {
    if (residualIds.has(entity.id)) return;
    residualIds.add(entity.id);
    errors.push(cleanupResidualError(entity.id, error));
  };

  for (const entity of unique.values()) {
    try {
      const endpoint = detailEndpoint(entity.id);
      const detail = await requestContext.get(endpoint);
      if (detail.status() === 404) {
        onResolved?.(entity);
        continue;
      }
      if (detail.status() !== 200) {
        throw new Error(`Team ownership GET returned ${detail.status()}.`);
      }
      assertExactTeamCleanupTarget(await detail.json(), entity);
      deleteTargets.push({ entity, endpoint });
    } catch (error) {
      markResidual(entity, error);
    }
  }

  for (const { entity, endpoint } of [...deleteTargets].reverse()) {
    try {
      await assertSandbox();
      const deleted = await requestContext.delete(endpoint);
      if (deleted.status() !== 204) {
        throw new Error(`Team exact-ID DELETE returned ${deleted.status()}.`);
      }
      onResolved?.(entity);
    } catch (error) {
      markResidual(entity, error);
    }
  }

  try {
    const collection = await requestContext.get(collectionEndpoint);
    if (collection.status() !== 200) {
      throw new Error(
        `Team collection GET returned ${collection.status()}.`,
      );
    }
    const rows = responseRows(await collection.json());
    const names = new Set([...unique.values()].map(({ name }) => name));
    for (const row of rows) {
      if (typeof row.id === 'string' && unique.has(row.id)) {
        const entity = unique.get(row.id)!;
        markResidual(
          entity,
          new Error('Team remained in its collection after exact-ID cleanup.'),
        );
      } else if (typeof row.name === 'string' && names.has(row.name)) {
        if (isCanonicalEntityId(row.id)) {
          if (!residualIds.has(row.id)) {
            residualIds.add(row.id);
            errors.push(
              new Error(
                `Team collection retained an uncaptured same-name exact residual ID: ${row.id}`,
              ),
            );
          }
        } else {
          errors.push(
            new Error(
              'Team collection retained a shared run-owned name without a canonical ID after exact-ID cleanup.',
            ),
          );
        }
      }
    }
  } catch (error) {
    errors.push(error);
  }

  try {
    await page.goto(uiPath, { waitUntil: 'domcontentloaded' });
    for (const { id } of unique.values()) {
      await expect(page.getByText(id, { exact: true })).toHaveCount(0);
    }
    for (const name of new Set(
      [...unique.values()].map(({ name }) => name),
    )) {
      await expect(page.getByText(name, { exact: true })).toHaveCount(0);
    }
  } catch (error) {
    errors.push(error);
  }

  if (errors.length > 0) {
    const residualLabel =
      residualIds.size > 0
        ? `; residual IDs: ${[...residualIds].join(', ')}`
        : '';
    throw new AggregateError(
      errors,
      `Team exact-ID batch cleanup failed${residualLabel}`,
    );
  }
};

type CaptureWorkspaceTeamPostsOptions = {
  page: Page;
  endpoint: string;
  requestedName: string;
  action: () => Promise<unknown>;
  register: (created: CreatedTeam) => unknown | Promise<unknown>;
  recoverRegistrationFailure: (
    created: CreatedTeam,
  ) => unknown | Promise<unknown>;
  recoverRejectedExactId: (
    id: string,
    requestedName: string,
  ) => unknown | Promise<unknown>;
  quietMs?: number;
  maxAdmissionMs?: number;
};

export const captureWorkspaceTeamPosts = async ({
  page,
  endpoint,
  requestedName,
  action,
  register,
  recoverRegistrationFailure,
  recoverRejectedExactId,
  quietMs,
  maxAdmissionMs,
}: CaptureWorkspaceTeamPostsOptions): Promise<
  ExactFeaturePostCapture<CreatedTeam>
> => {
  assertOwnedName(requestedName);
  const recoveryErrors: unknown[] = [];
  const recoveryResidualIds = new Set<string>();
  const capture = await captureExactFeaturePosts<CreatedTeam>({
    page,
    endpoint,
    action,
    parse: (body) => parseCreatedTeam(body, requestedName),
    extractResponseIdentity: extractWorkspaceOperationResponseIdentity,
    register: async (created) => {
      try {
        await register(created);
      } catch (registrationError) {
        try {
          await recoverRegistrationFailure(created);
        } catch (recoveryError) {
          const failure = new AggregateError(
            [registrationError, recoveryError],
            `Team registration and exact-ID recovery failed: ${created.id}`,
          );
          recoveryResidualIds.add(created.id);
          recoveryErrors.push(failure);
          throw failure;
        }
        throw registrationError;
      }
    },
    ...(quietMs === undefined ? {} : { quietMs }),
    ...(maxAdmissionMs === undefined ? {} : { maxAdmissionMs }),
  });

  for (const contractError of capture.contractErrors) {
    if (contractError.exactId === undefined) continue;
    try {
      await recoverRejectedExactId(contractError.exactId, requestedName);
    } catch (error) {
      recoveryResidualIds.add(contractError.exactId);
      recoveryErrors.push(
        new AggregateError(
          [error],
          `Rejected Team response left exact residual ID: ${contractError.exactId}`,
        ),
      );
    }
  }
  if (recoveryErrors.length > 0) {
    throw new AggregateError(
      recoveryErrors,
      `Team response recovery was incomplete; residual IDs: ${[
        ...recoveryResidualIds,
      ].join(', ')}`,
    );
  }
  return capture;
};
