test('registers and cleans an admitted late 201 before surfacing an action error', async ({
  page,
}) => {
  const endpoint = 'https://qa.invalid/modules/';
  const name = 'QA E2E late response module';
  let releaseResponse = () => {};
  const responseGate = new Promise<void>((resolve) => {
    releaseResponse = resolve;
  });
  await page.setContent('<main>synthetic feature collector</main>');
  await page.route(endpoint, async (route) => {
    await responseGate;
    await route.fulfill({
      status: 201,
      contentType: 'application/json',
      body: JSON.stringify({
        id: IDS.module,
        name,
        project_id: SANDBOX.projectId,
        start_date: '2026-07-27',
        target_date: '2026-07-28',
        status: 'backlog',
      }),
    });
  });

  const ledger = new CleanupLedger();
  const registeredIds: string[] = [];
  const cleanedIds: string[] = [];
  let settled = false;
  const settling = captureExactFeaturePosts({
    page,
    endpoint,
    quietMs: 30,
    maxAdmissionMs: 200,
    parse: (body) => parseCreatedModule(body, name),
    register: (created) => {
      registeredIds.push(created.id);
      ledger.register({
        kind: 'module',
        id: created.id,
        name,
        workspaceSlug: SANDBOX.workspaceSlug,
        cleanup: async () => {
          cleanedIds.push(created.id);
        },
      });
    },
    action: async () => {
      let confirmAdmission = () => {};
      const admitted = new Promise<void>((resolve) => {
        confirmAdmission = resolve;
      });
      const observeRequest = (request: { method(): string; url(): string }) => {
        if (request.method() === 'POST' && request.url() === endpoint) {
          page.off('request', observeRequest);
          confirmAdmission();
        }
      };
      page.on('request', observeRequest);
      await page.evaluate((url) => {
        void fetch(url, { method: 'POST' }).catch(() => {});
      }, endpoint);
      await admitted;
      throw new Error('synthetic click timeout after request admission');
    },
  }).then((capture) => {
    settled = true;
    return capture;
  });

  await page.waitForTimeout(2_200);
  expect(settled).toBe(false);
  expect(registeredIds).toEqual([]);
  releaseResponse();
  const capture = await settling;

  expect(capture.actionError).toHaveProperty(
    'message',
    'synthetic click timeout after request admission',
  );
  expect(capture.attemptCount).toBe(1);
  expect(capture.responseCount).toBe(1);
  expect(capture.created.map(({ id }) => id)).toEqual([IDS.module]);
  expect(registeredIds).toEqual([IDS.module]);
  await expect(ledger.cleanupAll()).resolves.toEqual([]);
  expect(cleanedIds).toEqual([IDS.module]);
});

  page,
}) => {

      contentType: 'text/html',
      body: projectFormHtml(
        `window.syntheticCreateClicks += 1; void fetch('${PROJECTS_ENDPOINT}', { method: 'POST' })`,
      ).replace(
        '<button onclick=',
        '<script>window.syntheticCreateClicks = 0</script><button onclick=',
      ),
    }),
  );
  let mutationRequests = 0;
  await page.route(PROJECTS_ENDPOINT, (route) =>
    route.fulfill({ status: 201, contentType: 'application/json', body: '{}' }),
  );
  page.on('request', (request) => {
    if (
      ['POST', 'PATCH', 'DELETE'].includes(request.method()) &&
      request.url().startsWith('https://flow.nuanu.com/be/api/')
    ) {
      mutationRequests += 1;
    }
  });
  let guardCalls = 0;
  const features = new ProjectFeaturesPage(page, async () => {
    guardCalls += 1;
    throw new Error('synthetic sandbox guard rejected');
  });

  await expect(
    features.createProject(
      {
        name: 'QA E2E guarded project',
        identifier: 'QG123',
        description: 'Synthetic only.',
      },
      () => {},
    ),
  ).rejects.toThrow('synthetic sandbox guard rejected');
  expect(guardCalls).toBe(1);
  await expect
    .poll(() =>
      page.evaluate(
        () =>
          (window as typeof window & { syntheticCreateClicks: number })
            .syntheticCreateClicks,
      ),
    )
    .toBe(0);
  expect(mutationRequests).toBe(0);

  const harnessPost = page.waitForRequest(
    (request) =>
      request.method() === 'POST' && request.url() === PROJECTS_ENDPOINT,
  );
  await page
    .getByRole('button', { name: 'Create project', exact: true })
    .click();
  await harnessPost;
  expect(mutationRequests).toBe(1);
});

test('cycle create click failure without a date-check drains quickly and removes observers', async ({
  page,
}) => {
  await page.route(CYCLES_PAGE, (route) =>
    route.fulfill({
      contentType: 'text/html',
      body: cycleFormHtml({ blockCreate: true }),
    }),
  );
  await page.route('https://flow.nuanu.com/be/api/**', (route) =>
    route.abort('blockedbyclient'),
  );

  let mutationRequests = 0;
  const observeMutation = (request: { method(): string; url(): string }) => {
    if (
      request.method() === 'POST' &&
      [CYCLES_ENDPOINT, CYCLES_DATE_CHECK_ENDPOINT].includes(request.url())
    ) {
      mutationRequests += 1;
    }
  };
  page.on('request', observeMutation);
  const listenerPage = page as typeof page & {
    listenerCount(event: string): number;
  };
  const baselineListeners = {
    request: listenerPage.listenerCount('request'),
    response: listenerPage.listenerCount('response'),
    requestfailed: listenerPage.listenerCount('requestfailed'),
  };
  page.setDefaultTimeout(300);

  try {
    let guardCalls = 0;
    const features = new ProjectFeaturesPage(page, async () => {
      guardCalls += 1;
      expect(new URL(page.url()).pathname).toBe(
        `/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/cycles/`,
      );
    });
    const startedAt = Date.now();
    const completion = features.createCycle(
      {
        name: 'QA E2E blocked date-check cycle',
        description: 'Synthetic only.',
        startDate: new Date(2026, 6, 27, 12),
        endDate: new Date(2026, 6, 28, 12),
      },
      () => {
        throw new Error('Blocked cycle must not register.');
      },
    );
    const outcome = await Promise.race([
      completion.then(
        () => ({ kind: 'resolved' as const }),
        (error: unknown) => ({ kind: 'rejected' as const, error }),
      ),
      new Promise<{ kind: 'timed-out' }>((resolve) =>
        setTimeout(() => resolve({ kind: 'timed-out' }), 2_500),
      ),
    ]);

    expect(outcome.kind).toBe('rejected');
    expect(Date.now() - startedAt).toBeLessThan(2_500);
    expect(guardCalls).toBe(1);
    expect(mutationRequests).toBe(0);
    expect(listenerPage.listenerCount('request')).toBe(baselineListeners.request);
    expect(listenerPage.listenerCount('response')).toBe(baselineListeners.response);
    expect(listenerPage.listenerCount('requestfailed')).toBe(
      baselineListeners.requestfailed,
    );
  } finally {
    page.setDefaultTimeout(DEFAULT_ACTION_TIMEOUT_MS);
    page.off('request', observeMutation);
  }
});

      'POST',

              .catch(() => {});

          {
            checkUrl: preconditionEndpoint,
            createUrl: endpoint,
            releaseUrl: createReleaseEndpoint,
          },
        ),
      waitBeforeAdmissionClose: async (waitForExactAdmission) => {

        registeredIds.push(created.id);
        ledger.register({
          kind: 'module',
          id: created.id,
          name,
          workspaceSlug: SANDBOX.workspaceSlug,
          cleanup: async () => cleanedIds.push(created.id),
        });
      },

      settled = true;
      return capture;
    });

    await page.waitForTimeout(2_200);
    expect(settled).toBe(false);
    expect(registeredIds).toEqual([]);
    releasePrecondition();
    await new Promise((resolve) => setTimeout(resolve, 2_250));
    expect(settled).toBe(false);
    expect(registeredIds).toEqual([]);
    releaseCreate();
    const capture = await settling;
    expect((await precondition).status()).toBe(200);
    expect(capture.attemptCount).toBe(1);
    expect(capture.created.map(({ id }) => id)).toEqual([IDS.module]);
    expect(registeredIds).toEqual([IDS.module]);
    await expect(ledger.cleanupAll()).resolves.toEqual([]);
    expect(cleanedIds).toEqual([IDS.module]);
  } finally {
    releasePrecondition();
    releaseCreate();
    page.setDefaultTimeout(DEFAULT_ACTION_TIMEOUT_MS);
  }
});

const SYNTHETIC_ITEM_ID = IDS.item;
const PRE_EXISTING_ITEM_ID = '99999999-9999-4999-8999-999999999999';
const SYNTHETIC_ITEM_SEQUENCE = 77;
const SYNTHETIC_MODULE_NAME = 'QA E2E guarded association module';
const SYNTHETIC_ITEM_NAME = 'QA E2E guarded association item';
const MODULES_PATH =
  `/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/modules/`;
const MODULES_ENDPOINT_PATH =
  `/be/api/workspaces/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/modules/`;
const ISSUES_PATH =
  `/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/issues/`;
const ISSUES_ENDPOINT_PATH =
  `/be/api/workspaces/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/issues/`;
const ITEM_PATH =
  `/${SANDBOX.workspaceSlug}/browse/NFQAS-${SYNTHETIC_ITEM_SEQUENCE}/`;
const MODULE_ASSOCIATION_ENDPOINT_PATH =
  `/be/api/workspaces/${SANDBOX.workspaceSlug}/projects/${SANDBOX.projectId}/modules/${IDS.module}/issues/`;

type SyntheticAssociationOptions = {
  requestPath?: string;
  delayedRequestAfterPickerMs?: number;
  delayedSecondRequestMs?: number;
};

const startSyntheticAssociationServer = async (
  associationPayload: unknown,
  options: SyntheticAssociationOptions = {},
) => {
  let itemCreated = false;
  let associationRequests = 0;
  const receivedAssociationPayloads: unknown[] = [];
  const associationRequestPath =
    options.requestPath ?? MODULE_ASSOCIATION_ENDPOINT_PATH;

  const server = createServer(async (request, response) => {
    const method = request.method ?? 'GET';
    const path = new URL(request.url ?? '/', 'http://127.0.0.1').pathname;
    const sendHtml = (body: string) => {
      response.writeHead(200, { 'content-type': 'text/html; charset=utf-8' });
      response.end(body);
    };
    const sendJson = (status: number, body: unknown) => {
      response.writeHead(status, { 'content-type': 'application/json' });
      response.end(JSON.stringify(body));
    };
    const readJson = async () => {
      let raw = '';
      for await (const chunk of request) raw += chunk.toString();
      return JSON.parse(raw) as unknown;
    };

    if (method === 'GET' && path === MODULES_PATH) {
      sendHtml(`
        <button id="add-module">Add Module</button>
        <dialog id="module-dialog">
          <h2>Create module</h2>
          <input aria-label="Title" />
          <textarea aria-label="Description"></textarea>
          <button id="date-range">Start date End date</button>
          <div id="date-list" role="listbox" aria-label="Start date End date" hidden>
            <button type="button" role="option">July</button>
            <button type="button" data-date aria-label="Monday, July 27th, 2026">27</button>
            <button type="button" data-date aria-label="Tuesday, July 28th, 2026">28</button>
          </div>
          <button id="create-module">Create Module</button>
          <button>Cancel</button>
        </dialog>
        <script>
          const moduleDialog = document.getElementById('module-dialog');
          const dateList = document.getElementById('date-list');
          document.getElementById('add-module').onclick = () => moduleDialog.showModal();
          document.getElementById('date-range').onclick = () => { dateList.hidden = false; };
          let selectedDates = 0;
          for (const button of document.querySelectorAll('[data-date]')) {
            button.onclick = () => {
              selectedDates += 1;
              if (selectedDates === 2) dateList.hidden = true;
            };
          }
          document.getElementById('create-module').onclick = () => {
            void fetch('${MODULES_ENDPOINT_PATH}', {
              method: 'POST',
              headers: { 'content-type': 'application/json' },
              body: JSON.stringify({ name: '${SYNTHETIC_MODULE_NAME}' }),
            }).then(() => moduleDialog.close());
          };
        </script>
      `);
      return;
    }

    if (method === 'POST' && path === MODULES_ENDPOINT_PATH) {
      sendJson(201, {
        id: IDS.module,
        name: SYNTHETIC_MODULE_NAME,
        project_id: SANDBOX.projectId,
        start_date: '2026-07-27',
        target_date: '2026-07-28',
        status: 'backlog',
      });
      return;
    }

    if (method === 'GET' && path === ISSUES_PATH) {
      sendHtml(`
        <button id="add-work-item">Add work item</button>
        <dialog id="work-item-dialog">
          <h2>Create new work item</h2>
          <input aria-label="Title" />
          <div contenteditable="true"></div>
          <button id="save-work-item">Save</button>
        </dialog>
        ${
          itemCreated
            ? `<a href="${ITEM_PATH}">${SYNTHETIC_ITEM_NAME}</a>`
            : ''
        }
        <script>
          const itemDialog = document.getElementById('work-item-dialog');
          document.getElementById('add-work-item').onclick = () => itemDialog.showModal();
          document.getElementById('save-work-item').onclick = () => {
            void fetch('${ISSUES_ENDPOINT_PATH}', {
              method: 'POST',
              headers: { 'content-type': 'application/json' },
              body: JSON.stringify({ name: '${SYNTHETIC_ITEM_NAME}' }),
            }).then(() => itemDialog.close());
          };
        </script>
      `);
      return;
    }

    if (method === 'POST' && path === ISSUES_ENDPOINT_PATH) {
      itemCreated = true;
      sendJson(201, {
        id: SYNTHETIC_ITEM_ID,
        name: SYNTHETIC_ITEM_NAME,
        project_id: SANDBOX.projectId,
        sequence_id: SYNTHETIC_ITEM_SEQUENCE,
      });
      return;
    }

    if (method === 'GET' && path === ITEM_PATH) {
      const delayedRequest =
        options.delayedRequestAfterPickerMs === undefined
          ? ''
          : `
            blocker.hidden = false;
            setTimeout(sendAssociation, ${options.delayedRequestAfterPickerMs});
          `;
      const optionAction =
        options.delayedRequestAfterPickerMs === undefined
          ? `
            sendAssociation();
            ${
              options.delayedSecondRequestMs === undefined
                ? ''
                : `setTimeout(sendAssociation, ${options.delayedSecondRequestMs});`
            }
          `
          : '';
      sendHtml(`
        <main>
          <input aria-label="Work item title" value="${SYNTHETIC_ITEM_NAME}" />
          <div>
            <span>Modules</span>
            <button id="module-property">No module</button>
          </div>
          <div id="module-options" role="listbox" hidden>
            <button id="module-option" role="option">${SYNTHETIC_MODULE_NAME}</button>
          </div>
          <div id="blocker" hidden style="position:fixed;inset:0;z-index:9999"></div>
        </main>
        <script>
          const moduleProperty = document.getElementById('module-property');
          const moduleOptions = document.getElementById('module-options');
          const blocker = document.getElementById('blocker');
          const sendAssociation = () => {
            void fetch('${associationRequestPath}', {
              method: 'POST',
              headers: { 'content-type': 'application/json' },
              body: JSON.stringify(${JSON.stringify(associationPayload)}),
            }).then((result) => {
              if (!result.ok) return;
              moduleProperty.textContent = '${SYNTHETIC_MODULE_NAME}';
              moduleOptions.hidden = true;
            }).catch(() => {});
          };
          moduleProperty.onclick = () => {
            moduleOptions.hidden = false;
            ${delayedRequest}
          };
          document.getElementById('module-option').onclick = () => {
            ${optionAction}
          };
        </script>
      `);
      return;
    }

    if (
      method === 'POST' &&
      path !== MODULES_ENDPOINT_PATH &&
      path !== ISSUES_ENDPOINT_PATH
    ) {
      associationRequests += 1;
      receivedAssociationPayloads.push(await readJson());
      sendJson(201, {});
      return;
    }

    response.writeHead(404);
    response.end();
  });

  await new Promise<void>((resolve, reject) => {
    server.once('error', reject);
    server.listen(0, '127.0.0.1', resolve);
  });
  const address = server.address() as AddressInfo;
  return {
    origin: `http://127.0.0.1:${address.port}`,
    associationRequests: () => associationRequests,
    receivedAssociationPayloads,
    close: () =>
      new Promise<void>((resolve, reject) =>
        server.close((error) => (error ? reject(error) : resolve())),
      ),
  };
};

const openSyntheticOwnedModuleAssociation = async (
  browser: Browser,
  associationPayload: unknown,
  options: SyntheticAssociationOptions = {},
) => {
  const server = await startSyntheticAssociationServer(
    associationPayload,
    options,
  );
  const context = await browser.newContext({ baseURL: server.origin });
  try {
    const page = await context.newPage();
    const assertSandbox = async () => {
      expect(new URL(page.url()).pathname).toMatch(
        new RegExp(`^/${SANDBOX.workspaceSlug}/`),
      );
    };
    const features = new ProjectFeaturesPage(page, assertSandbox);
    const workItems = new WorkItemsPage(page);
    const module = await features.createModule(
      {
        name: SYNTHETIC_MODULE_NAME,
        description: 'Synthetic local module.',
        startDate: new Date(2026, 6, 27, 12),
        endDate: new Date(2026, 6, 28, 12),
      },
      () => {},
    );
    await workItems.goto();
    const item = await workItems.create(
      { title: SYNTHETIC_ITEM_NAME },
      () => {},
    );
    await workItems.goto();
    await workItems.openByName(SYNTHETIC_ITEM_NAME, item);
    return {
      context,
      features,
      item,
      module,
      page,
      server,
      workItems,
      close: async () => {
        await context.close();
        await server.close();
      },
    };
  } catch (error) {
    await context.close();
    await server.close();
    throw error;
  }
};
