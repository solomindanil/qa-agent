import { test, expect } from '../fixtures/authenticated.fixture';
import {
  AUTOMATION_ACTION_NAMES,
  AUTOMATION_TRIGGER_NAMES,
  PULSE_SUMMARY_SECTION_NAMES,
  PULSE_SUMMARY_TYPE_NAMES,
  WorkspaceOperationsPage,
} from '../pages/workspace-operations.page';

test.describe.configure({ mode: 'serial' });

test.describe('Nuanu Flow Pulse and automations', () => {
  test('distinguishes Pulse summaries and feed without publishing', async ({
    page,
    assertSandbox,
  }) => {
    const operations = new WorkspaceOperationsPage(page);
    const pulsePosts: string[] = [];
    const observePulsePost = (request: {
      method(): string;
      url(): string;
    }) => {
      if (
        request.method() === 'POST' &&
        new URL(request.url()).pathname.endsWith('/pulse/')
      ) {
        pulsePosts.push(request.url());
      }
    };
    page.on('request', observePulsePost);

    try {
      await operations.gotoPulse();
      await assertSandbox();
      await expect(operations.pulseSummariesButton).toBeVisible();
      await expect(operations.pulseFeedButton).toBeVisible();
      await expect(
        page.getByRole('button', {
          name: 'Generate Summaries',
          exact: true,
        }),
      ).toBeVisible();
      for (const section of PULSE_SUMMARY_SECTION_NAMES) {
        await expect(
          page.getByRole('heading', { name: section, exact: true }),
        ).toBeVisible();
      }
      await expect(
        page.getByText('No summary available yet', { exact: true }),
      ).toHaveCount(3);
      await expect(
        page.getByRole('button', {
          name: 'Generate this section',
          exact: true,
        }),
      ).toHaveCount(3);

      const dialog = await operations.openPulseFeedItem();
      await expect(
        page.getByRole('heading', {
          name: 'No summaries yet',
          exact: true,
        }),
      ).toBeVisible();
      await expect(
        page.getByText(
          'Create pulse summaries to track team progress and insights',
          { exact: true },
        ),
      ).toBeVisible();
      await expect(dialog.getByRole('option')).toHaveText(
        PULSE_SUMMARY_TYPE_NAMES,
      );
      expect(
        await dialog
          .getByRole('option', {
            name: 'Day Review',
            exact: true,
          })
          .evaluate((option) => (option as HTMLOptionElement).selected),
      ).toBe(true);
      await expect(
        dialog.getByRole('textbox', {
          name: 'Enter a title for this pulse',
          exact: true,
        }),
      ).toHaveValue('');
      await expect(
        dialog.getByRole('textbox', {
          name: 'Write your summary here...',
          exact: true,
        }),
      ).toHaveValue('');
      await expect(
        dialog.getByText('Period Start', { exact: true }),
      ).toBeVisible();
      await expect(
        dialog.getByText('Period End', { exact: true }),
      ).toBeVisible();
      const pulseTextboxes = dialog.getByRole('textbox');
      await expect(pulseTextboxes).toHaveCount(4);
      await expect(pulseTextboxes.nth(2)).toHaveValue('');
      await expect(pulseTextboxes.nth(3)).toHaveValue('');
      await expect(
        dialog.getByRole('button', {
          name: 'Create Pulse',
          exact: true,
        }),
      ).toBeEnabled();
      await operations.cancelCreate(dialog);
      expect(pulsePosts).toEqual([]);
    } finally {
      page.off('request', observePulsePost);
    }
  });

  test('maps automation schedules, reaction triggers, and effectful actions without saving', async ({
    page,
    assertSandbox,
  }) => {
    const operations = new WorkspaceOperationsPage(page);
    const automationPosts: string[] = [];
    const observeAutomationPost = (request: {
      method(): string;
      url(): string;
    }) => {
      if (
        request.method() === 'POST' &&
        new URL(request.url()).pathname.endsWith('/automations/')
      ) {
        automationPosts.push(request.url());
      }
    };
    page.on('request', observeAutomationPost);

    try {
      await operations.gotoAutomations();
      await assertSandbox();
      await expect(operations.addAutomationButtons).toHaveCount(2);
      await expect(operations.automationEmptyHeading).toBeVisible();

      const surface = await operations.openAutomationCreate();
      await expect(
        surface.getByRole('checkbox', { name: 'Enabled', exact: true }),
      ).toBeChecked();
      await expect(
        surface.getByRole('textbox', {
          name: 'e.g., Daily Standup Reminder',
          exact: true,
        }),
      ).toHaveValue('');
      await expect(
        surface.getByRole('button', { name: 'Recurrent', exact: true }),
      ).toBeVisible();
      await expect(
        surface.getByRole('button', { name: 'Every day', exact: true }),
      ).toBeVisible();
      await expect(
        surface.getByText('Schedule preview: Runs every day at 9:00 AM', {
          exact: true,
        }),
      ).toBeVisible();

      await surface
        .getByRole('button', { name: 'Reaction', exact: true })
        .click();
      for (const trigger of AUTOMATION_TRIGGER_NAMES) {
        await expect(
          surface.getByRole('button', { name: trigger, exact: true }),
        ).toBeVisible();
      }
      await surface
        .getByRole('button', { name: 'Actions', exact: true })
        .click();
      await surface
        .getByRole('button', { name: 'Add Action', exact: true })
        .click();
      const actions = page.getByRole('listbox', {
        name: 'Add Action',
        exact: true,
      });
      for (const action of AUTOMATION_ACTION_NAMES) {
        await expect(
          actions.getByRole('option').filter({ hasText: action }),
        ).toHaveCount(1);
      }

      await operations.cancelCreate(surface);
      await expect(operations.automationEmptyHeading).toBeVisible();
      expect(automationPosts).toEqual([]);
    } finally {
      page.off('request', observeAutomationPost);
    }
  });

  test('creates and removes one response-owned Pulse item', async () => {
    test.fixme(
      true,
      'Pulse publication has no proven response-owned exact-ID detail/delete and requestfailed recovery contract, so this campaign does not publish a feed item.',
    );
  });

  test('persists, enables, disables, and deletes a safe automation', async () => {
    test.fixme(
      true,
      'Every observed action can prompt AI, create an issue, or call a webhook; no non-effectful action or exact-ID requestfailed recovery contract exists, so save/enable/run remains blocked.',
    );
  });
});
