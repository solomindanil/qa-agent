const MUTATING_TEST_TIMEOUT_MS = 90_000;

const absoluteEndpoint = (pageUrl: string, path: string) =>
  new URL(path, pageUrl).href;

const expectHealthyCapture = <T>(
  capture: ProcessMutationCapture<T>,
  expectedSuccesses = 1,
) => {
  expect(capture.actionError).toBeUndefined();
  expect(capture.attemptCount).toBe(expectedSuccesses);
  expect(capture.responseCount).toBe(expectedSuccesses);
  expect(capture.successfulEntries).toHaveLength(expectedSuccesses);
  expect(capture.nonSuccessResponses).toEqual([]);
  expect(capture.contractErrors).toEqual([]);
  expect(capture.registrationErrors).toEqual([]);
  expect(capture.requestFailedOutcomes).toEqual([]);
  expect(capture.unexpectedMutationAttempts).toEqual([]);
};

const recoverRejectedCreateIds = async ({
  page,
  capture,
  requestedName,
  created,
}: {
  page: Page;
  capture: ProcessMutationCapture<CreatedProcess>;
  requestedName: string;
  created: CreatedProcess[];
}) => {
  await reconcileRejectedProcessExactIds({
    rejections: capture.contractErrors,
    requestedName,
    workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
    inspect: async (id) => {
      const response = await page.request.get(
        PROCESS_PATHS.detailApi(id),
      );
      return {
        status: response.status(),
        ...(response.status() === 200
          ? { body: await response.json() }
          : {}),
      };
    },
    cleanup: async (recovered) => {
      if (!created.some(({ id }) => id === recovered.id)) {
        created.push(recovered);
      }
    },
  });
};

const cleanupOwnedProcesses = async ({
  page,
  assertSandbox,
  created,
  allowedNames,
}: {
  page: Page;
  assertSandbox: () => Promise<void>;
  created: CreatedProcess[];
  allowedNames: ReadonlySet<string>;
}) =>
  cleanupCapturedProcessBatch({
    page,
    requestContext: page.request,
    assertSandbox,
    created,
    allowedNames,
    detailEndpoint: PROCESS_PATHS.detailApi,
    collectionEndpoint: PROCESS_PATHS.collectionApi,
    uiPath: PROCESS_PATHS.list,
  });

  test('handles a bracketed process name without silent failure (NUANU-8)', async ({
    page,
    assertSandbox,
    ownedName,
  }) => {
    test.setTimeout(MUTATING_TEST_TIMEOUT_MS);
    const processes = new ProcessesPage(page, assertSandbox);
    const bracketedName = ownedName('[BPMN] punctuation');
    const allowedNames = new Set([bracketedName]);
    const created: CreatedProcess[] = [];
    let behaviorFailure: unknown | undefined;

    try {
      const capture = await processes.createOwned({
        name: bracketedName,
        description: 'Guarded regression coverage for NUANU-8.',
        category: 'Operational',
        register: (responseOwned) => {
          created.push(responseOwned);
        },
        quietMs: 750,
        minAdmissionMs: 25_000,
        maxAdmissionMs: 26_000,
      });
      await recoverRejectedCreateIds({
        page,
        capture,
        requestedName: bracketedName,
        created,
      });

      const operationalErrors: unknown[] = [];
      if (capture.actionError !== undefined) {
        operationalErrors.push(capture.actionError);
      }
      operationalErrors.push(
        ...capture.nonSuccessResponses.map(
          ({ status, url }) =>
            new Error(`NUANU-8 create returned ${status} at ${url}.`),
        ),
        ...capture.contractErrors.map(({ error }) => error),
        ...capture.registrationErrors.map(({ error }) => error),
        ...capture.requestFailedOutcomes.map(
          ({ url }) =>
            new Error(`NUANU-8 create request failed at ${url}.`),
        ),
        ...capture.unexpectedMutationAttempts.map(
          ({ method, url }) =>
            new Error(
              `NUANU-8 blocked unexpected process mutation ${method} ${url}.`,
            ),
        ),
      );

      const dialogVisible = await processes.createDialog.isVisible();
      const validationMessages = dialogVisible
        ? await processes.createNameValidationMessages()
        : [];
      const actionableValidation = validationMessages.length > 0;
      const [createdProcess] = created;
      const exactProcessVisible =
        created.length === 1 &&
        createdProcess !== undefined &&
        (await processes.ownedProcessLink(createdProcess.id).count()) === 1;
      let exactNameAbsenceProven = false;
      if (
        capture.successfulEntries.length === 0 &&
        operationalErrors.length === 0
      ) {
        await assertExactProcessNameAbsent({
          page,
          requestContext: page.request,
          assertSandbox,
          name: bracketedName,
          collectionEndpoint: PROCESS_PATHS.collectionApi,
          uiPath: PROCESS_PATHS.list,
        });
        exactNameAbsenceProven = true;
      }

      behaviorFailure = classifyNuanu8ProcessOutcome({
        submitCompleted: capture.actionError === undefined,
        dialogVisible,
        actionableValidation,
        exactProcessVisible,
        successfulProcessCount: capture.successfulEntries.length,
        exactNameAbsenceProven,
        operationalErrors,
        cleanupResidualIds: [],
      });
    } finally {
      await cleanupOwnedProcesses({
        page,
        assertSandbox,
        created,
        allowedNames,
      });
    }

    const expectedFailure =
      requireNuanu8KnownBugFailure(behaviorFailure);
    test.fail(
      true,
      'NUANU-8: completed submit leaves the bracketed process form open without actionable validation or an exact created process.',
    );
    throw expectedFailure;
  });

  test('persists an inert Start → Human Task → End graph', async ({
    page,
    assertSandbox,
    ownedName,
  }) => {
    test.setTimeout(MUTATING_TEST_TIMEOUT_MS);
    const processes = new ProcessesPage(page, assertSandbox);
    const processName = ownedName('editor BPMN process');
    const humanTaskLabel = ownedName('human task');
    const allowedNames = new Set([processName]);
    const created: CreatedProcess[] = [];
    let humanTaskId: string | undefined;
    let finalGraph: InertHumanTaskGraph | undefined;
    let finalRequestXml: string | undefined;

    const requireOneId = (
      xml: string,
      localName: 'userTask' | 'sequenceFlow',
      relation?: { sourceRef: string; targetRef: string },
    ) => {
      const tags = [
        ...xml.matchAll(
          new RegExp(`<bpmn:${localName}\\b[^>]*>`, 'gu'),
        ),
      ].filter(({ 0: tag }) => {
        if (!relation) return true;
        return (
          tag.includes(`sourceRef="${relation.sourceRef}"`) &&
          tag.includes(`targetRef="${relation.targetRef}"`)
        );
      });
      expect(tags).toHaveLength(1);
      const id = tags[0]?.[0].match(/\bid="([^"]+)"/u)?.[1];
      expect(id).toMatch(/^[A-Za-z][A-Za-z0-9_-]*$/);
      if (!id) throw new Error(`Missing exact ${localName} ID.`);
      return id;
    };

    try {
      const createCapture = await processes.createOwned({
        name: processName,
        description:
          'Inert Start to Human Task to End BPMN lifecycle.',
        category: 'Operational',
        register: (responseOwned) => {
          created.push(responseOwned);
        },
      });
      await recoverRejectedCreateIds({
        page,
        capture: createCapture,
        requestedName: processName,
        created,
      });
      expectHealthyCapture(createCapture);
      const owned = created[0];
      expect(owned).toBeDefined();
      if (!owned) throw new Error('Missing response-owned process.');

      await processes.gotoOwned(owned);
      await assertSandbox();
      expect(new URL(page.url()).pathname).toBe(
        PROCESS_PATHS.detail(owned.id),
      );
      const editor = new ProcessEditorPage(page, assertSandbox);
      await expect(editor.autosaveSwitch).toBeChecked();
      await expect(editor.elementById('StartEvent_1')).toHaveCount(1);
      await expect(editor.startRunButton).toBeDisabled();

      const humanTaskCapture = await captureExactProcessSaves({
        page,
        created: owned,
        allowedNames,
        assertSandbox,
        action: () => editor.appendHumanTaskFrom('StartEvent_1'),
        validatePayload: (request) => {
          const xml = assertExactInertBpmnAutosavePayload(
            request.postDataJSON(),
            { requiredElementIds: ['StartEvent_1'] },
          );
          const taskId = requireOneId(xml, 'userTask');
          requireOneId(xml, 'sequenceFlow', {
            sourceRef: 'StartEvent_1',
            targetRef: taskId,
          });
          humanTaskId = taskId;
        },
        validateResponse: (body, request) => {
          const requestXml = assertExactInertBpmnAutosavePayload(
            request.postDataJSON(),
            { requiredElementIds: ['StartEvent_1'] },
          );
          assertExactInertBpmnAutosaveResponse(body, {
            processId: owned.id,
            workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
            requestXml,
          });
        },
        quietMs: 800,
        maxAdmissionMs: 5_000,
      });
      expectHealthyCapture(humanTaskCapture.capture);
      expect(humanTaskCapture.foreignWriteAttempts).toEqual([]);
      expect(humanTaskId).toBeDefined();
      if (!humanTaskId) {
        throw new Error('Human Task save missed its exact ID.');
      }
      const exactHumanTaskId = humanTaskId;
      await expect(editor.elementById(exactHumanTaskId)).toHaveCount(1);

      const labelCapture = await captureExactProcessSaves({
        page,
        created: owned,
        allowedNames,
        assertSandbox,
        action: async () => {
          await editor.elementById(exactHumanTaskId).click();
          await editor.renameSelectedHumanTask(humanTaskLabel);
        },
        validatePayload: (request) => {
          assertExactInertBpmnAutosavePayload(
            request.postDataJSON(),
            {
              requiredElementIds: [
                'StartEvent_1',
                exactHumanTaskId,
              ],
              requiredLabel: humanTaskLabel,
            },
          );
        },
        validateResponse: (body, request) => {
          const requestXml = assertExactInertBpmnAutosavePayload(
            request.postDataJSON(),
            {
              requiredElementIds: [
                'StartEvent_1',
                exactHumanTaskId,
              ],
              requiredLabel: humanTaskLabel,
            },
          );
          assertExactInertBpmnAutosaveResponse(body, {
            processId: owned.id,
            workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
            requestXml,
          });
        },
        quietMs: 800,
        maxAdmissionMs: 5_000,
      });
      expectHealthyCapture(labelCapture.capture);
      expect(labelCapture.foreignWriteAttempts).toEqual([]);

      const endCapture = await captureExactProcessSaves({
        page,
        created: owned,
        allowedNames,
        assertSandbox,
        action: () =>
          editor.appendEndEventFrom(exactHumanTaskId),
        validatePayload: (request) => {
          const body = request.postDataJSON() as {
            bpmn_xml?: unknown;
          };
          if (typeof body.bpmn_xml !== 'string') {
            throw new Error('Final BPMN save missed its XML.');
          }
          finalGraph = assertExactInertHumanTaskGraph(
            body.bpmn_xml,
            {
              taskLabel: humanTaskLabel,
              expectedStartId: 'StartEvent_1',
            },
          );
          finalRequestXml = body.bpmn_xml;
        },
        validateResponse: (body, request) => {
          const requestBody = request.postDataJSON() as {
            bpmn_xml?: unknown;
          };
          if (typeof requestBody.bpmn_xml !== 'string') {
            throw new Error('Final BPMN response missed request XML.');
          }
          assertExactInertHumanTaskGraph(requestBody.bpmn_xml, {
            taskLabel: humanTaskLabel,
            expectedStartId: 'StartEvent_1',
          });
          assertExactInertBpmnAutosaveResponse(body, {
            processId: owned.id,
            workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
            requestXml: requestBody.bpmn_xml,
          });
        },
        quietMs: 800,
        maxAdmissionMs: 5_000,
      });
      expectHealthyCapture(endCapture.capture);
      expect(endCapture.foreignWriteAttempts).toEqual([]);
      expect(finalGraph).toBeDefined();
      expect(finalRequestXml).toBeDefined();
      if (!finalGraph || !finalRequestXml) {
        throw new Error('Final inert BPMN graph was not captured.');
      }

      await expect(editor.savedIndicator).toBeVisible();
      await expect(editor.zeroErrorsIndicator).toBeVisible();
      await expect(
        editor.elementById(finalGraph.startId),
      ).toHaveCount(1);
      await expect(
        editor.elementById(finalGraph.humanTaskId),
      ).toContainText(humanTaskLabel);
      await expect(editor.elementById(finalGraph.endId)).toHaveCount(1);
      await expect(
        editor.elementById(finalGraph.startToTaskFlowId),
      ).toHaveCount(1);
      await expect(
        editor.elementById(finalGraph.taskToEndFlowId),
      ).toHaveCount(1);

      await page.reload({ waitUntil: 'domcontentloaded' });
      await assertSandbox();
      await expect(editor.savedIndicator).toBeVisible();
      await expect(editor.zeroErrorsIndicator).toBeVisible();
      await expect(
        editor.elementById(finalGraph.humanTaskId),
      ).toContainText(humanTaskLabel);
      for (const id of [
        finalGraph.startId,
        finalGraph.endId,
        finalGraph.startToTaskFlowId,
        finalGraph.taskToEndFlowId,
      ]) {
        await expect(editor.elementById(id)).toHaveCount(1);
      }

      const persisted = await page.request.get(
        PROCESS_PATHS.detailApi(owned.id),
      );
      expect(persisted.status()).toBe(200);
      const persistedBody: unknown = await persisted.json();
      const persistedXml = assertExactInertBpmnAutosaveResponse(
        persistedBody,
        {
          processId: owned.id,
          workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
          requestXml: finalRequestXml,
        },
      );
      expect(
        assertExactInertHumanTaskGraph(persistedXml, {
          taskLabel: humanTaskLabel,
          expectedStartId: 'StartEvent_1',
        }),
      ).toEqual(finalGraph);
    } finally {
      await cleanupOwnedProcesses({
        page,
        assertSandbox,
        created,
        allowedNames,
      });
    }
  });

      expectHealthyCapture(finalFlowCapture.capture);
      expect(finalFlowCapture.foreignWriteAttempts).toEqual([]);
      if (!finalGraph || !finalRequestXml) {
        throw new Error('Final Exclusive Gateway graph was not captured.');
      }
      const exactFinalGraph = finalGraph;
      const exactFinalRequestXml = finalRequestXml;
      expect(exactFinalGraph.gatewayId).toBe(exactGatewayId);
      expect(exactFinalGraph.endId).toBe(exactEndId);
      expect(exactFinalGraph.startToGatewayFlowId).toBe(
        exactStartToGatewayFlowId,
      );
      await expect(editor.savedIndicator).toBeVisible();
      await expect(editor.noIssuesIndicator).toBeVisible();

      const undoCapture = await captureSave(
        () => editor.undoWithButton(),
        (xml) => {
          assertExactInertBpmnAutosavePayload(
            { bpmn_xml: xml },
            {
              requiredElementIds: [
                exactFinalGraph.startId,
                exactFinalGraph.gatewayId,
                exactFinalGraph.endId,
                exactFinalGraph.startToGatewayFlowId,
              ],
            },
          );
          expect(
            xml.match(/<bpmn:sequenceFlow\b/gu) ?? [],
          ).toHaveLength(1);
          expect(xml).not.toContain(
            `id="${exactFinalGraph.gatewayToEndFlowId}"`,
          );
        },
      );
      expectHealthyCapture(undoCapture.capture);
      expect(undoCapture.foreignWriteAttempts).toEqual([]);
      await expect(editor.oneWarningIndicator).toBeVisible();

      const redoCapture = await captureSave(
        () => editor.redoWithButton(),
        (xml) => {
          expect(
            assertExactInertExclusiveGatewayGraph(xml, {
              expectedStartId: 'StartEvent_1',
            }),
          ).toEqual(exactFinalGraph);
        },
      );
      expectHealthyCapture(redoCapture.capture);
      expect(redoCapture.foreignWriteAttempts).toEqual([]);
      await expect(editor.noIssuesIndicator).toBeVisible();

      await page.reload({ waitUntil: 'domcontentloaded' });
      await assertSandbox();
      await expect(editor.savedIndicator).toBeVisible();
      await expect(editor.noIssuesIndicator).toBeVisible();
      for (const id of Object.values(exactFinalGraph)) {
        await expect(editor.elementById(id)).toHaveCount(1);
      }
      const persisted = await page.request.get(
        PROCESS_PATHS.detailApi(owned.id),
      );
      expect(persisted.status()).toBe(200);
      const persistedXml = assertExactInertBpmnAutosaveResponse(
        await persisted.json(),
        {
          processId: owned.id,
          workspaceId: PROCESS_SANDBOX_WORKSPACE_ID,
          requestXml: exactFinalRequestXml,
        },
      );
      expect(
        assertExactInertExclusiveGatewayGraph(persistedXml, {
          expectedStartId: 'StartEvent_1',
        }),
      ).toEqual(exactFinalGraph);
    } finally {
      await cleanupOwnedProcesses({
        page,
        assertSandbox,
        created,
        allowedNames,
      });
    }
  });
