const REPOSITORY_ROOT = '/qa/repository';
const EVIDENCE_ROOT =
  `${REPOSITORY_ROOT}/docs/local/nuanuflow/evidence`;
const VALID_PATH = `${EVIDENCE_ROOT}/${APPROVED_AUDIO_BASENAME}`;

type FakeOverrides = Partial<{
  candidateRealPath: string;
  evidenceRealPath: string;
  repositoryRealPath: string;
  symbolicLink: boolean;
  regularFile: boolean;
  size: number;
  magic: Uint8Array;
  mime: AudioFixtureCommandResult;
  ignored: AudioFixtureCommandResult;
  tracked: AudioFixtureCommandResult;
  probe: AudioFixtureCommandResult;
  readFileError: Error;
  commandError: { command: string; error: Error };
  fileBytes: Uint8Array;
  onInputCommand: (command: string, input: Uint8Array) => void;
}>;

const ok = (stdout = ''): AudioFixtureCommandResult => ({
  exitCode: 0,
  stdout,
  timedOut: false,
});

const failed = (stdout = ''): AudioFixtureCommandResult => ({
  exitCode: 1,
  stdout,
  timedOut: false,
});

const validProbe = JSON.stringify({
  streams: [{ codec_name: 'mp3', codec_type: 'audio' }],
  format: { duration: '5.25', format_name: 'mp3' },
});

const createDependencies = (overrides: FakeOverrides = {}) => {
  const calls: Array<{
    command: string;
    args: readonly string[];
    cwd: string;
    timeoutMs: number;
    maxBufferBytes: number;
    input?: Uint8Array;
  }> = [];
  const dependencies: AudioFixtureDependencies = {
    realpath: async (path) => {
      if (path === REPOSITORY_ROOT) {
        return overrides.repositoryRealPath ?? REPOSITORY_ROOT;
      }
      if (path === EVIDENCE_ROOT) {
        return overrides.evidenceRealPath ?? EVIDENCE_ROOT;
      }
      return overrides.candidateRealPath ?? VALID_PATH;
    },
    lstat: async () => ({
      isFile: () => overrides.regularFile ?? true,
      isSymbolicLink: () => overrides.symbolicLink ?? false,
      size: overrides.size ?? 41_536,
    }),
    readStableFile: async () => {
      if (overrides.readFileError) throw overrides.readFileError;
      if (overrides.fileBytes) return overrides.fileBytes;
      const bytes = new Uint8Array(41_536).fill(0x11);
      bytes.set(
        overrides.magic ?? new Uint8Array([0x49, 0x44, 0x33, 0x04]),
      );
      return bytes;
    },
    execFile: async (command, args, options) => {
      calls.push({ command, args: [...args], ...options });
      if (overrides.commandError?.command === command) {
        throw overrides.commandError.error;
      }
      if (command === 'git' && args[0] === 'check-ignore') {
        return overrides.ignored ?? ok();
      }
      if (command === 'git' && args[0] === 'ls-files') {
        return overrides.tracked ?? failed();
      }
      throw new Error('Unexpected fake command.');
    },
    execFileWithInput: async (command, args, input, options) => {
      const inputCopy = Uint8Array.from(input);
      calls.push({ command, args: [...args], input: inputCopy, ...options });
      overrides.onInputCommand?.(command, inputCopy);
      if (overrides.commandError?.command === command) {
        throw overrides.commandError.error;
      }
      if (command === 'file') return overrides.mime ?? ok('audio/mpeg\n');
      if (command === 'ffprobe') return overrides.probe ?? ok(validProbe);
      throw new Error('Unexpected fake input command.');
    },
  };
  return { calls, dependencies };
};

const validEnvironment = (overrides: Record<string, string | undefined> = {}) => ({
  NUANUFLOW_AUDIO_FIXTURE: VALID_PATH,
  NUANUFLOW_AUDIO_ORACLE_TEXT: '  Synthetic expected sentence.  ',
  NUANUFLOW_AUDIO_ORACLE_SPEAKER: '  QA Speaker  ',
  ...overrides,
});

const validationCode = async (
  environment: Record<string, string | undefined>,
  overrides: FakeOverrides = {},
) => {
  const { dependencies } = createDependencies(overrides);
  try {
    await validatePrivateAudioFixture({
      dependencies,
      environment,
      repositoryRoot: REPOSITORY_ROOT,
    });
  } catch (error) {
    return {
      code: (error as { code?: unknown }).code,
      message: error instanceof Error ? error.message : String(error),
    };
  }
  throw new Error('Expected fixture validation to fail.');
};

  let dependencyCalled = false;
  const dependencies: AudioFixtureDependencies = {
    realpath: async () => {
      dependencyCalled = true;
      throw new Error('unexpected realpath');
    },
    lstat: async () => {
      dependencyCalled = true;
      throw new Error('unexpected lstat');
    },
    readStableFile: async () => {
      dependencyCalled = true;
      throw new Error('unexpected file read');
    },
    execFile: async () => {
      dependencyCalled = true;
      throw new Error('unexpected command');
    },
    execFileWithInput: async () => {
      dependencyCalled = true;
      throw new Error('unexpected input command');
    },
  };

  await expect(
    validatePrivateAudioFixture({
      dependencies,
      environment: {},
      repositoryRoot: REPOSITORY_ROOT,
    }),
  ).resolves.toBeUndefined();
  expect(dependencyCalled).toBe(false);
});

test('valid fixture returns a bounded descriptor and uses argument-array commands', async () => {
  const { calls, dependencies } = createDependencies();
  const descriptor = await validatePrivateAudioFixture({
    dependencies,
    environment: validEnvironment(),
    repositoryRoot: REPOSITORY_ROOT,
  });

  expect(descriptor).toMatchObject({
    basename: APPROVED_AUDIO_BASENAME,
    byteCount: 41_536,
    codecName: 'mp3',
    codecType: 'audio',
    durationSeconds: 5.25,
    formatName: 'mp3',
  });
  expect(descriptor?.oracleSpeaker).toBe('QA Speaker');
  expect(descriptor?.oracleText).toBe('Synthetic expected sentence.');
  expect(Object.keys(descriptor!)).not.toContain('oracleSpeaker');
  expect(Object.keys(descriptor!)).not.toContain('oracleText');
  expect(JSON.stringify(descriptor)).not.toContain('Synthetic expected sentence.');
  expect(JSON.stringify(descriptor)).not.toContain('QA Speaker');
  expect(calls.map(({ command }) => command)).toEqual([
    'git',
    'git',
    'file',
    'ffprobe',
  ]);
  for (const call of calls) {
    if (call.command === 'git') {
      expect(call.args.at(-1)).toBe(VALID_PATH);
      expect(call.args).toContain('--');
      expect(call.input).toBeUndefined();
    } else {
      expect(call.args).not.toContain(VALID_PATH);
      expect(call.input).toHaveLength(41_536);
      expect(call.input?.slice(0, 4)).toEqual(
        new Uint8Array([0x49, 0x44, 0x33, 0x04]),
      );
    }
    expect(call.cwd).toBe(REPOSITORY_ROOT);
    expect(call.timeoutMs).toBeGreaterThan(0);
    expect(call.maxBufferBytes).toBeLessThanOrEqual(65_536);
  }
});

test('validated fixture exposes no path and loads a bounded in-memory upload payload', async () => {
  const fileBytes = new Uint8Array(41_536).fill(0x22);
  fileBytes.set([0x49, 0x44, 0x33, 0x04]);
  const { dependencies } = createDependencies({ fileBytes });
  const descriptor = await validatePrivateAudioFixture({
    dependencies,
    environment: validEnvironment(),
    repositoryRoot: REPOSITORY_ROOT,
  });
  expect(descriptor).toBeDefined();
  expect(Object.keys(descriptor!)).not.toContain('filePath');
  expect(JSON.stringify(descriptor)).not.toContain(REPOSITORY_ROOT);

  const payload = await loadValidatedAudioFixture(descriptor!);
  expect(payload.name).toBe(APPROVED_AUDIO_BASENAME);
  expect(payload.mimeType).toBe('audio/mpeg');
  expect(payload.buffer).toEqual(Buffer.from(fileBytes));

  payload.buffer.fill(0);
  const secondPayload = await loadValidatedAudioFixture(descriptor!);
  expect(secondPayload.buffer).toEqual(Buffer.from(fileBytes));
  expect(secondPayload.buffer).not.toBe(payload.buffer);
});

test('MIME and ffprobe validate the one retained byte snapshot even when the path changes', async () => {
  const original = new Uint8Array(41_536).fill(0x41);
  original.set([0x49, 0x44, 0x33, 0x04]);
  const replacement = new Uint8Array(41_536).fill(0x52);
  replacement.set([0x49, 0x44, 0x33, 0x04]);
  const validatedInputs: Array<{ command: string; input: Uint8Array }> = [];
  const state: FakeOverrides = {
    fileBytes: original,
    onInputCommand: (command, input) => {
      validatedInputs.push({ command, input: Uint8Array.from(input) });
      state.fileBytes = replacement;
    },
  };
  const { dependencies } = createDependencies(state);
  const descriptor = await validatePrivateAudioFixture({
    dependencies,
    environment: validEnvironment(),
    repositoryRoot: REPOSITORY_ROOT,
  });

  expect(validatedInputs.map(({ command }) => command))
    .toEqual(['file', 'ffprobe']);
  for (const { input } of validatedInputs) {
    expect(input).toEqual(original);
    expect(input).not.toEqual(replacement);
  }
  await expect(loadValidatedAudioFixture(descriptor!)).rejects.toMatchObject({
    code: 'AUDIO_FIXTURE_CHANGED',
  });
});

test('transcript fidelity normalizes only Unicode and whitespace and redacts mismatch data', async () => {
  const { dependencies } = createDependencies();
  const descriptor = await validatePrivateAudioFixture({
    dependencies,
    environment: validEnvironment({
      NUANUFLOW_AUDIO_ORACLE_TEXT: 'Cafe\u0301: Synthetic sentence!',
      NUANUFLOW_AUDIO_ORACLE_SPEAKER: 'QA Speaker',
    }),
    repositoryRoot: REPOSITORY_ROOT,
  });

  expect(() => assertRedactedTranscriptFidelity(descriptor!, {
    transcriptText: 'Prefix\nCaf\u00e9:   Synthetic sentence!\nSuffix',
    speakerLabels: ['QA Speaker'],
  })).not.toThrow();

  for (const [actual, code] of [
    [
      { transcriptText: 'Caf\u00e9: synthetic sentence!', speakerLabels: ['QA Speaker'] },
      'AUDIO_TRANSCRIPT_TEXT_MISMATCH',
    ],
    [
      { transcriptText: 'Caf\u00e9: Synthetic sentence!', speakerLabels: ['Speaker 1'] },
      'AUDIO_TRANSCRIPT_SPEAKER_MISMATCH',
    ],
  ] as const) {
    let captured: unknown;
    try {
      assertRedactedTranscriptFidelity(descriptor!, actual);
    } catch (error) {
      captured = error;
    }
    expect(captured).toMatchObject({ code });
    expect(String(captured)).not.toContain('Caf');
    expect(String(captured)).not.toContain('Speaker 1');
    expect(String(captured)).not.toContain('QA Speaker');
  }
});

test('rejects malformed, oversized, and invalid ffprobe documents', async () => {
  const cases: Array<[string, AudioFixtureCommandResult, string]> = [
    ['malformed JSON', ok('{'), 'AUDIO_FIXTURE_PROBE_JSON'],
    ['oversized JSON', ok('x'.repeat(65_537)), 'AUDIO_FIXTURE_PROBE_OUTPUT'],
    [
      'no stream',
      ok(JSON.stringify({ streams: [], format: { duration: '5', format_name: 'mp3' } })),
      'AUDIO_FIXTURE_STREAM_COUNT',
    ],
    [
      'video stream',
      ok(JSON.stringify({
        streams: [{ codec_name: 'h264', codec_type: 'video' }],
        format: { duration: '5', format_name: 'mp3' },
      })),
      'AUDIO_FIXTURE_STREAM_TYPE',
    ],
    [
      'wrong codec',
      ok(JSON.stringify({
        streams: [{ codec_name: 'aac', codec_type: 'audio' }],
        format: { duration: '5', format_name: 'mp3' },
      })),
      'AUDIO_FIXTURE_CODEC',
    ],
    [
      'wrong container',
      ok(JSON.stringify({
        streams: [{ codec_name: 'mp3', codec_type: 'audio' }],
        format: { duration: '5', format_name: 'wav' },
      })),
      'AUDIO_FIXTURE_CONTAINER',
    ],
    [
      'non-finite duration',
      ok(JSON.stringify({
        streams: [{ codec_name: 'mp3', codec_type: 'audio' }],
        format: { duration: 'NaN', format_name: 'mp3' },
      })),
      'AUDIO_FIXTURE_DURATION',
    ],
    [
      'too-short duration',
      ok(JSON.stringify({
        streams: [{ codec_name: 'mp3', codec_type: 'audio' }],
        format: { duration: '0.5', format_name: 'mp3' },
      })),
      'AUDIO_FIXTURE_DURATION',
    ],
    [
      'too-long duration',
      ok(JSON.stringify({
        streams: [{ codec_name: 'mp3', codec_type: 'audio' }],
        format: { duration: '31', format_name: 'mp3' },
      })),
      'AUDIO_FIXTURE_DURATION',
    ],
  ];

  for (const [, probe, expectedCode] of cases) {
    expect((await validationCode(validEnvironment(), { probe })).code)
      .toBe(expectedCode);
  }
});

test('meeting cleanup continues after one GET or DELETE failure and aggregates exact residual IDs', async () => {
  const first = parsedOwnedEntity({ id: MEETING_IDS.first, requestOrdinal: 0 });
  const second = parsedOwnedEntity({ id: MEETING_IDS.second, requestOrdinal: 1 });

      return exactInspection(entity);

    residualIds: [MEETING_IDS.first, MEETING_IDS.second],
  });

});
