export type ResponseOwnedCleanupEntry<T extends { id: string }> = {
  requestOrdinal: number;
  created: T;
};

export const orderResponseOwnedCleanupBatch = <T extends { id: string }>(
  entries: Array<ResponseOwnedCleanupEntry<T>>,
): T[] => {
  const latestById = new Map<string, ResponseOwnedCleanupEntry<T>>();
  for (const entry of entries) {
    if (
      !Number.isInteger(entry.requestOrdinal) ||
      entry.requestOrdinal < 0 ||
      !isCanonicalEntityId(entry.created.id)
    ) {
      throw new Error('Cleanup ordering requires an exact request ordinal and ID.');
    }
    const current = latestById.get(entry.created.id);
    if (!current || current.requestOrdinal < entry.requestOrdinal) {
      latestById.set(entry.created.id, entry);
    }
  }
  return [...latestById.values()]
    .sort((left, right) => left.requestOrdinal - right.requestOrdinal)
    .map(({ created }) => created);
};

type JsonRecord = Record<string, unknown>;

export const SANDBOX_WORKSPACE_ID =
  '8f081389-298d-4dc2-93d5-7cd80c5109da';

const isRecord = (value: unknown): value is JsonRecord =>
  typeof value === 'object' && value !== null;

const assertOwnedName = (name: string) => {
  if (!name.startsWith(`${QA_PREFIX} `)) {
    throw new Error(`Knowledge names must start with ${QA_PREFIX}.`);
  }
};

const exactSandboxProjectIds = (value: unknown): value is string[] =>
  Array.isArray(value) &&
  value.length === 1 &&
  value[0] === SANDBOX.projectId;

const exactProjectLink = (value: unknown) =>
  isRecord(value) &&
  value.entity_type === 'project' &&
  value.entity_id === SANDBOX.projectId &&
  value.relation === 'about';

const hasAnyProjectLink = (links: unknown[]) =>
  links.some((link) => isRecord(link) && link.entity_type === 'project');

export const parseCreatedSavedView = (
  body: unknown,
  requestedName: string,
  requestedStateId: string,
): CreatedSavedView => {
  assertOwnedName(requestedName);
  if (!isCanonicalEntityId(requestedStateId)) {
    throw new Error('Saved-view state filter must be a canonical exact ID.');
  }
  if (!isRecord(body)) {
    throw new Error('Saved-view create response was not an object.');
  }
  const { id, name, project } = body;
  if (!isCanonicalEntityId(id) || typeof name !== 'string') {
    throw new Error('Saved-view create response omitted its exact ID contract.');
  }
  if (name !== requestedName || project !== SANDBOX.projectId) {
    throw new Error('Saved-view create response crossed its exact project scope.');
  }
  const richFilters = body.rich_filters;
  const clauses = isRecord(richFilters) ? richFilters.and : undefined;
  if (
    body.access !== 1 ||
    !Array.isArray(clauses) ||
    clauses.length !== 2 ||
    !clauses.some(
      (clause) => isRecord(clause) && clause.priority__in === 'urgent',
    ) ||
    !clauses.some(
      (clause) =>
        isRecord(clause) &&
        clause.state_id__in === requestedStateId,
    )
  ) {
    throw new Error('Saved-view create response omitted its deterministic filters.');
  }
  return {
    kind: 'saved-view',
    id,
    name,
    projectId: SANDBOX.projectId,
    stateId: requestedStateId,
  };
};

const artifactResponseRecords = (body: JsonRecord) => {
  const records: JsonRecord[] = [body];
  if (isRecord(body.artifact)) records.push(body.artifact);
  if (isRecord(body.data)) {
    records.push(body.data);
    if (isRecord(body.data.artifact)) records.push(body.data.artifact);
  }
  return records;
};

const artifactResponseIdCandidates = (body: JsonRecord) => {
  const artifact = isRecord(body.artifact) ? body.artifact : undefined;
  const data = isRecord(body.data) ? body.data : undefined;
  const dataArtifact = data && isRecord(data.artifact) ? data.artifact : undefined;
  const semanticCandidates: unknown[] = [
    body.artifact_id,
    typeof body.artifact === 'string' ? body.artifact : undefined,
    artifact?.artifact_id,
    artifact?.id,
    data?.artifact_id,
    typeof data?.artifact === 'string' ? data.artifact : undefined,
    dataArtifact?.artifact_id,
    dataArtifact?.id,
  ];
  const semanticIds = semanticCandidates.filter(
    (value): value is string => typeof value === 'string' && value.length > 0,
  );
  if (semanticIds.length > 0) return semanticIds;
  const fallbackCandidates = [body.id, data?.id].filter(
    (value): value is string => typeof value === 'string' && value.length > 0,
  );
  return fallbackCandidates;
};

export const extractArtifactResponseIdentity = (body: unknown) => {
  if (!isRecord(body)) return {};
  const [observedId] = artifactResponseIdCandidates(body);
  const exactId = isCanonicalEntityId(observedId) ? observedId : undefined;
  return { observedId, exactId };
};

const assertOptionalArtifactReservationFields = (
  records: JsonRecord[],
  requestedName: string,
  expectedScope: ArtifactScope,
) => {
  for (const record of records) {
    if (record.name !== undefined && record.name !== requestedName) {
      throw new Error('Artifact reservation crossed its exact requested name.');
    }
    if (
      record.workspace !== undefined &&
      record.workspace !== SANDBOX_WORKSPACE_ID
    ) {
      throw new Error('Artifact reservation crossed its exact workspace UUID.');
    }
    if (record.scope !== undefined && record.scope !== expectedScope) {
      throw new Error('Artifact reservation crossed its exact requested scope.');
    }
    if (
      record.project_id !== undefined &&
      (expectedScope !== 'project' || record.project_id !== SANDBOX.projectId)
    ) {
      throw new Error('Artifact reservation crossed its exact project scope.');
    }
    if (record.links !== undefined) {
      if (!Array.isArray(record.links)) {
        throw new Error('Artifact reservation links were not an array.');
      }
      const projectLinks = record.links.filter(
        (link) => isRecord(link) && link.entity_type === 'project',
      );
      if (
        (expectedScope === 'private' && projectLinks.length > 0) ||
        (expectedScope === 'project' &&
          (projectLinks.length !== 1 || !exactProjectLink(projectLinks[0])))
      ) {
        throw new Error('Artifact reservation crossed its exact project scope.');
      }
    }
  }
};

export const parseArtifactReservation = (
  body: unknown,
  requestedName: string,
  expectedScope: ArtifactScope,
): CreatedArtifact => {
  assertOwnedName(requestedName);
  if (!isRecord(body)) {
    throw new Error('Artifact reservation response was not an object.');
  }
  const candidates = artifactResponseIdCandidates(body);
  const uniqueCandidates = [...new Set(candidates)];
  const { exactId } = extractArtifactResponseIdentity(body);
  if (!exactId || uniqueCandidates.length !== 1) {
    throw new Error('Artifact reservation omitted one exact canonical artifact ID.');
  }
  assertOptionalArtifactReservationFields(
    artifactResponseRecords(body),
    requestedName,
    expectedScope,
  );
  return {
    kind: 'artifact',
    id: exactId,
    name: requestedName,
    workspaceId: SANDBOX_WORKSPACE_ID,
    scope: expectedScope,
    ...(expectedScope === 'project' ? { projectId: SANDBOX.projectId } : {}),
  };
};

export const parseCreatedArtifact = (
  body: unknown,
  requestedName: string,
  expectedScope: ArtifactScope,
): CreatedArtifact => {
  assertOwnedName(requestedName);
  if (!isRecord(body)) {
    throw new Error('Artifact detail response was not an object.');
  }
  const {
    id,
    name,
    workspace,
    scope,
    kind,
    status,
    links: rawLinks,
  } = body;
  if (
    !isCanonicalEntityId(id) ||
    typeof name !== 'string'
  ) {
    throw new Error('Artifact detail response omitted its exact ID contract.');
  }
  if (
    name !== requestedName ||
    workspace !== SANDBOX_WORKSPACE_ID ||
    scope !== expectedScope ||
    kind !== 'document' ||
    status !== 'stored' ||
    !Array.isArray(rawLinks)
  ) {
    throw new Error(
      workspace !== SANDBOX_WORKSPACE_ID
        ? 'Artifact detail response crossed its exact workspace UUID.'
        : 'Artifact detail response did not match the requested identity.',
    );
  }

  const projectLinks = rawLinks.filter(
    (link) => isRecord(link) && link.entity_type === 'project',
  );
  if (expectedScope === 'private') {
    if (hasAnyProjectLink(rawLinks)) {
      throw new Error('Private artifact response contained a cross-scope project link.');
    }
    return {
      kind: 'artifact',
      id,
      name,
      workspaceId: SANDBOX_WORKSPACE_ID,
      scope: expectedScope,
    };
  }

  if (projectLinks.length !== 1 || !exactProjectLink(projectLinks[0])) {
    throw new Error('Project artifact response crossed its exact project scope.');
  }
  return {
    kind: 'artifact',
    id,
    name,
    workspaceId: SANDBOX_WORKSPACE_ID,
    scope: expectedScope,
    projectId: SANDBOX.projectId,
  };
};

export const assertExactArtifactCleanupTarget = (
  body: unknown,
  owned: CreatedArtifact,
) => {
  assertOwnedName(owned.name);
  const ownedScopeIsExact =
    (owned.scope === 'private' && owned.projectId === undefined) ||
    (owned.scope === 'project' && owned.projectId === SANDBOX.projectId);
  if (
    !isCanonicalEntityId(owned.id) ||
    owned.workspaceId !== SANDBOX_WORKSPACE_ID ||
    !ownedScopeIsExact ||
    !isRecord(body) ||
    body.id !== owned.id ||
    body.name !== owned.name ||
    body.workspace !== SANDBOX_WORKSPACE_ID ||
    body.scope !== owned.scope ||
    !Array.isArray(body.links)
  ) {
    throw new Error(
      'Refusing artifact cleanup without an exact minimal ownership match.',
    );
  }
  if (
    (owned.scope === 'private' && body.links.length !== 0) ||
    (owned.scope === 'project' &&
      (body.links.length !== 1 || !exactProjectLink(body.links[0])))
  ) {
    throw new Error(
      'Refusing artifact cleanup without an exact project-link match.',
    );
  }
  return body;
};

const residualRecoveryError = (
  kind: string,
  id: string,
  error: unknown,
) =>
  new AggregateError(
    [error],
    `${kind} exact-ID recovery refused or failed; residual ID: ${id}`,
  );

export const reconcileRejectedSavedViewExactId = async ({
  id,
  requestedName,
  requestedStateId,
  inspect,
  cleanup,
}: ReconcileSavedViewExactIdOptions) => {
  if (!isCanonicalEntityId(id)) {
    throw new Error(`Saved-view recovery received a non-canonical ID: ${id}`);
  }
  let inspection: RecoveryInspection;
  try {
    inspection = await inspect(id);
  } catch (error) {
    throw residualRecoveryError('Saved-view', id, error);
  }
  if (inspection.status === 404) return;
  if (inspection.status !== 200) {
    throw residualRecoveryError(
      'Saved-view',
      id,
      new Error(`Ownership GET returned ${inspection.status}.`),
    );
  }
  let created: CreatedSavedView;
  try {
    created = assertExactSavedViewRecoveryTarget(
      inspection.body,
      id,
      requestedName,
      requestedStateId,
    );
  } catch (error) {
    throw residualRecoveryError('Saved-view', id, error);
  }
  try {
    await cleanup(created);
  } catch (error) {
    throw residualRecoveryError('Saved-view', id, error);
  }
};

  id,
  requestedName,
  inspect,
  cleanup,

  assertOwnedName(requestedName);
  if (!isCanonicalEntityId(id)) {
    throw new Error(`Wiki recovery received a non-canonical ID: ${id}`);
  }
  const provisional: ProvisionalWikiPage = {
    kind: 'wiki-provisional',
    id,
    name: '',
    projectIds: [SANDBOX.projectId],

  let inspection: RecoveryInspection;
  try {
    inspection = await inspect(id);
  } catch (error) {
    throw residualRecoveryError('Wiki', id, error);

  if (inspection.status === 404) return;
  if (inspection.status !== 200) {
    throw residualRecoveryError(
      'Wiki',
      id,

    );

  try {
    assertExactProvisionalWikiCleanupTarget(
      inspection.body,
      provisional,
      requestedName,
    );
  } catch (error) {
    throw residualRecoveryError('Wiki', id, error);
  }
  try {
    await cleanup(provisional, requestedName);
  } catch (error) {
    throw residualRecoveryError('Wiki', id, error);
  }
};

    !exactSandboxProjectIds(provisional.projectIds)

      'Refusing Wiki cleanup without an exact provisional ownership match.',
    );

type ProvisionalWikiRecoveryEntry = {
  provisional: ProvisionalWikiPage;
  cleanup: () => Promise<unknown>;
};

export type ProvisionalWikiResidual = {
  kind: 'wiki-provisional';
  id: string;
};

export class ProvisionalWikiRecoveryLedger {
  readonly #entries = new Map<string, ProvisionalWikiRecoveryEntry>();

  get size() {
    return this.#entries.size;
  }

  register(
    provisional: ProvisionalWikiPage,
    cleanup: () => Promise<unknown>,
  ) {
    if (
      provisional.name !== '' ||
      !isCanonicalEntityId(provisional.id) ||
      !exactSandboxProjectIds(provisional.projectIds) ||
      this.#entries.has(provisional.id)
    ) {
      throw new Error('Invalid provisional Wiki recovery registration.');
    }
    this.#entries.set(provisional.id, { provisional, cleanup });
  }

  promote(provisional: ProvisionalWikiPage, named: CreatedWikiPage) {
    const entry = this.#entries.get(provisional.id);
    if (
      !entry ||
      named.id !== provisional.id ||
      !named.name.startsWith(`${QA_PREFIX} `) ||
      !exactSandboxProjectIds(named.projectIds)
    ) {
      throw new Error('Cannot promote an unregistered provisional Wiki identity.');
    }
    this.#entries.delete(provisional.id);
  }

  async cleanupAll(): Promise<ProvisionalWikiResidual[]> {
    const residuals: ProvisionalWikiResidual[] = [];
    for (const [id, entry] of [...this.#entries.entries()].reverse()) {
      try {
        await entry.cleanup();
        this.#entries.delete(id);
      } catch {
        residuals.push({ kind: 'wiki-provisional', id });
      }
    }
    return residuals;
  }
}

type CleanupWikiAfterFailedSaveOptions = {
  promoted: boolean;
  named?: CreatedWikiPage;
  provisionalRecovery: ProvisionalWikiRecoveryLedger;
  cleanupNamed: (created: CreatedWikiPage) => Promise<unknown>;
};

export const cleanupWikiAfterFailedSave = async ({
  promoted,
  named,
  provisionalRecovery,
  cleanupNamed,
}: CleanupWikiAfterFailedSaveOptions): Promise<ProvisionalWikiResidual[]> => {
  const cleanupErrors: unknown[] = [];
  if (promoted) {
    if (!named) {
      cleanupErrors.push(
        new Error('Promoted Wiki recovery omitted its exact named identity.'),
      );
    } else {
      try {
        await cleanupNamed(named);
      } catch (error) {
        cleanupErrors.push(
          residualRecoveryError('Wiki', named.id, error),
        );
      }
    }
  }
  const residuals = await provisionalRecovery.cleanupAll();
  for (const { id } of residuals) {
    cleanupErrors.push(
      new Error(`Wiki provisional recovery left residual ID: ${id}`),
    );
  }
  aggregateErrors('Wiki save failure recovery was incomplete.', cleanupErrors);
  return residuals;
};

const responseRows = (body: unknown): JsonRecord[] => {
  if (Array.isArray(body)) return body.filter(isRecord);
  if (!isRecord(body)) return [];
  for (const key of ['results', 'views', 'pages', 'artifacts']) {
    const value = body[key];
    if (Array.isArray(value)) return value.filter(isRecord);
  }
  return [];
};

const responseWorkItems = (body: unknown) => {
  const items = new Map<string, JsonRecord>();
  const visit = (value: unknown) => {
    if (Array.isArray(value)) {
      for (const entry of value) visit(entry);
      return;
    }
    if (!isRecord(value)) return;
    if (
      isCanonicalEntityId(value.id) &&
      typeof value.name === 'string' &&
      value.project_id === SANDBOX.projectId &&
      typeof value.sequence_id === 'number'
    ) {
      items.set(value.id, value);
    }
    for (const nested of Object.values(value)) visit(nested);
  };
  visit(body);
  return [...items.values()];
};

const queryTokens = (url: URL) => {
  const tokens = new Set<string>();
  const visit = (value: unknown) => {
    if (typeof value === 'string') {
      tokens.add(value);
      for (const token of value.split(/[\s,\[\]{}":]+/).filter(Boolean)) {
        tokens.add(token);
      }
      try {
        visit(JSON.parse(value));
      } catch {
        // Non-JSON query values are already represented by exact tokens above.
      }
      return;
    }
    if (Array.isArray(value)) {
      for (const entry of value) visit(entry);
      return;
    }
    if (isRecord(value)) {
      for (const [key, nested] of Object.entries(value)) {
        tokens.add(key);
        visit(nested);
      }
    }
  };
  for (const [key, value] of url.searchParams) {
    tokens.add(key);
    visit(value);
  }
  return tokens;
};

const aggregateErrors = (message: string, errors: unknown[]) => {
  if (errors.length === 0) return;
  if (errors.length === 1) throw errors[0];
  throw new AggregateError(errors, message);
};

type CleanupResponse = {
  status: () => number;
  json: () => Promise<unknown>;
};

  delete: (url: string) => Promise<{ status: () => number }>;
};

type ExactBatchIdentity = {
  id: string;
  name: string;
};

type ExactBatchCleanupOptions<T extends ExactBatchIdentity> = {
  page: Page;
  requestContext: KnowledgeCleanupRequestContext;
  assertSandbox: () => Promise<void>;
  created: T[];
  detailEndpoint: (id: string) => string;
  collectionEndpoint: string;
  uiPath: string;
  kind: string;
  assertIdentity: (created: T) => void;
  assertTarget: (body: unknown, created: T) => void;
  alreadyResolvedIds?: ReadonlySet<string>;
  onResolved?: (created: T) => void;
};

const assertExactOwnedArtifactIdentity = (created: CreatedArtifact) => {
  assertOwnedName(created.name);
  if (
    !isCanonicalEntityId(created.id) ||
    created.workspaceId !== SANDBOX_WORKSPACE_ID ||
    (created.scope === 'private' && created.projectId !== undefined) ||
    (created.scope === 'project' && created.projectId !== SANDBOX.projectId)
  ) {
    throw new Error('Artifact cleanup identity was not exact and run-owned.');
  }
};

const throwBatchCleanupErrors = (
  kind: string,
  errors: unknown[],
  residualIds: Set<string>,
) => {
  if (errors.length === 0) return;
  const ids = [...residualIds];
  if (ids.length === 0) {
    aggregateErrors(`${kind} batch cleanup was incomplete.`, errors);
    return;
  }
  const residualLabel =
    ids.length === 1
      ? `residual ID: ${ids[0]}`
      : `residual IDs: ${ids.join(', ')}`;
  throw new AggregateError(
    errors,
    `${kind} exact-ID batch cleanup refused or failed; ${residualLabel}`,
  );
};

const cleanupCapturedExactBatch = async <T extends ExactBatchIdentity>({
  page,
  requestContext,
  assertSandbox,
  created,
  detailEndpoint,
  collectionEndpoint,
  uiPath,
  kind,
  assertIdentity,
  assertTarget,
  alreadyResolvedIds,
  onResolved,
}: ExactBatchCleanupOptions<T>) => {
  if (created.length === 0) return;
  const unique = new Map<string, T>();
  for (const entity of created) {
    assertIdentity(entity);
    if (unique.has(entity.id)) {
      throw new Error(`${kind} batch contained a duplicate exact response ID.`);
    }
    unique.set(entity.id, entity);
  }

  const errors: unknown[] = [];
  const residualIds = new Set<string>();
  const deleteTargets: Array<{ entity: T; endpoint: string }> = [];
  const recordResidual = (entity: T, error: unknown) => {
    residualIds.add(entity.id);
    errors.push(residualRecoveryError(kind, entity.id, error));
  };
  const markResolved = (entity: T) => {
    try {
      onResolved?.(entity);
    } catch (error) {
      errors.push(error);
    }
  };

  for (const entity of unique.values()) {
    if (alreadyResolvedIds?.has(entity.id)) continue;
    try {
      const endpoint = detailEndpoint(entity.id);
      const inspection = await requestContext.get(endpoint);
      if (inspection.status() === 404) {
        markResolved(entity);
        continue;
      }
      if (inspection.status() !== 200) {
        throw new Error(`Ownership GET returned ${inspection.status()}.`);
      }
      assertTarget(await inspection.json(), entity);
      deleteTargets.push({ entity, endpoint });
    } catch (error) {
      recordResidual(entity, error);
    }
  }

  for (const { entity, endpoint } of [...deleteTargets].reverse()) {
    try {
      await assertSandbox();
      const deleted = await requestContext.delete(endpoint);
      if (deleted.status() !== 204) {
        throw new Error(`Exact-ID DELETE returned ${deleted.status()}.`);
      }
      markResolved(entity);
    } catch (error) {
      recordResidual(entity, error);
    }
  }

  let rows: JsonRecord[] | undefined;
  try {
    const collection = await requestContext.get(collectionEndpoint);
    if (collection.status() !== 200) {
      throw new Error(`${kind} collection GET returned ${collection.status()}.`);
    }
    rows = responseRows(await collection.json());
  } catch (error) {
    errors.push(error);
  }

  const names = new Set([...unique.values()].map(({ name }) => name));
  if (rows) {
    for (const row of rows) {
      if (typeof row.id !== 'string' || !unique.has(row.id)) continue;
      const entity = unique.get(row.id)!;
      if (!residualIds.has(entity.id)) {
        recordResidual(
          entity,
          new Error(`${kind} remained in its exact collection after cleanup.`),
        );
      }
    }
    if (
      rows.some(
        (row) =>
          typeof row.name === 'string' &&
          names.has(row.name) &&
          !(typeof row.id === 'string' && unique.has(row.id)),
      )
    ) {
      errors.push(
        new Error(
          `${kind} collection retained a shared current-run name after all exact-ID deletes.`,
        ),
      );
    }
  }

  let uiLoaded = false;
  try {
    await page.goto(uiPath, { waitUntil: 'domcontentloaded' });
    uiLoaded = true;
  } catch (error) {
    errors.push(error);
  }
  if (uiLoaded) {
    for (const { id } of unique.values()) {
      try {
        await expect(page.getByText(id, { exact: true })).toHaveCount(0);
      } catch (error) {
        errors.push(error);
      }
    }
    for (const name of names) {
      try {
        await expect(page.getByText(name, { exact: true })).toHaveCount(0);
      } catch (error) {
        errors.push(error);
      }
    }
  }

  throwBatchCleanupErrors(kind, errors, residualIds);
};

export const cleanupCapturedArtifactBatch = (
  options: ArtifactBatchCleanupOptions,
) =>
  cleanupCapturedExactBatch({
    ...options,
    kind: 'Artifact',
    assertIdentity: assertExactOwnedArtifactIdentity,
    assertTarget: assertExactArtifactCleanupTarget,
  });

  async createSavedView(
    name: string,
    stateId: string,
    register: KnowledgeRegistration<CreatedSavedView>,
  ): Promise<CreatedSavedView> {
    assertOwnedName(name);
    if (!isCanonicalEntityId(stateId)) {
      throw new Error('Saved-view state filter must be a canonical exact ID.');
    }
    await this.gotoViews();
    const endpoint = this.absolute(VIEWS_API_PATH);
    const payload = {
      name,
      description: 'Synthetic QA saved view with no private data.',
      access: 1,
      rich_filters: {
        and: [
          { priority__in: 'urgent' },
          { state_id__in: stateId },
        ],
      },
    };
    const capture = await captureExactFeaturePosts<CreatedSavedView>({
      page: this.page,
      endpoint,
      parse: (body) => parseCreatedSavedView(body, name, stateId),
      register: async (created) => {
        this.registerOwned(created);
        await register(created);
      },
      action: () =>
        runSandboxMutation(this.assertSandbox, () =>
          this.page.evaluate(
            async ({ requestUrl, requestBody }) => {
              const response = await fetch(requestUrl, {
                method: 'POST',
                credentials: 'include',
                headers: { 'content-type': 'application/json' },
                body: JSON.stringify(requestBody),
              });
              if (!response.ok) {
                throw new Error(`Saved-view create failed with ${response.status}.`);
              }
            },
            { requestUrl: endpoint, requestBody: payload },
          ),
        ),
    });

    const recoveryErrors: unknown[] = [];
    const responseOwnedForCleanup: Array<
      ResponseOwnedCleanupEntry<CreatedSavedView>
    > = [
      ...capture.createdEntries,
      ...capture.registrationErrors.map(({ requestOrdinal, created }) => ({
        requestOrdinal,
        created,
      })),
    ];
    for (const failure of capture.contractErrors) {
      if (!failure.exactId) continue;
      responseOwnedForCleanup.push({
        requestOrdinal: failure.requestOrdinal,
        created: {
          kind: 'saved-view',
          id: failure.exactId,
          name,
          projectId: SANDBOX.projectId,
          stateId,
        },
      });
      try {
        await this.recoverSavedView(
          failure.exactId,
          name,
          stateId,
          () => {},
        );
      } catch (error) {
        recoveryErrors.push(error);
      }
    }
    const contractFailures: unknown[] = [
      ...capture.contractErrors.map(({ error }) => error),
      ...capture.registrationErrors.map(({ error }) => error),
      ...capture.non201.map(({ status }) =>
        new Error(`Saved-view create returned ${status}.`),
      ),
      ...capture.failedAttempts.map(() =>
        new Error('Saved-view create request failed before a response.'),
      ),
      ...recoveryErrors,
    ];
    if (capture.actionError !== undefined) contractFailures.push(capture.actionError);
    if (
      capture.attemptCount !== 1 ||
      capture.responseCount !== 1 ||
      capture.created.length !== 1
    ) {
      contractFailures.push(new Error('Saved-view create did not produce one exact response-owned entity.'));
    }
    if (contractFailures.length > 0) {
      try {
        await this.cleanupCapturedSavedViews(
          orderResponseOwnedCleanupBatch(responseOwnedForCleanup),
        );
      } catch (error) {
        contractFailures.push(error);
      }
    }
    aggregateErrors('Saved-view create failed after guarded recovery.', contractFailures);
    const created = capture.created[0];
    if (!created) throw new Error('Saved-view create omitted its response-owned result.');
    return created;
  }

  async assertSavedViewMembership(
    created: CreatedSavedView,
    item: CreatedWorkItem,

      item.projectId !== SANDBOX.projectId ||

  async createWikiPage(
    name: string,
    body: string,
    register: KnowledgeRegistration<CreatedWikiPage>,
  ): Promise<CreatedWikiPage> {
    assertOwnedName(name);
    await this.gotoWiki();
    const create = this.page
      .getByRole('button', { name: 'Create your first Page', exact: true })
      .or(this.page.getByRole('button', { name: 'Add context', exact: true }));
    await expect(create).toHaveCount(1);
    const endpoint = this.absolute(WIKI_API_PATH);
    const provisionalCapture = await captureExactFeaturePosts<ProvisionalWikiPage>({
      page: this.page,
      endpoint,
      parse: parseProvisionalWikiPage,
      register: (provisional) => {
        this.#provisionalWiki.register(provisional, () =>
          this.cleanupProvisionalWiki(provisional, name),
        );
      },
      action: () => runSandboxMutation(this.assertSandbox, () => create.click()),
    });
    const provisional = provisionalCapture.created[0];
    const provisionalRecoveryErrors: unknown[] = [];
    const rejectedIds = new Set<string>();
    for (const failure of provisionalCapture.contractErrors) {
      if (!failure.exactId || rejectedIds.has(failure.exactId)) continue;
      rejectedIds.add(failure.exactId);
      try {
        await this.recoverRejectedWiki(failure.exactId, name);
      } catch (error) {
        provisionalRecoveryErrors.push(error);
      }
    }
    for (const failure of provisionalCapture.registrationErrors) {
      if (rejectedIds.has(failure.created.id)) continue;
      rejectedIds.add(failure.created.id);
      try {
        await this.recoverRejectedWiki(failure.created.id, name);
      } catch (error) {
        provisionalRecoveryErrors.push(error);
      }
    }
    const invalidProvisionalCapture =
      !provisional ||
      provisionalCapture.attemptCount !== 1 ||
      provisionalCapture.responseCount !== 1 ||
      provisionalCapture.contractErrors.length > 0 ||
      provisionalCapture.registrationErrors.length > 0 ||
      provisionalCapture.non201.length > 0 ||
      provisionalCapture.failedAttempts.length > 0 ||
      provisionalCapture.actionError !== undefined;
    if (invalidProvisionalCapture) {
      const residuals = await this.#provisionalWiki.cleanupAll();
      const failures: unknown[] = [
        ...provisionalCapture.contractErrors.map(({ error }) => error),
        ...provisionalCapture.registrationErrors.map(({ error }) => error),
        ...provisionalCapture.non201.map(
          ({ status }) => new Error(`Wiki create returned ${status}.`),
        ),
        ...provisionalCapture.failedAttempts.map(
          ({ url }) => new Error(`Wiki create request failed: ${url}`),
        ),
        ...provisionalRecoveryErrors,
      ];
      if (provisionalCapture.actionError !== undefined) {
        failures.push(provisionalCapture.actionError);
      }
      if (
        provisionalCapture.attemptCount !== 1 ||
        provisionalCapture.responseCount !== 1 ||
        provisionalCapture.created.length !== 1
      ) {
        failures.push(
          new Error('Wiki create did not produce one exact response-owned entity.'),
        );
      }
      if (residuals.length > 0) {
        failures.push(
          new Error(
            `Residual provisional Wiki IDs: ${residuals
              .map(({ id }) => id)
              .join(', ')}`,
          ),
        );
      }
      aggregateErrors('Wiki provisional create failed after guarded recovery.', failures);
      throw new Error('Wiki provisional create failed without an explicit cause.');
    }

    let named: CreatedWikiPage | undefined;
    let promoted = false;
    try {
      const detailEndpoint = this.absolute(`${WIKI_API_PATH}${provisional.id}/`);
      const titleResponse = waitForExactResponse(
        this.page,
        'PATCH',
        detailEndpoint,
        { timeout: 15_000 },
      );
      const title = this.page.getByRole('textbox', { name: 'Untitled', exact: true });
      await expect(title).toHaveAttribute('maxlength', '255');
      await this.assertSandbox();
      await title.fill(name);
      const titleSaved = await titleResponse;
      expect(titleSaved.status()).toBe(200);
      const created = parseNamedWikiPage(
        await titleSaved.json(),
        provisional,
        name,
      );
      named = created;
      this.registerOwned(created);
      await register(created);
      this.#provisionalWiki.promote(provisional, created);
      promoted = true;

      const editor = this.page.locator('main [contenteditable="true"]');
      await expect(editor).toHaveCount(1);
      const descriptionEndpoint = this.absolute(
        `${WIKI_API_PATH}${created.id}/description/`,
      );
      const bodyResponse = waitForExactResponse(
        this.page,
        'PATCH',
        descriptionEndpoint,
        { timeout: 15_000 },
      );
      await this.assertSandbox();
      await editor.fill(body);
      expect((await bodyResponse).status()).toBe(200);
      await editor.selectText();
      const bold = this.page.locator('button:has(svg.lucide-bold)');
      await expect(bold).toHaveCount(1);
      const formattedResponse = waitForExactResponse(
        this.page,
        'PATCH',
        descriptionEndpoint,
        { timeout: 15_000 },
      );
      await this.assertSandbox();
      await bold.click();
      expect((await formattedResponse).status()).toBe(200);
      return created;
    } catch (error) {
      const failures: unknown[] = [error];
      try {
        await cleanupWikiAfterFailedSave({
          promoted,
          named,
          provisionalRecovery: this.#provisionalWiki,
          cleanupNamed: (created) => this.cleanupOwnedWiki(created),
        });
      } catch (cleanupError) {
        failures.push(cleanupError);
      }
      aggregateErrors('Wiki title/body save failed after guarded recovery.', failures);
      throw new Error('Wiki title/body save failed without an explicit cause.');
    }
  }

  async createWikiPageWithDescriptionHtml(
    name: string,
    descriptionHtml: string,
    register: KnowledgeRegistration<CreatedWikiPage>,
  ): Promise<DirectWikiCreateObservation> {
    assertOwnedName(name);
    if (
      descriptionHtml.trim().length === 0 ||
      !descriptionHtml.includes(QA_PREFIX)
    ) {
      throw new Error('Direct Wiki HTML must be non-empty synthetic QA E2E content.');
    }
    await this.gotoWiki();
    const endpoint = this.absolute(WIKI_API_PATH);
    if (new URL(endpoint).origin !== new URL(this.page.url()).origin) {
      throw new Error('Refusing a cross-origin direct Wiki create.');
    }

    await this.assertSandbox();
    const response = await this.requestContext.post(endpoint, {
      data: { name, description_html: descriptionHtml },
      failOnStatusCode: false,
      maxRedirects: 0,
    });
    const status = response.status();
    let responseBody: unknown;
    try {
      responseBody = await response.json();
    } catch {
      responseBody = undefined;
    }

    const exactId =
      isRecord(responseBody) && isCanonicalEntityId(responseBody.id)
        ? responseBody.id
        : undefined;
    if (!exactId) {
      await this.assertWikiNameAbsent(name);
      return { status };
    }

    let created: CreatedWikiPage;
    try {
      created = parseDirectCreatedWikiPage(responseBody, name);
    } catch (error) {
      try {
        await this.recoverRejectedWiki(exactId, name);
      } catch (recoveryError) {
        throw new AggregateError(
          [error, recoveryError],
          `Direct Wiki response recovery failed; residual ID: ${exactId}`,
        );
      }
      throw error;
    }

    this.registerOwned(created);
    try {
      await register(created);
    } catch (error) {
      const failures: unknown[] = [error];
      try {
        await this.cleanupOwnedWiki(created);
      } catch (cleanupError) {
        failures.push(cleanupError);
      }
      throw new AggregateError(
        failures,
        'Direct Wiki cleanup registration failed after guarded recovery.',
      );
    }
    return { status, created };
  }

  async assertWikiNameAbsent(name: string) {
    assertOwnedName(name);

  async uploadArtifact(
    input: ArtifactUploadInput,
    register: KnowledgeRegistration<CreatedArtifact>,
  ): Promise<CreatedArtifact> {
    assertArtifactUploadPayload(input.payload, input.name);
    if (input.scope === 'private') await this.gotoWorkspaceArtifacts();
    else await this.gotoProjectArtifacts();
    const endpoint = this.absolute(ARTIFACTS_API_PATH);
    const upload = this.page.getByRole('button', { name: 'Upload', exact: true });
    await expect(upload).toHaveCount(1);

    const capture = await captureExactFeaturePosts<CreatedArtifact>({
      page: this.page,
      endpoint,
      extractResponseIdentity: extractArtifactResponseIdentity,
      parse: (body) =>
        parseArtifactReservation(body, input.name, input.scope),
      register: async (reserved) => {
        this.registerOwned(reserved);
        await register(reserved);
      },
      waitBeforeAdmissionClose: waitForArtifactAdmission,
      action: async () => {
        const chooserPromise = this.page.waitForEvent('filechooser');
        await upload.click();
        const chooser = await chooserPromise;
        expect(chooser.isMultiple()).toBe(false);
        await runSandboxMutation(this.assertSandbox, () =>
          chooser.setFiles(input.payload),
        );
      },
    });

    const responseOwnedForCleanup: Array<
      ResponseOwnedCleanupEntry<CreatedArtifact>
    > = [
      ...capture.createdEntries,
      ...capture.registrationErrors.map(({ requestOrdinal, created }) => ({
        requestOrdinal,
        created,
      })),
    ];
    for (const failure of capture.contractErrors) {
      if (!failure.exactId) continue;
      responseOwnedForCleanup.push({
        requestOrdinal: failure.requestOrdinal,
        created: {
          kind: 'artifact',
          id: failure.exactId,
          name: input.name,
          workspaceId: SANDBOX_WORKSPACE_ID,
          scope: input.scope,
          ...(input.scope === 'project'
            ? { projectId: SANDBOX.projectId }
            : {}),
        },
      });
    }

    const failures: unknown[] = [
      ...capture.contractErrors.map(({ error }) => error),
      ...capture.registrationErrors.map(({ error }) => error),
      ...capture.non201.map(
        ({ status }) => new Error(`Artifact reservation returned ${status}.`),
      ),
      ...capture.failedAttempts.map(
        ({ url }) => new Error(`Artifact reservation request failed: ${url}`),
      ),
    ];
    if (capture.actionError !== undefined) failures.push(capture.actionError);
    if (
      capture.attemptCount !== 1 ||
      capture.responseCount !== 1 ||
      capture.created.length !== 1
    ) {
      failures.push(
        new Error('Artifact upload did not produce one exact POST-201 reservation.'),
      );
    }

    if (failures.length > 0) {
      try {
        await this.cleanupCapturedArtifacts(
          orderResponseOwnedCleanupBatch(responseOwnedForCleanup),
        );
      } catch (error) {
        failures.push(error);
      }
      aggregateErrors('Artifact upload failed after guarded recovery.', failures);
      throw new Error('Artifact upload failed without an explicit cause.');
    }

    const reserved = capture.created[0];
    if (!reserved) {
      throw new Error('Artifact upload omitted its response-owned reservation.');
    }
    try {
      return await this.awaitStoredArtifact(reserved);
    } catch (error) {
      const failuresAfterReservation: unknown[] = [error];
      try {
        await this.cleanupCapturedArtifact(reserved);
      } catch (cleanupError) {
        failuresAfterReservation.push(cleanupError);
      }
      aggregateErrors(
        'Artifact detail validation failed after guarded recovery.',
        failuresAfterReservation,
      );
      throw new Error('Artifact detail validation failed without an explicit cause.');
    }
  }

  private async cleanupWikiIdentity(
    provisional: ProvisionalWikiPage,
    requestedName: string,
  ) {
    const endpoint = this.absolute(`${WIKI_API_PATH}${provisional.id}/`);
    let inspection = await this.requestContext.get(endpoint);
    if (inspection.status() === 404) return;
    expect(inspection.status()).toBe(200);
    let body: unknown = await inspection.json();
    assertExactProvisionalWikiCleanupTarget(body, provisional, requestedName);
    if (!isRecord(body) || !body.archived_at) {
      await this.page.goto(`${WIKI_PATH}${provisional.id}/`, {
        waitUntil: 'domcontentloaded',
      });
      const menu = this.page.locator(
        'main main button:not([aria-label]):has(svg.lucide-ellipsis)',
      );
      await expect(menu).toHaveCount(1);
      await menu.click();
      const archive = this.page.getByRole('menuitem', { name: 'Archive', exact: true });
      await expect(archive).toHaveCount(1);
      await this.assertSandbox();
      await archive.click();
      await expect.poll(async () => {
        inspection = await this.requestContext.get(endpoint);
        if (inspection.status() !== 200) return false;
        body = await inspection.json();
        assertExactProvisionalWikiCleanupTarget(body, provisional, requestedName);
        return isRecord(body) && Boolean(body.archived_at);
      }).toBe(true);
    }
    await this.assertSandbox();
    const deleted = await this.requestContext.delete(endpoint);
    expect(deleted.status()).toBe(204);
    expect((await this.requestContext.get(endpoint)).status()).toBe(404);
    await this.gotoWiki();
    await expect(this.page.getByText(requestedName, { exact: true })).toHaveCount(0);
  }

  private async cleanupCapturedSavedViews(created: CreatedSavedView[]) {
    const unique = [...new Map(created.map((view) => [view.id, view])).values()];
    const alreadyResolvedIds = new Set(
      unique
        .filter((view) => {
          const record = this.#owned.get(view.id);
          return (
            record?.deleted === true &&
            record.created.kind === 'saved-view' &&
            record.created.name === view.name &&
            record.created.projectId === view.projectId &&
            record.created.stateId === view.stateId
          );
        })
        .map(({ id }) => id),
    );
    await cleanupCapturedSavedViewBatch({
      page: this.page,
      requestContext: this.requestContext,
      assertSandbox: this.assertSandbox,
      created: unique,
      detailEndpoint: (id) => this.absolute(`${VIEWS_API_PATH}${id}/`),
      collectionEndpoint: this.absolute(VIEWS_API_PATH),
      uiPath: VIEWS_PATH,
      alreadyResolvedIds,
      onResolved: ({ id }) => {
        const record = this.#owned.get(id);
        if (record?.created.kind === 'saved-view') record.deleted = true;
      },
    });
  }

  private async cleanupCapturedArtifacts(created: CreatedArtifact[]) {
    const unique = [
      ...new Map(created.map((artifact) => [artifact.id, artifact])).values(),
    ];
    if (unique.length === 0) return;
    const scope = unique[0]!.scope;
    if (unique.some((artifact) => artifact.scope !== scope)) {
      throw new Error('Artifact cleanup batch crossed its exact requested scope.');
    }
    const collectionPath =
      scope === 'project'
        ? `${ARTIFACTS_API_PATH}?entity_type=project&entity_id=${SANDBOX.projectId}`
        : ARTIFACTS_API_PATH;
    const uiPath =
      scope === 'project' ? PROJECT_ARTIFACTS_PATH : WORKSPACE_ARTIFACTS_PATH;
    await cleanupCapturedArtifactBatch({
      page: this.page,
      requestContext: this.requestContext,
      assertSandbox: this.assertSandbox,
      created: unique,
      detailEndpoint: (id) => this.absolute(`${ARTIFACTS_API_PATH}${id}/`),
      collectionEndpoint: this.absolute(collectionPath),
      uiPath,
      onResolved: ({ id }) => {
        const record = this.#owned.get(id);
        if (record?.created.kind === 'artifact') record.deleted = true;
      },
    });
  }

  private async recoverSavedView(
    id: string,
    name: string,
    stateId: string,
    cleanup: KnowledgeRegistration<CreatedSavedView> = async (created) => {
      this.registerOwned(created);
      await this.cleanupOwnedSavedView(created);
    },
  ) {
    await reconcileRejectedSavedViewExactId({
      id,
      requestedName: name,
      requestedStateId: stateId,
      inspect: async () => {
        const response = await this.requestContext.get(
          this.absolute(`${VIEWS_API_PATH}${id}/`),
        );
        return {
          status: response.status(),
          ...(response.status() === 200 ? { body: await response.json() } : {}),
        };
      },
      cleanup: async (created) => cleanup(created),
    });
  }

  private async recoverRejectedWiki(id: string, name: string) {
    await reconcileRejectedWikiExactId({
      id,
      requestedName: name,
      inspect: async () => {
        const response = await this.requestContext.get(
          this.absolute(`${WIKI_API_PATH}${id}/`),
        );
        return {
          status: response.status(),
          ...(response.status() === 200 ? { body: await response.json() } : {}),
        };
      },
      cleanup: (provisional, requestedName) =>
        this.cleanupWikiIdentity(provisional, requestedName),
    });
  }

  private async awaitStoredArtifact(
    reserved: CreatedArtifact,
  ): Promise<CreatedArtifact> {
    let stored: CreatedArtifact | undefined;
    await expect
      .poll(
        async () => {
          const response = await this.requestContext.get(
            this.absolute(`${ARTIFACTS_API_PATH}${reserved.id}/`),
          );
          if (response.status() !== 200) return response.status();
          const parsed = parseCreatedArtifact(
            await response.json(),
            reserved.name,
            reserved.scope,
          );
          if (
            parsed.id !== reserved.id ||
            parsed.workspaceId !== reserved.workspaceId ||
            parsed.projectId !== reserved.projectId
          ) {
            throw new Error('Artifact detail crossed its exact reservation identity.');
          }
          stored = parsed;
          return 200;
        },
        { timeout: 30_000 },
      )
      .toBe(200);
    if (!stored) {
      throw new Error(
        `Artifact detail never confirmed stored identity for ${reserved.id}.`,
      );
    }
    return stored;
  }

  private async cleanupCapturedArtifact(reserved: CreatedArtifact) {
    await this.cleanupCapturedArtifacts([reserved]);
  }

  private registerOwned(created: ResponseOwnedKnowledge) {
    if (this.#owned.has(created.id)) {
      throw new Error('Knowledge entity was already registered by exact response ID.');
    }
    this.#owned.set(created.id, { created, deleted: false });
  }

  private assertRegistered(created: ResponseOwnedKnowledge) {
    const record = this.#owned.get(created.id);
    if (!record || record.created.kind !== created.kind) {
      throw new Error('Knowledge entity is absent from the exact response ledger.');
    }
  }
