const expectHealthyCapture = (
  capture: MeetingMutationCapture,
  expectedSuccesses = 1,
) => {
  expect(capture.actionError).toBeUndefined();
  expect(capture.attemptCount).toBe(expectedSuccesses);
  expect(capture.responseCount).toBe(expectedSuccesses);
  expect(capture.successfulEntries).toHaveLength(expectedSuccesses);
  expect(capture.nonSuccessResponses).toEqual([]);
  expect(capture.contractErrors).toEqual([]);
  expect(capture.registrationErrors).toEqual([]);
  expect(capture.requestFailedOutcomes).toEqual([]);
  expect(capture.unexpectedMutationAttempts).toEqual([]);
};

    page,
    assertSandbox,
  }) => {
    const meetings = new MeetingsPage(page, assertSandbox);
    await meetings.gotoHome();
    await assertSandbox();
    await expect(meetings.addTranscriptButton).toBeVisible();
    await expect(meetings.uploadAudioButton).toBeVisible();

  test('validates and cancels a manual transcript with zero mutations', async ({
    page,
    assertSandbox,
    ownedName,
  }) => {
    const meetings = new MeetingsPage(page, assertSandbox);
    const cancelledTitle = ownedName('cancelled manual transcript');
    await meetings.gotoHome();
    await assertSandbox();
    await meetings.openManualTranscript();

    await expect(meetings.manualTitleInput).toBeVisible();
    await expect(meetings.manualDateTimeInput).toHaveCount(1);
    await expect(meetings.manualParticipantsInput).toBeVisible();
    await expect(meetings.manualTranscriptInput).toBeVisible();
    await expect(meetings.manualTitleInput).toHaveAttribute('maxlength', '255');

    const emptyCapture = await captureBlockedMeetingMutations({
      page,
      guardPatterns: meetings.mutationGuardPatterns(),
      action: () => meetings.manualSubmitButton.click(),
    });
    expect(emptyCapture.actionError).toBeUndefined();
    expect(emptyCapture.attemptCount).toBe(0);
    await expect(page.getByText('Title is required', { exact: true }))
      .toBeVisible();
    await expect(page.getByText('Transcript is required', { exact: true }))
      .toBeVisible();

    await meetings.fillManualTranscript({
      title: cancelledTitle,
      transcriptText: MANUAL_TRANSCRIPT,
    });
    const cancelCapture = await captureBlockedMeetingMutations({
      page,
      guardPatterns: meetings.mutationGuardPatterns(),
      action: () => meetings.cancelManualTranscript(),
    });
    expect(cancelCapture.actionError).toBeUndefined();
    expect(cancelCapture.attemptCount).toBe(0);
    await expect(meetings.meetingLink(cancelledTitle)).toHaveCount(0);
  });

  test('creates, reads back, reloads, and exactly cleans one manual transcript', async ({
    page,
    assertSandbox,
    cleanupLedger,
    ownedName,
  }) => {
    test.fixme(
      true,
      'Live browser recon could not attach a supported response-time interceptor; exact-title recovery cleaned the single discovery record but must not be repeated.',
    );
    test.setTimeout(90_000);
    const meetings = new MeetingsPage(page, assertSandbox);
    const title = ownedName('manual transcript lifecycle');
    const owned: OwnedMeetingEntity[] = [];
    let cleanupAnchorRegistered = false;

    const register = (entity: OwnedMeetingEntity) => {
      owned.push(entity);
      if (cleanupAnchorRegistered) return;
      cleanupAnchorRegistered = true;
      cleanupLedger.register({
        kind: 'meeting-transaction',
        id: entity.id,
        name: title,
        workspaceSlug: entity.workspaceSlug,
        cleanup: () => meetings.cleanupOwnedEntities(owned),
      });
    };

    try {
      await meetings.gotoHome();
      await assertSandbox();
      const capture = await meetings.createOwnedManualTranscript({
        title,
        transcriptText: MANUAL_TRANSCRIPT,
        register,
      });
      expectHealthyCapture(capture);
      const entity = owned[0];
      expect(entity).toBeDefined();
      if (!entity) throw new Error('MEETING_RESPONSE_ID_MISSING');

      await expect(meetings.manualDialog).toBeHidden();
      await expect(meetings.meetingLink(title)).toHaveCount(1);
      await meetings.openTranscript(title);
      await expect(meetings.transcriptTitle(title)).toBeVisible();
      for (const line of MANUAL_TRANSCRIPT_LINES) {
        await expect(page.getByText(line, { exact: false })).toHaveCount(1);
      }

      const detail = await page.request.get(MEETING_PATHS.detailApi(entity.id));
      expect(detail.status()).toBe(200);
      const body = await detail.json() as Record<string, unknown>;
      expect(body.id).toBe(entity.id);
      expect(body.title).toBe(title);
      expect(body.status).toBe('ready');
      expect(body.transcript_text === MANUAL_TRANSCRIPT).toBe(true);

      await meetings.gotoHome();
      await assertSandbox();
      await expect(meetings.meetingLink(title)).toHaveCount(1);
      await meetings.openTranscript(title);
      for (const line of MANUAL_TRANSCRIPT_LINES) {
        await expect(page.getByText(line, { exact: false })).toHaveCount(1);
      }
    } finally {
      if (owned.length > 0) await meetings.cleanupOwnedEntities(owned);
    }
  });

  test('shows actionable feedback after audio asset reservation returns 504 (NUANU-10)', async ({
    page,
    assertSandbox,
    ownedName,
  }) => {
    test.setTimeout(90_000);
    const payload = {
      name: 'qa-e2e-silent-250ms.mp3',
      mimeType: 'audio/mpeg',
      buffer: Buffer.from(SYNTHETIC_SILENT_MP3_250MS_BASE64, 'base64'),
    };
    expect(payload.buffer).toHaveLength(1_254);
    const meetings = new MeetingsPage(page, assertSandbox);
    const title = ownedName('audio upload 504 feedback');
    const unexpectedMutations: Array<{
      method: string;
      path: string;
    }> = [];
    let assetPostCount = 0;
    let releaseAssetResponse: () => void = () => {};
    const assetResponseGate = new Promise<void>((resolve) => {
      releaseAssetResponse = resolve;
    });
    let behaviorFailure: unknown | undefined;

    await meetings.gotoHome();
    await assertSandbox();

    const assetEndpoint = new URL(
      MEETING_PATHS.assetCollectionApi,
      page.url(),
    ).href;
    const mutationGuard = async (route: Route) => {
      const request = route.request();
      const method = request.method();
      if (method === 'POST' && request.url() === assetEndpoint) {
        assetPostCount += 1;
        await assetResponseGate;
        await route.fulfill({
          status: 504,
          contentType: 'text/html; charset=utf-8',
          body: '<html><title>504 Gateway Time-out</title></html>',
        });
        return;
      }
      if (MUTATION_METHODS.has(method)) {
        unexpectedMutations.push({
          method,
          path: new URL(request.url()).pathname,
        });
        await route.abort('blockedbyclient');
        return;
      }
      await route.fallback();
    };

    await page.route('**/*', mutationGuard);
    try {
      await meetings.openUploadAudio();
      await meetings.uploadTitleInput.fill(title);
      await meetings.uploadFileInput.setInputFiles(payload);

      const assetRequest = page.waitForRequest(
        (request) =>
          request.method() === 'POST' && request.url() === assetEndpoint,
        { timeout: 10_000 },
      );
      await meetings.uploadSubmitButton.click();
      await assetRequest;

      const assetResponse = page.waitForResponse(
        (response) =>
          response.request().method() === 'POST' &&
          response.url() === assetEndpoint,
        { timeout: 10_000 },
      );
      releaseAssetResponse();
      expect((await assetResponse).status()).toBe(504);

      const feedback = page
        .getByText(ACTIONABLE_AUDIO_UPLOAD_FAILURE)
        .first();
      try {
        await expect(feedback).toBeVisible({ timeout: 5_000 });
        const text = (await feedback.innerText()).replace(/\s+/gu, ' ').trim();
        expect(text.length).toBeGreaterThan(0);
        expect(text.length).toBeLessThanOrEqual(240);
        expect(RAW_OR_GENERIC_VALIDATION.test(text)).toBe(false);
        expect(RAW_AUDIO_UPLOAD_FAILURE.test(text)).toBe(false);
      } catch (error) {
        behaviorFailure = error;
      }
    } finally {
      releaseAssetResponse();
      if (await meetings.uploadDialog.isVisible()) {
        await meetings.cancelUploadAudio();
      }
      await page.unroute('**/*', mutationGuard);
    }

    expect(assetPostCount).toBe(1);
    expect(unexpectedMutations).toEqual([]);
    test.fail(
      true,
      'NUANU-10: audio asset 504 resets the form without actionable feedback.',
    );
    if (behaviorFailure !== undefined) throw behaviorFailure;
  });

  for (const invalidCase of INVALID_AUDIO_CASES) {
    test(`rejects ${invalidCase.label} with actionable validation and zero mutation`, async ({
      page,
      assertSandbox,
      ownedName,
    }) => {
      const meetings = new MeetingsPage(page, assertSandbox);
      const title = ownedName(invalidCase.suffix);
      await meetings.gotoHome();
      await assertSandbox();
      await meetings.openUploadAudio();
      await expect(meetings.uploadFileInput).toHaveAttribute('accept', 'audio/*');
      await meetings.uploadTitleInput.fill(title);
      const feedback = meetings.uploadValidationFeedback.filter({
        hasText: invalidCase.validation,
      }).first();
      let validationAssessment = {
        actionable: false,
        bounded: false,
        redacted: false,
      };

      const capture = await captureBlockedMeetingMutations({
        page,
        guardPatterns: meetings.mutationGuardPatterns(),
        action: async () => {
          await meetings.uploadFileInput.setInputFiles(invalidCase.file);
          await meetings.uploadSubmitButton.click();
          await feedback.waitFor({ state: 'visible' });
          const text = (await feedback.innerText()).trim();
          validationAssessment = {
            actionable: invalidCase.validation.test(text),
            bounded: text.length > 0 && text.length <= 240,
            redacted: !RAW_OR_GENERIC_VALIDATION.test(text),
          };
          await meetings.cancelUploadAudio();
        },
      });

      expect(capture.actionError).toBeUndefined();
      expect(capture.attemptCount).toBe(0);
      expect(validationAssessment).toEqual({
        actionable: true,
        bounded: true,
        redacted: true,
      });
      await expect(meetings.uploadDialog).toBeHidden();
      await expect(meetings.meetingLink(title)).toHaveCount(0);
    });
  }
