function variableStrings(value) {
  return [
    value.symptom,
    value.affectedObjectOrRoute,
    ...value.preconditions,
    ...value.steps,
    value.actual,
    value.expected,
    value.impact,
    value.evidenceSummary,
    ...value.attachments.flatMap((item) => [item.evidenceId, item.fileName, item.mediaType]),
    ...value.graph.nodeIds,
    ...(value.regression.testId === null ? [] : [value.regression.testId]),
    value.regression.design,
    value.deduplication.searchedAt,
  ];
}

function variableTextBytes(value) {
  return variableStrings(value).reduce((total, item) => total + Buffer.byteLength(item), 0);
}

function repeatedPattern(bytes, pattern = "&<>'\"") {
  return pattern.repeat(Math.ceil(bytes / pattern.length)).slice(0, bytes);
}

function findingAtAggregateBytes(target, pattern = "x") {
  const value = finding({
    symptom: "x",
    affectedObjectOrRoute: "/x",
    preconditions: ["x"],
    steps: ["x"],
    actual: "x",
    expected: "x",
    impact: "x",
    evidenceSummary: "x",
    graph: { graphDigest: SHA_C, nodeIds: ["nuanu-app:x"] },
    regression: { status: "AUTOMATED", testId: "NAPP-X-1", design: "x" },
  });
  let remaining = target - variableTextBytes(value);
  assert.ok(remaining >= 0);
  for (const key of ["actual", "expected", "impact", "evidenceSummary"]) {
    const add = Math.min(remaining, MAX_TEXT_BYTES - Buffer.byteLength(value[key]));
    value[key] += repeatedPattern(add, pattern);
    remaining -= add;
  }
  assert.equal(remaining, 0, "test fixture has enough bounded fields for aggregate bytes");
  assert.equal(variableTextBytes(value), target);
  return value;
}

function actionManifest(entry, {
  kind = "search_issues",
  issueId = remoteIdentity().id,
  actionId,
  batchDigest = exactBatchDigest(entry),
  proposedAt = NOW,
  descriptorCapturedAt,
  name,
  descriptionHtml,
  previousOutbox = { version: 1, entries: [entry] },
} = {}) {
  const codeOwnedCreatePayload = expectedCreatePayload(entry);
  const kindFields = {
    search_issues: {
      mutation: false,
      arguments: {
        workspaceSlug: "nuanu-app",
        projectIdentifier: "NAPP",
        fingerprint: entry.fingerprint,
        scopes: ["active", "recently_closed"],
        requireComplete: true,
      },
      expectedObservation: "search",
    },
    get_issue: {
      mutation: false,
      arguments: {
        workspaceSlug: "nuanu-app",
        projectIdentifier: "NAPP",
        issueId,
      },
      expectedObservation: "issue",
    },
    create_issue: {
      mutation: true,
      arguments: {
        project_identifier: "NAPP",
        name: name ?? codeOwnedCreatePayload.name,
        description_html: descriptionHtml ?? codeOwnedCreatePayload.description_html,
        state_name: "Bugs",
        label_names: ["Bug", entry.finding.platform].sort(),
        priority: entry.finding.priority,
      },
      expectedObservation: "mutation_ack",
    },
  }[kind];
  assert.ok(kindFields);
  const semantic = {
    version: 1,
    productId: "nuanu-app",
    fingerprint: entry.fingerprint,
    findingDigest: entry.findingDigest,
    batchDigest,
    proposedAt,
    expiresAt: new Date(Date.parse(proposedAt) + MAX_ACTION_AGE_MS).toISOString(),
    kind,
    mutation: kindFields.mutation,
    descriptorBinding: descriptorBinding(descriptorCapturedAt ?? proposedAt),
    arguments: kindFields.arguments,
    expectedObservation: kindFields.expectedObservation,
  };
  const derivedActionId = `napp-flow-action-v1-${digestJson({
    version: 1,
    outboxRevision: exactOutboxRevision(previousOutbox),
    action: semantic,
  })}`;
  const preceding = {
    version: semantic.version,
    productId: semantic.productId,
    actionId: actionId ?? derivedActionId,
    fingerprint: semantic.fingerprint,
    findingDigest: semantic.findingDigest,
    batchDigest: semantic.batchDigest,
    proposedAt: semantic.proposedAt,
    expiresAt: semantic.expiresAt,
    kind: semantic.kind,
    mutation: semantic.mutation,
    descriptorBinding: semantic.descriptorBinding,
    arguments: semantic.arguments,
    expectedObservation: semantic.expectedObservation,
  };
  return { ...preceding, actionDigest: `sha256:${digestJson(preceding)}` };
}

async function fixture(t, { ioOptions = {}, lockFs, journalFs, limits, randomUUID } = {}) {
  const createdRoot = await mkdtemp(join(tmpdir(), "napp-flow-outbox-"));
  const root = await realpath(createdRoot);
  t.after(() => rm(createdRoot, { recursive: true, force: true }));
  const io = createJsonIo({ privateRoot: root, ...ioOptions });
  const outboxPath = join(root, OUTBOX_RELATIVE_PATH);
  const lockPath = join(root, OUTBOX_LOCK_RELATIVE_PATH);
  const journalPath = join(root, OUTBOX_JOURNAL_RELATIVE_PATH);
  const snapshotPath = join(root, OUTBOX_SNAPSHOT_RELATIVE_PATH);
  const service = createFlowOutboxService({
    io, journalPath, snapshotPath, outboxPath, legacyLockPath: lockPath,
    journalFs, randomUUID, limits,
  });
  return { root, io, outboxPath, lockPath, journalPath, snapshotPath, service };
}

async function readJournalEvent(journalPath, sequence) {

async function readJournalEvents(journalPath) {
  const names = (await readdir(journalPath)).sort();
  const events = [];
  for (const name of names) {
    events.push(JSON.parse(await readFile(join(journalPath, name), "utf8")));
  }
  return events;
}

async function readJsonFile(path) {
  return JSON.parse(await readFile(path, "utf8"));
}

async function writeReadonlyJson(path, value) {
  await chmod(path, 0o600).catch(() => {});
  await writeFile(path, `${JSON.stringify(value)}\n`, { mode: 0o400 });
  await chmod(path, 0o400);
}

async function rewriteJournalChain(journalPath, mutate) {
  const names = (await readdir(journalPath)).sort();
  let previousEventDigest = null;
  let previousOutboxDigest = null;
  let currentOutbox = null;
  for (const [index, name] of names.entries()) {
    const path = join(journalPath, name);
    const event = await readJsonFile(path);
    if (typeof mutate === "function") {
      await mutate(event, index + 1, path);
    }
    event.sequence = index + 1;
    event.previousEventDigest = previousEventDigest;
    event.previousOutboxDigest = previousOutboxDigest;
    if (event.kind === "bootstrap_v1" || event.kind === "outbox_committed") {
      currentOutbox = structuredClone(event.payload.outbox);
    }
    event.nextOutboxDigest = `sha256:${digestJson(currentOutbox)}`;
    const { eventDigest: _ignored, ...preceding } = event;
    event.eventDigest = `sha256:${digestJson(preceding)}`;
    await writeReadonlyJson(path, event);
    previousEventDigest = event.eventDigest;
    previousOutboxDigest = event.nextOutboxDigest;
  }
}

function makeJournalFs({
  operations = [],
  onOpen,
  onLstat,
  onReaddir,
  onMkdir,
  onLink,
  onUnlink,
} = {}) {
  return {
    async open(...args) {
      operations.push(["open", args[0], args[1] ?? null, args[2] ?? null]);
      return onOpen ? onOpen(...args) : open(...args);
    },
    async lstat(...args) {
      operations.push(["lstat", args[0]]);
      return onLstat ? onLstat(...args) : lstat(...args);
    },
    async readdir(...args) {
      operations.push(["readdir", args[0]]);
      return onReaddir ? onReaddir(...args) : readdir(...args);
    },
    async mkdir(...args) {
      operations.push(["mkdir", args[0], args[1] ?? null]);
      return onMkdir ? onMkdir(...args) : mkdir(...args);
    },
    async link(...args) {
      operations.push(["link", args[0], args[1]]);
      return onLink ? onLink(...args) : link(...args);
    },
    async unlink(...args) {
      operations.push(["unlink", args[0]]);
      return onUnlink ? onUnlink(...args) : unlink(...args);
    },
  };
}

    },
    async write(...args) {
      return overrides.write ? overrides.write(...args) : handle.write(...args);
    },
    async sync(...args) {
      return overrides.sync ? overrides.sync(...args) : handle.sync(...args);
    },
    async chmod(...args) {
      return overrides.chmod ? overrides.chmod(...args) : handle.chmod(...args);
    },
    async stat(...args) {

    },
    async truncate(...args) {
      return overrides.truncate ? overrides.truncate(...args) : handle.truncate(...args);
    },
    async close(...args) {
      return overrides.close ? overrides.close(...args) : handle.close(...args);

  };
}

test("fingerprint is stable when object keys are reordered", () => {
  const input = {
    productId: "nuanu-app",
    symptom: "unknown app route returns http 200",
    affectedObjectOrRoute: "/qa-napp-not-found-probe",
    platform: "Desktop",
    runtimeIdentity: SHA_A,
    evidenceSignature: SHA_B,
  };
  const reordered = Object.fromEntries(Object.entries(input).reverse());
  const expected = "napp-v1-fa70a0b62405675a8edc909311e7830e8933654766573ba2db21bad0c0ddd866";
  assert.equal(fingerprintFinding(input), expected);
  assert.equal(fingerprintFinding(reordered), expected);
});

test("fingerprint changes when runtime or evidence signature changes", () => {
  const base = finding();
  assert.notEqual(
    fingerprintFinding(base),
    fingerprintFinding({ ...base, runtimeIdentity: shaFor("runtime-2") }),
  );
  assert.notEqual(
    fingerprintFinding(base),
    fingerprintFinding({ ...base, evidenceSignature: shaFor("evidence-2") }),
  );
});

      affectedObjectOrRoute: "/safe",

      runtimeIdentity: SHA_A,

          },

test("concurrent journal directory creators adopt one inode", async (t) => {
  const operations = [];
  let mkdirCalls = 0;
  let mkdirFailures = 0;
  const openedDirInodes = [];
  let signalFirstMkdir;
  let releaseFirstMkdir;
  let signalSecondMkdir;
  const firstMkdirReady = new Promise((resolve) => { signalFirstMkdir = resolve; });
  const continueFirstMkdir = new Promise((resolve) => { releaseFirstMkdir = resolve; });
  const secondMkdirReady = new Promise((resolve) => { signalSecondMkdir = resolve; });
  let base;
  const journalFs = makeJournalFs({
    operations,
    onOpen: async (path, flags, mode) => {
      const handle = await open(path, flags, mode);
      if (base && path === base.journalPath) {
        return wrapHandle(handle, {
          async stat(...args) {
            const statValue = await handle.stat(...args);
            openedDirInodes.push([Number(statValue.dev), Number(statValue.ino)]);
            return statValue;
          },
        });
      }
      return handle;
    },
    onMkdir: async (path, options) => {
      mkdirCalls += 1;
      if (mkdirCalls === 1) {
        signalFirstMkdir();
        await continueFirstMkdir;
        try {
          return await mkdir(path, options);
        } catch (error) {
          if (error?.code === "EEXIST") mkdirFailures += 1;
          throw error;
        }
      }
      signalSecondMkdir();
      try {
        return await mkdir(path, options);
      } catch (error) {
        if (error?.code === "EEXIST") mkdirFailures += 1;
        throw error;
      }
    },
  });
  base = await fixture(t, { journalFs, randomUUID: deterministicUuidFactory() });
  const barrierUuid = deterministicUuidFactory(100)();
  await writeFile(base.lockPath, `${JSON.stringify(barrierRecord(barrierUuid))}\n`, { mode: 0o600 });
  const second = createFlowOutboxService({
    io: base.io,
    journalPath: base.journalPath,
    snapshotPath: base.snapshotPath,
    outboxPath: base.outboxPath,
    legacyLockPath: base.lockPath,
    journalFs,
    randomUUID: deterministicUuidFactory(110),
  });
  const firstRun = enqueue(base.service, finding({ evidenceSignature: shaFor("mkdir-race-a") }), { ownerId: "mkdir-race-a" }).catch((error) => error);
  await firstMkdirReady;
  const secondRun = enqueue(second, finding({ evidenceSignature: shaFor("mkdir-race-b") }), { ownerId: "mkdir-race-b" }).catch((error) => error);
  await secondMkdirReady;
  releaseFirstMkdir();
  await Promise.all([firstRun, secondRun]);
  const journalStat = await lstat(base.journalPath);
  assert.equal(journalStat.isDirectory(), true);
  assert.equal(mkdirCalls, 2);
  assert.equal(mkdirFailures, 1);
  assert.equal(openedDirInodes.length >= 2, true);
  assert.deepEqual(openedDirInodes[0], openedDirInodes[1]);
  assert.deepEqual(openedDirInodes[0], [Number(journalStat.dev), Number(journalStat.ino)]);
});

    operations,

      if (String(path).includes(".flow-outbox-journal-candidate-v1-")) {
        candidatePath ??= path;
        return wrapHandle(handle, {
          async stat(...args) {
            candidateOps.push("stat");
            return handle.stat(...args);
          },
          async write(...args) {
            candidateOps.push("write");
            return handle.write(...args);
          },
          async sync(...args) {
            candidateOps.push("sync");
            return handle.sync(...args);
          },
          async chmod(...args) {
            candidateOps.push("chmod");
            return handle.chmod(...args);
          },
          async close(...args) {
            candidateOps.push("close");
            return handle.close(...args);
          },
        });
      }
      return handle;
    },
  });
  const base = await fixture(t, { journalFs, randomUUID: deterministicUuidFactory() });
  await enqueue(base.service, finding({ evidenceSignature: shaFor("oexcl-order") }));

    entry[1] === candidatePath &&
    entry[2] === (fsConstants.O_WRONLY | fsConstants.O_CREAT | fsConstants.O_EXCL | (fsConstants.O_NOFOLLOW ?? 0)));
  const candidateReadOpenIndex = operations.findIndex((entry, index) =>
    index > candidateCreateOpenIndex &&
    entry[0] === "open" &&
    entry[1] === candidatePath &&
    entry[2] === (fsConstants.O_RDONLY | (fsConstants.O_NOFOLLOW ?? 0)));
  const linkIndex = operations.findIndex((entry, index) =>
    index > candidateCreateOpenIndex &&
    entry[0] === "link" &&
    entry[1] === candidatePath);
  assert.deepEqual(candidateOps.slice(0, 6), ["stat", "write", "sync", "chmod", "sync", "close"]);
  assert.ok(candidateReadOpenIndex > candidateCreateOpenIndex);
  assert.ok(linkIndex > candidateReadOpenIndex);

  {
    let replacedPath = null;
    let replaced = false;
    let linkCalls = 0;
    const replacedFs = makeJournalFs({
      onOpen: async (path, flags, mode) => {
        if (
          String(path).includes(".flow-outbox-journal-candidate-v1-") &&
          flags === (fsConstants.O_WRONLY | fsConstants.O_CREAT | fsConstants.O_EXCL | (fsConstants.O_NOFOLLOW ?? 0))

          }

      },
      onLink: async (from, to) => {
        if (from === replacedPath) linkCalls += 1;
        return link(from, to);
      },
    });
    const replacedBase = await fixture(t, { journalFs: replacedFs, randomUUID: deterministicUuidFactory(40) });
    await assert.rejects(

        ownerId: "oexcl-prelink-replaced-owner",
      }),
      { code: "INVALID_OUTBOX_JOURNAL_CANDIDATE" },
    );
    assert.equal(linkCalls, 0);
  }
});

test("barrier hard-link reconciliation uses the literal oracle", async (t) => {
  const cases = [

    ["RETURNED_DIFFERENT_VALID", "DIFFERENT_VALID", "RETURNED", "OUTBOX_JOURNAL_PUBLICATION_INCONSISTENT"],

    ["EXDEV_DIFFERENT_VALID", "DIFFERENT_VALID", "EXDEV", "FULFILLED"],
    ["EXDEV_INVALID", "INVALID", "EXDEV", "OUTBOX_LEGACY_BUSY"],
    ["EXDEV_ABSENT", "ABSENT", "EXDEV", "OUTBOX_JOURNAL_PUBLICATION_UNSUPPORTED"],
  ];
  const results = {};
  for (const [label, destinationState, linkCode, expected] of cases) {
    const uuidStart = 1000 + Object.keys(results).length * 10;
    const expectedUuid = `00000000-0000-4000-8000-${String(uuidStart + 1).padStart(12, "0")}`;
    let linkCalls = 0;
    let rootPath = null;
    const operations = [];
    const handleOps = [];
    let linkedSameInode = false;
    const journalFs = makeJournalFs({
      operations,
      onOpen: async (path, flags, mode) => {
        const handle = await open(path, flags, mode);
        if (path === rootPath) {
          return wrapHandle(handle, {
            async sync(...args) {
              handleOps.push(["sync", path, operations.length]);
              return handle.sync(...args);
            },
          });
        }
        return handle;
      },
      onLink: async (from, to) => {
        if (String(to) !== base.lockPath) return link(from, to);
        linkCalls += 1;
        if (destinationState === "MATCHING") {
          await link(from, to);
          const [fromStat, toStat] = await Promise.all([
            lstat(from, { bigint: true }),
            lstat(to, { bigint: true }),

        }

          await writeFile(to, `${JSON.stringify(barrierRecord(`00000000-0000-4000-8000-${String(uuidStart + 9).padStart(12, "0")}`))}\n`, { mode: 0o600 });
        }
        else if (destinationState === "INVALID") {
          await writeFile(to, "{bad-barrier}\n", { mode: 0o600 });
        }
        if (linkCode === "RETURNED") {
          return undefined;
        }
        const error = new Error(`synthetic ${linkCode}`);
        error.code = linkCode;
        throw error;
      },
    });
    const base = await fixture(t, { journalFs, randomUUID: deterministicUuidFactory(uuidStart) });
    rootPath = base.root;
    const candidatePath = migrationBarrierCandidatePath(base.root, expectedUuid);
    results[label] = await outcomeCode(() => enqueue(base.service, finding({ evidenceSignature: shaFor(`barrier-oracle-${label}`) }), {
      ownerId: `barrier-oracle-case-${Object.keys(results).length + 1}`,

      "EEXIST_MATCHING",
      "EEXIST_DIFFERENT_VALID",
      "EIO_MATCHING",
      "EIO_DIFFERENT_VALID",
      "EIO_ABSENT",

      "EXDEV_DIFFERENT_VALID",

    } else {
      assert.equal(await readFile(candidatePath, "utf8").then(Boolean), true, `${label}:retained`);

      assert.equal((await readJsonFile(base.lockPath)).ownerId.startsWith("journal-v1-barrier-"), true);
    }
    if (destinationState === "MATCHING") {
      assert.equal(linkedSameInode, true, `${label}:matching-inode`);
    }
    if (label === "RETURNED_ABSENT") {
      const candidate = await readJsonFile(candidatePath);
      assert.equal(candidate.ownerId, `journal-v1-barrier-${expectedUuid}`);
    }

  }

test("strictly expired lease contenders publish exactly one new leaseId and fence", async (t) => {

  let signalContenderAcquired;
  let allowWinnerRelease;
  let releaseContenders;
  let firstContenderLabel = null;
  const heldAcquired = new Promise((resolve) => { signalHeldAcquired = resolve; });
  const contenderAcquired = new Promise((resolve) => { signalContenderAcquired = resolve; });
  const allowWinner = new Promise((resolve) => { allowWinnerRelease = resolve; });
  const contendersGate = new Promise((resolve) => { releaseContenders = resolve; });

  const held = base.service.withOutboxTransaction(transactionInput(async function (transaction) {
    assert.equal(transaction.expiresAt, "2026-08-26T00:00:01.000Z");
    signalHeldAcquired();

    await assert.rejects(
      () => transaction.commit(transaction.snapshot),
      { code: "OUTBOX_LEASE_FENCED" },
    );
  }, {
    ownerId: "expired-holder",
    leaseMs: 1_000,
  }));
  await heldAcquired;

  const captureOutcome = (promise) => promise.then(
    (value) => ({ status: "fulfilled", value }),
    (reason) => ({ status: "rejected", reason }),
  );
  const contenderOperation = (label) => async () => {
    if (firstContenderLabel === null) {
      firstContenderLabel = label;
      signalContenderAcquired();
    }
    await allowWinner;
    return label;
  };

  const winner = captureOutcome((async () => {
    await contendersGate;
    return base.service.withOutboxTransaction({
      ownerId: "expired-contender-winner",
      now: () => new Date("2026-08-26T00:00:01.001Z"),
      leaseMs: 30_000,
      operation: contenderOperation("expired-contender-winner"),
    });

    await contendersGate;
    return base.service.withOutboxTransaction({
      ownerId: "expired-contender-loser",
      now: () => new Date("2026-08-26T00:00:01.001Z"),
      leaseMs: 30_000,
      operation: contenderOperation("expired-contender-loser"),
    });
  })());
  releaseContenders();
  await contenderAcquired;

  await held;
  allowWinnerRelease();
  const settled = await Promise.all([winner, loser]);
  const fulfilled = settled.filter((item) => item.status === "fulfilled");
  const rejected = settled.filter((item) => item.status === "rejected");
  assert.equal(fulfilled.length, 1);
  assert.equal(rejected.length, 1);
  assert.match(fulfilled[0].value, /^expired-contender-(winner|loser)$/);

test("expired old writer paused before candidate creation cannot erase a newer owner commit", async (t) => {
  const rows = [
    { mode: "success", expectedCode: "OUTBOX_LEASE_FENCED" },
    { mode: "cleanup_pre_absence_failure", expectedCode: "OUTBOX_JOURNAL_ORPHAN_CLEANUP_FAILED" },
    { mode: "cleanup_post_absence_fsync_failure", expectedCode: "OUTBOX_JOURNAL_CANDIDATE_CLEANUP_DURABILITY_UNKNOWN" },
  ];

  for (const [index, row] of rows.entries()) {
    const operations = [];
    const handleOps = [];
    let signalOldCommitReady;
    let signalContinueOldCommit;
    const oldCommitReady = new Promise((resolve) => { signalOldCommitReady = resolve; });
    const continueOldCommit = new Promise((resolve) => { signalContinueOldCommit = resolve; });
    let privateRootPath = null;
    let oldCommitCandidatePath = null;
    let oldLinkCalls = 0;
    let journalCandidateOpens = 0;
    let armCleanupRootSyncFailure = false;
    const candidateBasenamePattern = /^\.flow-outbox-journal-candidate-v1-[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-8[0-9a-f]{3}-[0-9a-f]{12}\.json$/;
    const journalFs = makeJournalFs({
      operations,
      onOpen: async (path, flags, mode) => {
        if (
          typeof path === "string" &&
          privateRootPath &&
          path.startsWith(`${privateRootPath}/`) &&
          candidateBasenamePattern.test(basename(path)) &&
          flags === (fsConstants.O_WRONLY | fsConstants.O_CREAT | fsConstants.O_EXCL | (fsConstants.O_NOFOLLOW ?? 0))
        ) {
          journalCandidateOpens += 1;
          if (journalCandidateOpens === 3 && oldCommitCandidatePath === null) {
            oldCommitCandidatePath = path;
            signalOldCommitReady();
            await continueOldCommit;
          }
          return open(path, flags, mode);
        }
        const handle = await open(path, flags, mode);
        if (path === privateRootPath) {
          return wrapHandle(handle, {
            async stat(...args) {
              const statValue = await handle.stat(...args);
              handleOps.push(["stat", path, Number(statValue.dev), Number(statValue.ino)]);
              return statValue;
            },
            async sync(...args) {
              handleOps.push(["sync", path]);
              if (armCleanupRootSyncFailure) {
                armCleanupRootSyncFailure = false;
                const error = new Error("synthetic cleanup root fsync failure");
                error.code = "EIO";
                throw error;
              }
              return handle.sync(...args);
            },
            async close(...args) {
              handleOps.push(["close", path]);
              return handle.close(...args);
            },
          });
        }
        return handle;
      },
      onLink: async (from, to) => {
        if (from === oldCommitCandidatePath && String(to).endsWith(".json")) {
          oldLinkCalls += 1;
        }
        return link(from, to);
      },
      onUnlink: async (path) => {
        if (path === oldCommitCandidatePath && row.mode === "cleanup_pre_absence_failure") {
          const error = new Error("synthetic cleanup failure");
          error.code = "EIO";
          throw error;
        }
        return unlink(path);
      },
    });
    const raced = await fixture(t, { journalFs, randomUUID: deterministicUuidFactory(200 + (index * 20)) });
    privateRootPath = raced.root;
    const oldFinding = finding({ evidenceSignature: shaFor(`old-in-flight-writer-${row.mode}`) });
    const newFinding = finding({ evidenceSignature: shaFor(`new-fenced-writer-${row.mode}`) });

    const oldAttempt = enqueue(raced.service, oldFinding, {
      ownerId: "expiring-old-writer",
      now: () => new Date(NOW),
      leaseMs: 1_000,
    });
    await oldCommitReady;
    await assert.rejects(() => stat(oldCommitCandidatePath), { code: "ENOENT" }, `${row.mode}:pre-open`);
    await enqueue(raced.service, newFinding, {
      ownerId: "new-fenced-writer",
      now: () => new Date("2026-08-26T00:00:01.001Z"),
      leaseMs: 30_000,
    });
    armCleanupRootSyncFailure = row.mode === "cleanup_post_absence_fsync_failure";
    signalContinueOldCommit();
    await assert.rejects(oldAttempt, { code: row.expectedCode }, row.mode);

    if (row.mode === "cleanup_pre_absence_failure") {
      const residue = await lstat(oldCommitCandidatePath);
      assert.equal(residue.isFile(), true, row.mode);
    } else {
      await assert.rejects(() => stat(oldCommitCandidatePath), { code: "ENOENT" }, row.mode);
    }

    assert.match(basename(oldCommitCandidatePath), candidateBasenamePattern, row.mode);
    assert.equal(oldLinkCalls, 0, row.mode);
    assert.deepEqual(
      (await raced.service.readOutbox()).entries.map(({ fingerprint }) => fingerprint),
      [fingerprintFinding(newFinding)],
      row.mode,
    );

    const oldFingerprint = fingerprintFinding(oldFinding);
    const events = await readJournalEvents(raced.journalPath);
    const oldLeaseId = events.find((event) =>
      event.kind === "lease_acquired" && event.payload.ownerId === "expiring-old-writer")?.leaseId;
    assert.equal(
      events.some((event) =>
        event.kind === "outbox_committed" &&
        event.payload.outbox.entries.some((entry) => entry.fingerprint === oldFingerprint)),
      false,
      row.mode,
    );
    assert.equal(
      events.some((event) => event.kind === "lease_released" && event.leaseId === oldLeaseId),
      false,
      row.mode,
    );

    if (row.mode !== "cleanup_pre_absence_failure") {
      const unlinkIndex = operations.findIndex((entry) => entry[0] === "unlink" && entry[1] === oldCommitCandidatePath);
      const reattestIndex = operations.findIndex((entry) => entry[0] === "lstat" && entry[1] === oldCommitCandidatePath);
      const rootOpenIndex = operations.findIndex((entry, itemIndex) =>
        itemIndex > unlinkIndex && entry[0] === "open" && entry[1] === privateRootPath);
      assert.ok(reattestIndex >= 0, row.mode);
      assert.ok(unlinkIndex > reattestIndex, row.mode);
      assert.ok(rootOpenIndex > unlinkIndex, row.mode);
      const cleanupRootOps = handleOps
        .filter((entry) => entry[1] === privateRootPath)
        .slice(handleOps.findIndex((entry) => entry[0] === "stat" && entry[1] === privateRootPath), undefined)
        .map((entry) => entry[0]);
      assert.deepEqual(
        cleanupRootOps.slice(-3),
        ["stat", "sync", "close"],
        row.mode,
      );
    }
  }
});

          updatedAt: "2026-08-26T00:00:01.001Z",
        }],
      }),
    }),
    { code: "OUTBOX_LEASE_EXPIRED" },
  );
  const releaseCountBefore = (await readJournalEvents(base.journalPath))
    .filter((event) => event.kind === "lease_released")
    .length;

  let signalOldCommitReady;
  let signalContinueOldCommit;
  const oldCommitReady = new Promise((resolve) => { signalOldCommitReady = resolve; });
  const continueOldCommit = new Promise((resolve) => { signalContinueOldCommit = resolve; });
  let gateOldCommit = true;
  let oldCommitCandidatePath = null;
  let journalCandidateOpens = 0;
  const racedJournalFs = makeJournalFs({
    onOpen: async (path, flags, mode) => {
      if (String(path).includes(".flow-outbox-journal-candidate-v1-")) {
        journalCandidateOpens += 1;
        if (journalCandidateOpens === 3 && gateOldCommit) {
          gateOldCommit = false;
          oldCommitCandidatePath = path;
          signalOldCommitReady();
          await continueOldCommit;
        }
      }
      return open(path, flags, mode);
    },
  });
  const raced = await fixture(t, { journalFs: racedJournalFs, randomUUID: deterministicUuidFactory(300) });
  const oldAttempt = enqueue(raced.service, finding({ evidenceSignature: shaFor("expired-cleanup-old") }), {
    ownerId: "expired-cleanup-old-owner",
    leaseMs: 1_000,
  });
  await oldCommitReady;
  await raced.service.withOutboxTransaction({
    ownerId: "expired-cleanup-new-owner",
    now: () => new Date("2026-08-26T00:00:01.001Z"),
    leaseMs: 30_000,
    operation: async () => {},
  });
  signalContinueOldCommit();
  await assert.rejects(oldAttempt, { code: "OUTBOX_LEASE_FENCED" });
  await assert.rejects(() => stat(oldCommitCandidatePath), { code: "ENOENT" });
  const releaseCountAfter = (await readJournalEvents(raced.journalPath))
    .filter((event) => event.kind === "lease_released")
    .length;
  assert.equal(releaseCountBefore, 1);
  assert.equal(releaseCountAfter, 1);
});

test("invalid transition and generic recovery cannot clear a durable create receipt", async (t) => {
  const { io, outboxPath, service } = await fixture(t);
  const base = entryFor(260);
  const receipt = createReceipt();
  const failed = {
    ...base,
    state: "failed",
    attempts: 1,
    mutationReceipt: receipt,
    failure: {
      code: "FLOW_READ_FAILED",
      failedAt: NOW,
      recoveryClass: "RETRY_SAFE_READ",
    },
  };
  await io.writeJsonAtomic(outboxPath, { version: 1, entries: [failed] });
  const before = await readFile(outboxPath);

  await assert.rejects(
    () => service.recoverOutbox({
      fingerprint: base.fingerprint,
      ownerId: OWNER,
      now: () => new Date(NOW),
      leaseMs: 30_000,
    }),
    { code: "OUTBOX_RECOVERY_BLOCKED" },
  );
  await assert.rejects(
    () => service.withOutboxTransaction(transactionInput((transaction) =>
      transaction.commit({
        version: 1,
        entries: [{
          ...transaction.snapshot.entries[0],
          state: "pending_sync",
          mutationReceipt: null,
          failure: null,
        }],
      }))),
    { code: "INVALID_OUTBOX_TRANSITION" },
  );
  assert.deepEqual(await readFile(outboxPath), before);
});

test("V1 never compacts deletes truncates renames or replaces a published event", async (t) => {
  const operations = [];
  const journalFs = {
    async open(...args) {
      operations.push(["open", args[0]]);
      return open(...args);
    },
    async lstat(...args) {
      operations.push(["lstat", args[0]]);
      return lstat(...args);
    },
    async readdir(...args) {
      operations.push(["readdir", args[0]]);
      return readdir(...args);
    },
    async mkdir(...args) {
      operations.push(["mkdir", args[0]]);
      return mkdir(...args);
    },
    async link(from, to) {
      operations.push(["link", from, to]);
      return link(from, to);
    },
    async unlink(path) {
      operations.push(["unlink", path]);
      return unlink(path);
    },
  };
  const base = await fixture(t, { journalFs, randomUUID: deterministicUuidFactory(600) });
  await enqueue(base.service, finding({ evidenceSignature: shaFor("no-compaction-1") }));
  await enqueue(base.service, finding({ evidenceSignature: shaFor("no-compaction-2") }), {
    ownerId: "no-compaction-2",
  });
  assert.equal((await readJournalEvents(base.journalPath)).length, 5);
  assert.equal(
    operations.some(([kind, path]) =>
      (kind === "unlink") && typeof path === "string" && /^[0-9]{20}\.json$/.test(path.split("/").at(-1))),
    false,
  );
});
